   157→function computeBreach(
   158→  elapsedHours: number,
   159→  targetHours: number,
   160→): "none" | "due_soon" | "breached" | "hard" {
   161→  if (elapsedHours >= targetHours * 2) return "hard";
   162→  if (elapsedHours >= targetHours) return "breached";
   163→  if (targetHours > 0 && elapsedHours / targetHours >= 0.75) return "due_soon";
   164→  return "none";
   165→}

   197→/**
   198→ * Build a synthetic payload view from the schema + employee.
   199→ * In production this would come from the request row itself; here we
   200→ * synthesise a plausible payload using prefillFrom hints so the
   201→ * service owner sees the same shape of data they would in production.
   202→ */
   203→function buildPayloadView(schema: IntakeSchema, employee: Employee): Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> {
   204→  const out: Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> = {};
   205→  for (const field of schema.fields) {
   206→    let value = "";
   207→    let prefilled = false;
   208→    if (field.prefillFrom === "employee.location") {
   209→      value = employee.location;
   210→      prefilled = true;
   211→    } else if (field.prefillFrom === "employee.stage") {
   212→      value = employee.stage;
   213→      prefilled = true;
   214→    } else if (field.prefillFrom === "employee.startDate") {
   215→      value = employee.startDate.slice(0, 10);
   216→      prefilled = true;
   217→    } else if (field.prefillFrom === "employee.role") {
   218→      value = employee.role;
   219→      prefilled = true;
   220→    } else if (field.prefillFrom === "step.key") {
   221→      value = "—";
   222→      prefilled = true;
   223→    } else if (field.prefillFrom === "step.dueAt") {
   224→      value = "—";
   225→      prefilled = true;
   226→    } else {
   227→      // Synthesise plausible free-text values
   228→      if (field.type === "checkbox") {
   229→        value = field.required ? "Yes" : "—";
   230→      } else if (field.type === "select" && field.options && field.options.length > 0) {
   231→        value = field.options[0].label;
   232→      } else if (field.type === "number") {
   233→        value = "1";
   234→      } else if (field.type === "textarea") {
   235→        value = "—";
   236→      } else if (field.type === "date") {
   237→        value = "2024-10-15";
   238→      } else {
   239→        value = "—";
   240→      }
   241→    }
   242→    out[field.key] = { label: field.label, labelAr: field.labelAr, value, prefilled };
   243→  }
   244→  return out;
   245→}
   246→
   247→function fieldTypeDef(t: IntakeField["type"]): string {
   248→  switch (t) {
   249→    case "select": return "select";
   250→    case "date": return "date";
   251→    case "checkbox": return "checkbox";
   252→    case "textarea": return "textarea";
   253→    case "number": return "number";
   254→    default: return "text";
   255→  }
   256→}
