   236→/* ============================================================
   237→   Component
   238→   ============================================================ */
   239→export function ManagerScreen() {
   240→  const locale = useAppStore((s) => s.locale);
   241→  const view = useAppStore((s) => s.managerView);
   242→  const dir = locale === "ar" ? "rtl" : "ltr";
   243→
   244→  const [items, setItems] = useState<CohortItem[] | null>(null);
   245→  const [err, setErr] = useState<string | null>(null);
   246→  const [loading, setLoading] = useState(true);
   247→
   248→  const fetchCohort = useCallback(async () => {
   249→    setLoading(true);
   250→    setErr(null);
   251→    try {
   252→      const res = await fetch("/api/v1/manager/cohort?managerId=MGR-1", { cache: "no-store" });
   253→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   254→      const json: ApiEnvelope<{ managerId: string; items: CohortItem[] }> = await res.json();
   255→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   256→      setItems(json.data.items);
   257→    } catch (e) {
   258→      setErr(e instanceof Error ? e.message : "Network error");
   259→    } finally {
   260→      setLoading(false);
   261→    }
   262→  }, []);
   263→
   264→  useEffect(() => { fetchCohort(); }, [fetchCohort]);
   265→
   266→  // Aggregate metrics
   267→  const stats = useMemo(() => {
   268→    if (!items) return { joiners: 0, waitingOnMe: 0, avgCompletion: 0 };
   269→    const joiners = items.length;
   270→    let waiting = 0;
   271→    let sumCompletion = 0;
   272→    for (const it of items) {
   273→      sumCompletion += it.aggregateCompletion;
   274→      for (const s of it.ownedSteps) {
   275→        if (s.status === "waiting_owner") waiting += 1;
   276→      }
   277→    }
   278→    return {
   279→      joiners,
   280→      waitingOnMe: waiting,
   281→      avgCompletion: joiners > 0 ? Math.round(sumCompletion / joiners) : 0,
   282→    };
   283→  }, [items]);
   284→
   285→  // Flatten owned steps across cohort (for "my obligations")
   286→  const allObligations = useMemo(() => {
   287→    if (!items) return [] as { employee: CohortItem; step: OwnedStep }[];
   288→    const out: { employee: CohortItem; step: OwnedStep }[] = [];
   289→    for (const it of items) {
   290→      for (const s of it.ownedSteps) {
   291→        out.push({ employee: it, step: s });
   292→      }
   293→    }
   294→    // Sort: waiting_owner first, then by dueAt asc
   295→    out.sort((a, b) => {
   296→      const aWait = a.step.status === "waiting_owner" ? 0 : 1;
   297→      const bWait = b.step.status === "waiting_owner" ? 0 : 1;
   298→      if (aWait !== bWait) return aWait - bWait;
   299→      const aDue = a.step.dueAt ? new Date(a.step.dueAt).getTime() : Number.MAX_SAFE_INTEGER;
   300→      const bDue = b.step.dueAt ? new Date(b.step.dueAt).getTime() : Number.MAX_SAFE_INTEGER;
   301→      return aDue - bDue;
   302→    });
   303→    return out;
   304→  }, [items]);
   305→
   306→  return (
   307→    <div
   308→      dir={dir}
   309→      className="max-w-[1180px] mx-auto px-4 sm:px-6 lg:px-8 py-6 lg:py-10"
   310→    >
   311→      <header className="flex items-center justify-between mb-6 flex-wrap gap-3">
   312→        <div>
   313→          <div className="flex items-center gap-2 text-[11px] font-semibold uppercase tracking-[0.18em]" style={{ color: "var(--brand-400)" }}>
   314→            <Users className="w-3.5 h-3.5" strokeWidth={1.75} />
   315→            <span>{locale === "ar" ? "وحدة المدير" : "Manager console"}</span>
   316→          </div>
   317→          <h1 className="mt-2 text-[26px] sm:text-[30px] font-semibold tracking-tight" style={{ color: "var(--text-hi)" }}>
   318→            {locale === "ar" ? "التزاماتك تجاه فريقك" : "Your obligations to your team"}
   319→          </h1>
   320→        </div>
   321→        <ManagerTabs />
   322→      </header>
   323→
   324→      <PrivacyBanner locale={locale} />
   325→
   326→      {loading ? (
   327→        <ManagerSkeleton />
   328→      ) : err ? (
   329→        <ErrorBanner message={err} onRetry={fetchCohort} />
   330→      ) : !items ? null : items.length === 0 ? (
   331→        <div className="ast-card p-10 flex flex-col items-center text-center">
   332→          <div
   333→            className="flex items-center justify-center w-14 h-14 rounded-full mb-4"
   334→            style={{ background: "var(--surface-2)", color: "var(--text-mid)" }}
   335→          >
   336→            <Inbox className="w-6 h-6" strokeWidth={1.75} />
   337→          </div>
   338→          <p className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   339→            {locale === "ar" ? "لا شيء بانتظارك." : "Nothing waiting on you."}
   340→          </p>
   341→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   342→            {locale === "ar"
   343→              ? "عندما يصل موظف جديد أو ينتقل أحد، ستظهر التزاماتك هنا."
   344→              : "When a joiner or mover lands on your team, your obligations will appear here."}
   345→          </p>
   346→        </div>
   347→      ) : (
   348→        <>
   349→          {/* Hero numbers */}
   350→          <section className="grid grid-cols-1 sm:grid-cols-3 gap-3 mb-5">
   351→            {/* Waiting on me — sand-gold (primary) */}
   352→            <div
   353→              className="ast-card p-5 relative overflow-hidden"
   354→              style={{
   355→                borderColor: "rgba(201,162,39,0.40)",
   356→                background: "linear-gradient(135deg, rgba(201,162,39,0.10) 0%, var(--surface-1) 60%)",
   357→              }}
   358→            >
   359→              <div
   360→                className="absolute top-0 left-0 right-0 h-[3px]"
   361→                style={{ background: "var(--accent-500)" }}
   362→              />
   363→              <p className="text-[11px] uppercase tracking-[0.14em] font-medium" style={{ color: "var(--accent-500)" }}>
   364→                {locale === "ar" ? "بانتظارك" : "Waiting on you"}
   365→              </p>
   366→              <p className="text-[44px] leading-none font-semibold mt-2 tabular ast-brass">
   367→                {stats.waitingOnMe}
   368→              </p>
   369→              <p className="text-[12px] mt-2" style={{ color: "var(--text-mid)" }}>
   370→                {locale === "ar"
   371→                  ? `خطوة تنتظر إكمالك عبر ${stats.joiners} موظف`
   372→                  : `step${stats.waitingOnMe === 1 ? "" : "s"} you own across ${stats.joiners} joiner${stats.joiners === 1 ? "" : "s"}`}
   373→              </p>
   374→            </div>
   375→
   376→            {/* Joiners count */}
   377→            <div className="ast-card p-5">
   378→              <p className="text-[11px] uppercase tracking-[0.14em] font-medium" style={{ color: "var(--text-lo)" }}>
   379→                {locale === "ar" ? "المنضمّون" : "Joiners"}
   380→              </p>

   382→                {stats.joiners}
   383→              </p>
   384→              <p className="text-[12px] mt-2" style={{ color: "var(--text-mid)" }}>
   385→                {locale === "ar" ? "في رحلتك الحالية" : "in your current cohort"}

   387→            </div>

   570→function ObligationRow({
   571→  employee,
   572→  step,
   573→}: {
   574→  employee: CohortItem;
   575→  step: OwnedStep;
   576→}) {
   577→  const locale = useAppStore((s) => s.locale);
   578→  const [completing, setCompleting] = useState(false);
   579→  const [completed, setCompleted] = useState(false);
   580→
   581→  const isWaitingOwner = step.status === "waiting_owner";
   582→  const due = step.dueAt ? new Date(step.dueAt) : null;
   583→  const dueDistance = due ? formatDistanceStrict(due, new Date("2024-10-19T10:00:00Z"), { addSuffix: false }) : null;
   584→  const overdue = due ? due.getTime() < Date.now() : false;
   585→
   586→  const handleComplete = useCallback(async () => {
   587→    setCompleting(true);
   588→    // No related request ref is returned by the cohort API (privacy boundary),
   589→    // so for the prototype we simulate completion with a success toast.
   590→    await new Promise((r) => setTimeout(r, 380));
   591→    setCompleting(false);
   592→    setCompleted(true);
   593→    toast.success("Step complete", {
   594→      description: `${step.label} · ${employee.employeeName}`,

   599→    <motion.li
   600→      initial={false}
   601→      className={cn("ast-card p-4 flex items-center gap-3 flex-wrap", completed && "opacity-70")}
   602→      style={{
   603→        borderColor: isWaitingOwner && !completed ? "rgba(201,162,39,0.32)" : "var(--border)",
   604→      }}
   605→    >
   606→      {/* Status glyph */}
   607→      <div
   608→        className="flex items-center justify-center w-9 h-9 rounded-full shrink-0"
   609→        style={{
   610→          background: completed
   611→            ? "rgba(14,159,110,0.12)"
   612→            : isWaitingOwner
   613→            ? "rgba(201,162,39,0.12)"
   614→            : "var(--surface-2)",
   615→          color: completed
   616→            ? "var(--status-steady)"
   617→            : isWaitingOwner
   618→            ? "var(--accent-500)"
   619→            : "var(--text-mid)",
   620→        }}

   623→          <CheckCircle2 className="w-4 h-4" strokeWidth={1.75} />

   625→          <Hourglass className="w-4 h-4" strokeWidth={1.75} />
   626→        ) : (
   627→          <Clock className="w-4 h-4" strokeWidth={1.75} />
   628→        )}
   629→      </div>
   630→
   631→      {/* Step + employee */}
   632→      <div className="flex-1 min-w-0">
   633→        <p className="text-[14px] font-medium" style={{ color: "var(--text-hi)" }}>
   634→          {step.label}
   635→          {completed && (
   636→            <span className="ml-2 text-[11px] font-medium" style={{ color: "var(--status-steady)" }}>
   637→              {locale === "ar" ? "تم" : "done"}
   638→            </span>
   639→          )}
   640→        </p>
   641→        <p className="text-[12px] mt-0.5" style={{ color: "var(--text-mid)" }}>
   642→          {locale === "ar" ? "لـ" : "for"} <span className="font-medium" style={{ color: "var(--text-hi)" }}>{employee.employeeName}</span>
   643→          {due && !completed && (
   644→            <>
   645→              <span className="mx-1.5">·</span>
   646→              <span
   647→                className="tabular inline-flex items-center gap-1"
   648→                style={{ color: overdue ? "var(--status-friction)" : "var(--text-lo)" }}
   649→              >
   650→                <Clock className="w-3 h-3" strokeWidth={1.75} />

   652→                <span style={{ color: "var(--text-lo)" }}>·</span>

   660→      </div>
   661→
   662→      {/* Status chip + action */}

   667→            onClick={handleComplete}
   668→            disabled={completing}
   669→            className="ast-focus inline-flex items-center gap-1.5 h-9 px-3 rounded-full text-[12px] font-medium transition-colors disabled:opacity-50"

   675→          >
   676→            <CheckCircle2 className="w-3.5 h-3.5" strokeWidth={1.75} />

   688→   30-60-90 outline card (PRD §11-207)
   689→   - Collapsible per-joiner card titled "30-60-90: {name}"
   690→   - Three columns: 30 / 60 / 90 days — each with editable goals
   691→   - Buddy assignment text input
   692→   - Save button → success toast (local state only, no backend)
   693→   ============================================================ */
   694→
   695→type OutlinePhase = "day30" | "day60" | "day90";
   696→
   697→function Outline3090Card({
   698→  item,
   699→  locale,
   700→}: {
   701→  item: CohortItem;
   702→  locale: "en" | "ar";
   703→}) {
   704→  const [open, setOpen] = useState(false);
   705→  const [outline, setOutline] = useState<Outline3090>(() =>
   706→    defaultOutline(item.journeyType),
   707→  );
   708→  const [saving, setSaving] = useState(false);
   709→  const [saved, setSaved] = useState(false);
   710→
   711→  const updateGoal = (phase: OutlinePhase, idx: number, value: string) => {
   712→    setOutline((prev) => {
   713→      const next: Outline3090 = { ...prev, [phase]: [...prev[phase]] };
   714→      next[phase][idx] = value;
   715→      return next;
   716→    });
   717→    // Mark as unsaved the moment any field is edited after a save.
   718→    if (saved) setSaved(false);
   719→  };
   720→
   721→  const updateBuddy = (value: string) => {
   722→    setOutline((prev) => ({ ...prev, buddy: value }));
   723→    if (saved) setSaved(false);
   724→  };
   725→
   726→  const handleSave = () => {
   727→    setSaving(true);
   728→    // Simulate a save round-trip — local state only (no backend).
   729→    setTimeout(() => {
   730→      setSaving(false);
   731→      setSaved(true);
   732→      toast.success(
   733→        locale === "ar" ? "تم حفظ المخطط" : "Outline saved",
   734→        {
   735→          description:
   736→            locale === "ar"
   737→              ? `مخطط 30-60-90 لـ ${item.employeeName}`
   738→              : `30-60-90 outline for ${item.employeeName}`,
   739→        },
   740→      );
   741→    }, 420);
   742→  };
   743→
   744→  const journeyLabel =
   745→    item.journeyType === "onboarding"
   746→      ? locale === "ar"
   747→        ? "التحاق"
   748→        : "Onboarding"
   749→      : locale === "ar"
   750→        ? "تنقل"
   751→        : "Mobility";
   752→
   753→  return (
   754→    <li className="ast-card overflow-hidden">
   755→      {/* Header — click to collapse / expand */}
   756→      <button
   757→        type="button"
   758→        onClick={() => setOpen((v) => !v)}
   759→        aria-expanded={open}
   760→        className="ast-focus w-full flex items-center gap-3 p-4 sm:p-5 text-left"
   761→      >
   762→        <div
   763→          className="flex items-center justify-center w-9 h-9 rounded-full shrink-0"
   764→          style={{
   765→            background: "rgba(22,154,165,0.10)",
   766→            color: "var(--brand-400)",
   767→          }}
   768→          aria-hidden="true"
   769→        >
   770→          <ClipboardList className="w-4 h-4" strokeWidth={1.75} />
   771→        </div>
   772→        <div className="flex-1 min-w-0">
   773→          <p className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   774→            {locale === "ar" ? "مخطط 30-60-90: " : "30-60-90: "}
   775→            {item.employeeName}
   776→          </p>
   777→          <p className="text-[12px] mt-0.5" style={{ color: "var(--text-mid)" }}>
   778→            {journeyLabel}
   779→            <span className="mx-1.5">·</span>
   780→            {stageLabel(item.stage, locale)}
   781→            <span className="mx-1.5">·</span>
   782→            <span className="tabular">
   783→              {locale === "ar" ? `يوم ${item.dayInStage}` : `Day ${item.dayInStage}`}
   784→            </span>
   785→          </p>
   786→        </div>
   787→        {saved && (
   788→          <span
   789→            className="ast-chip shrink-0"
   790→            style={{
   791→              color: "var(--status-steady)",
   792→              background: "rgba(14,159,110,0.10)",
   793→              borderColor: "rgba(14,159,110,0.30)",
   794→            }}
   795→          >
   796→            <CheckCircle2 className="w-3 h-3" strokeWidth={1.75} />
   797→            {locale === "ar" ? "محفوظ" : "Saved"}
   798→          </span>
   799→        )}
   800→        {open ? (
   801→          <ChevronUp className="w-4 h-4 shrink-0" strokeWidth={1.75} style={{ color: "var(--text-lo)" }} />
   802→        ) : (
   803→          <ChevronDown className="w-4 h-4 shrink-0" strokeWidth={1.75} style={{ color: "var(--text-lo)" }} />
   804→        )}
   805→      </button>
   806→
   807→      {/* Body — three columns + buddy + save */}
   808→      <AnimatePresence initial={false}>
   809→        {open && (
   810→          <motion.div
   811→            initial={{ height: 0, opacity: 0 }}
   812→            animate={{ height: "auto", opacity: 1 }}
   813→            exit={{ height: 0, opacity: 0 }}
   814→            transition={{ duration: 0.2 }}
   815→            className="overflow-hidden"
   816→          >
   817→            <div className="p-4 sm:p-5 pt-0 space-y-4">
   818→              <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
   819→                <OutlineColumn
   820→                  title={locale === "ar" ? "30 يوماً" : "30 days"}
   821→                  accent="var(--brand-400)"
   822→                  goals={outline.day30}
   823→                  onChange={(idx, v) => updateGoal("day30", idx, v)}
   824→                  locale={locale}
   825→                />
   826→                <OutlineColumn
   827→                  title={locale === "ar" ? "60 يوماً" : "60 days"}
   828→                  accent="var(--brand-400)"
   829→                  goals={outline.day60}
   830→                  onChange={(idx, v) => updateGoal("day60", idx, v)}
   831→                  locale={locale}
   832→                />
   833→                <OutlineColumn
   834→                  title={locale === "ar" ? "90 يوماً" : "90 days"}
   835→                  accent="var(--accent-500)"
   836→                  goals={outline.day90}
   837→                  onChange={(idx, v) => updateGoal("day90", idx, v)}
   838→                  locale={locale}
   839→                />
   840→              </div>
   841→
   842→              {/* Buddy assignment — named person selector (text input for prototype) */}
   843→              <div>
   844→                <label
   845→                  htmlFor={`buddy-${item.employeeId}`}
   846→                  className="text-[11px] font-semibold uppercase tracking-[0.12em]"
   847→                  style={{ color: "var(--text-lo)" }}
   848→                >
   849→                  {locale === "ar" ? "تكليف الرفيق" : "Buddy assignment"}
   850→                </label>
   851→                <input
   852→                  id={`buddy-${item.employeeId}`}
   853→                  type="text"
   854→                  value={outline.buddy}
   855→                  onChange={(e) => updateBuddy(e.target.value)}
   856→                  placeholder={
   857→                    locale === "ar"
   858→                      ? "اسم الشخص المكلف كرفيق"
   859→                      : "Named person assigned as buddy"
   860→                  }
   861→                  className="ast-focus mt-1.5 w-full h-11 px-3 rounded-[10px] text-[14px] outline-none"
   862→                  style={{
   863→                    background: "var(--surface-2)",
   864→                    color: "var(--text-hi)",
   865→                    border: "1px solid var(--border)",
   866→                  }}
   867→                />
   868→              </div>
   869→
   870→              {/* Save button */}
   871→              <div className="flex items-center justify-end gap-2 pt-1">
   872→                <button
   873→                  type="button"
   874→                  onClick={handleSave}
   875→                  disabled={saving}
   876→                  className="ast-focus inline-flex items-center justify-center gap-2 h-11 px-5 rounded-[10px] text-[13px] font-semibold transition-transform hover:-translate-y-0.5 disabled:opacity-50 disabled:hover:translate-y-0"
   877→                  style={{
   878→                    background: "var(--brand-500)",
   879→                    color: "var(--surface-0)",
   880→                    boxShadow: "0 4px 14px rgba(22,154,165,0.22)",
   881→                  }}
   882→                >
   883→                  <CheckCircle2 className="w-4 h-4" strokeWidth={1.75} />

   888→              </div>
   889→            </div>
   890→          </motion.div>
   891→        )}
   892→      </AnimatePresence>
   893→    </li>
