   127→   127→/** Strip inline [^n] refs and footnote definitions; return clean markdown. */
   128→   128→function stripCitations(md: string): string {
   129→   129→  return md
   130→   130→    // Strip footnote definition lines FIRST (before removing the [^n] markers they start with)
   131→   131→    .replace(/^\[\^\d+\]:.*$/gm, "")
   132→   132→    // Then strip inline [^n] references
   133→   133→    .replace(/\[\^\d+\]/g, "")
   134→   134→    .replace(/\n{3,}/g, "\n\n")
   135→   135→    .trim();
   136→   136→}
   137→   137→
   138→   138→/**
   139→   139→ * Extract the first "decisive" sentence from a markdown answer.
   140→   140→ * Used by the "Say it simply" toggle (PRD §01-016) so an employee can
   141→   141→ * collapse a long answer to just the headline + citations.
   142→   142→ *
   143→   143→ * Strategy: walk the text looking for the first sentence-ending
   144→   144→ * punctuation (`.`, `!`, `?`) that is followed by whitespace or end of
   145→   145→ * string. If none found, return the whole text. Markdown emphasis
   146→   146→ * markers (`*`, `_`, `**`) are preserved so ReactMarkdown still renders
   147→   147→ * them — we only slice on punctuation.
   148→   148→ */
   149→   149→function firstSentence(md: string): string {
   150→   150→  const text = md.trim();
   151→   151→  if (!text) return "";
   152→   152→  // Match up to and including the first . ! or ? that is followed by
   153→   153→  // whitespace, end-of-string, or a newline.
   154→   154→  const match = text.match(/^[^.!?\n]*[.!?](?=\s|$)/);
   155→   155→  if (match) return match[0].trim();
   156→   156→  // Fallback: first line, no terminal punctuation.
   157→   157→  const firstLine = text.split(/\n/)[0];
   158→   158→  return firstLine.trim();
   159→   159→}
   160→   160→
   161→   161→function fmtMs(ms: number): string {
   162→   162→  if (ms < 1000) return `${ms}ms`;
   163→   163→  return `${(ms / 1000).toFixed(1)}s`;
   164→   164→}
