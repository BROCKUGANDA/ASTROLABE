   258→function QueueView({ onOpenPacket }: { onOpenPacket: (employeeId: string) => void }) {
   259→  const locale = useAppStore((s) => s.locale);
   260→  const [items, setItems] = useState<QueueItem[] | null>(null);
   261→  const [err, setErr] = useState<string | null>(null);
   262→  const [loading, setLoading] = useState(true);
   263→  const [claiming, setClaiming] = useState<string | null>(null);
   264→
   265→  const fetchQ = useCallback(async () => {
   266→    setLoading(true);
   267→    setErr(null);
   268→    try {
   269→      const res = await fetch("/api/v1/hrbp/queue", { cache: "no-store" });
   270→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   271→      const json: ApiEnvelope<{ items: QueueItem[] }> = await res.json();
   272→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   273→      setItems(json.data.items);
   274→    } catch (e) {
   275→      setErr(e instanceof Error ? e.message : "Network error");
   276→    } finally {
   277→      setLoading(false);
   278→    }
   279→  }, []);
   280→
   281→  useEffect(() => { fetchQ(); }, [fetchQ]);
   282→
   283→  const handleClaim = useCallback(async (item: QueueItem) => {
   284→    setClaiming(item.employeeId);
   285→    // Fake audit-event write — there's no API yet, so we just simulate the receipt.
   286→    await new Promise((r) => setTimeout(r, 320));
   287→    toast.success("Case claimed by Fatima — employee notified", {
   288→      description: `${item.employeeName} · ${item.employeeId}`,
   289→    });
   290→    setClaiming(null);
   291→  }, []);
   292→
   293→  if (loading) return <QueueSkeleton />;
   294→  if (err) return <ErrorBanner message={err} onRetry={fetchQ} />;
   295→  if (!items) return null;
   296→
   297→  const active = items.filter((i) => bandSortIndex(i.band) >= 2); // friction + blocked
   298→  const rest = items.filter((i) => bandSortIndex(i.band) < 2);
   299→
   300→  return (
   301→    <div>
   302→      <div className="flex items-center justify-between mb-4 flex-wrap gap-3">
   303→        <div>
   304→          <h2 className="text-[17px] font-semibold" style={{ color: "var(--text-hi)" }}>
   305→            {locale === "ar" ? "قائمة أول من يحتاج مساعدة" : "Who needs help first"}
   306→          </h2>
   307→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   308→            {locale === "ar"
   309→              ? `${active.length} موظف في احتكاك أو أعلى اليوم.`
   310→              : `${active.length} employee${active.length === 1 ? "" : "s"} in Friction or above today.`}
   311→          </p>
   312→        </div>
   313→        <button
   314→          onClick={fetchQ}
   315→          className="ast-focus inline-flex items-center gap-2 h-9 px-3 rounded-full text-[12px] font-medium"
   316→          style={{ background: "var(--surface-2)", border: "1px solid var(--border)", color: "var(--text-mid)" }}
   317→        >
   318→          <RefreshCw className="w-3.5 h-3.5" strokeWidth={1.75} />
   319→          {locale === "ar" ? "تحديث" : "Refresh"}
   320→        </button>
   321→      </div>
   322→
   323→      {active.length === 0 ? (
   324→        <div className="ast-card p-10 flex flex-col items-center text-center">
   325→          <div
   326→            className="flex items-center justify-center w-14 h-14 rounded-full mb-4"
   327→            style={{ background: "rgba(14,159,110,0.10)", color: "var(--status-steady)" }}
   328→          >
   329→            <CheckCircle2 className="w-6 h-6" strokeWidth={1.75} />
   330→          </div>
   331→          <p className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   332→            {locale === "ar" ? "لا أحد في احتكاك أو أعلى اليوم." : "No one in Friction or above today."}
   333→          </p>
   334→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   335→            {locale === "ar" ? "تحقق لاحقاً، أو تصفّح بقية القائمة أدناه." : "Check back later, or browse the watch list below."}
   336→          </p>
   337→        </div>
   338→      ) : (
   339→        <ul className="space-y-3 mb-8">
   340→          {active.map((item, i) => (
   341→            <QueueRow
   342→              key={item.employeeId}
   343→              item={item}
   344→              rank={i + 1}
   345→              claiming={claiming === item.employeeId}
   346→              onClaim={() => handleClaim(item)}
   347→              onOpen={() => onOpenPacket(item.employeeId)}
   348→            />
   349→          ))}
   350→        </ul>
   351→      )}
   352→
   353→      {/* Watch + steady (collapsed by default-ish) */}
   354→      {rest.length > 0 && (
   355→        <details className="ast-card p-4" style={{ borderColor: "var(--border-soft)" }}>
   356→          <summary className="cursor-pointer text-[13px] font-medium flex items-center gap-2" style={{ color: "var(--text-mid)" }}>
   357→            <ChevronRight className="w-3.5 h-3.5" strokeWidth={1.75} />
   358→            {locale === "ar"
   359→              ? `${rest.length} موظف تحت المراقبة أو مستقر`
   360→              : `${rest.length} on watch or steady`}
   361→          </summary>
   362→          <ul className="mt-3 space-y-2">
   363→            {rest.map((item, i) => (
   364→              <QueueRow
   365→                key={item.employeeId}
   366→                item={item}
   367→                rank={active.length + i + 1}
   368→                claiming={claiming === item.employeeId}
   369→                onClaim={() => handleClaim(item)}
   370→                onOpen={() => onOpenPacket(item.employeeId)}
   371→                compact
   372→              />
   373→            ))}
   374→          </ul>
   375→        </details>
   376→      )}
   377→    </div>
   378→  );
   379→}

   492→  const reducedMotion = useAppStore((s) => s.reducedMotion);

   495→  const [packet, setPacket] = useState<EscalationPacket | null>(null);
   496→  const [loading, setLoading] = useState(true);

   501→  const [editMode, setEditMode] = useState(false);
   502→  const [building, setBuilding] = useState(false);

   504→  const [sent, setSent] = useState(false);
   505→  const [confirmOpen, setConfirmOpen] = useState(false);

   507→  const fetchAssessment = useCallback(async () => {
   508→    setLoading(true);
   509→    setErr(null);

   511→      const res = await fetch(`/api/v1/journey/${encodeURIComponent(employeeId)}/assessment`, { cache: "no-store" });

   515→      setAssessment(json.data.assessment);
   516→      setTrace(json.data.trace);

  1007→function StallsView() {
  1008→  const locale = useAppStore((s) => s.locale);
  1009→  const [cells, setCells] = useState<StallsCell[] | null>(null);
  1010→  const [err, setErr] = useState<string | null>(null);
  1011→  const [loading, setLoading] = useState(true);
  1012→
  1013→  const fetchCells = useCallback(async () => {
  1014→    setLoading(true);
  1015→    setErr(null);
  1016→    try {
  1017→      const res = await fetch("/api/v1/insights/stalls", { cache: "no-store" });
  1018→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
  1019→      const json: ApiEnvelope<{ cells: StallsCell[] }> = await res.json();
  1020→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
  1021→      setCells(json.data.cells);
  1022→    } catch (e) {
  1023→      setErr(e instanceof Error ? e.message : "Network error");
  1024→    } finally {
  1025→      setLoading(false);
  1026→    }
  1027→  }, []);
  1028→
  1029→  useEffect(() => { fetchCells(); }, [fetchCells]);
  1030→
  1031→  // Build matrix
  1032→  const matrix = useMemo(() => {
  1033→    if (!cells) return { stages: [], groups: [], map: new Map<string, StallsCell>() };
  1034→    const stageSet = new Set<string>();
  1035→    const groupSet = new Set<string>();
  1036→    const map = new Map<string, StallsCell>();
  1037→    for (const c of cells) {
  1038→      stageSet.add(c.stage);
  1039→      groupSet.add(c.ownerGroup);
  1040→      map.set(`${c.stage}|${c.ownerGroup}`, c);
  1041→    }
  1042→    // Preserve stage order via STAGE_LABELS insertion
  1043→    const stageOrder = Object.keys(STAGE_LABELS);
  1044→    const stages = Array.from(stageSet).sort((a, b) => stageOrder.indexOf(a) - stageOrder.indexOf(b));
  1045→    const groups = Array.from(groupSet).sort();
  1046→    return { stages, groups, map };
  1047→  }, [cells]);
  1048→
  1049→  // Color scale: teal → sand-gold → red (by medianDays)

  1051→    if (!c || c.suppressed) return "var(--surface-2)";
  1052→    const m = c.medianTimeToCompleteHours ?? 0;
  1053→    // Map 0h → teal, 48h → sand-gold, 168h+ → red
  1054→    if (m <= 24) return "rgba(22,154,165,0.18)";
  1055→    if (m <= 72) return "rgba(201,162,39,0.18)";
  1056→    if (m <= 168) return "rgba(217,119,6,0.20)";
  1057→    return "rgba(220,38,38,0.20)";

  1089→            {locale === "ar" ? "خريطة الاختناقات" : "Stall heatmap"}
  1090→          </h2>
  1091→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>

  1093→              ? "الوسيط لكل (مرحلة × مالك). الخلايا أقل من 5 مكتومة."
  1094→              : "Median days per stage × owner. Cells below k=5 are suppressed."}
