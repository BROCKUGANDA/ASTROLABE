   258→   258→function QueueView({ onOpenPacket }: { onOpenPacket: (employeeId: string) => void }) {
   259→   259→  const locale = useAppStore((s) => s.locale);
   260→   260→  const [items, setItems] = useState<QueueItem[] | null>(null);
   261→   261→  const [err, setErr] = useState<string | null>(null);
   262→   262→  const [loading, setLoading] = useState(true);
   263→   263→  const [claiming, setClaiming] = useState<string | null>(null);
   264→   264→
   265→   265→  const fetchQ = useCallback(async () => {
   266→   266→    setLoading(true);
   267→   267→    setErr(null);
   268→   268→    try {
   269→   269→      const res = await fetch("/api/v1/hrbp/queue", { cache: "no-store" });
   270→   270→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   271→   271→      const json: ApiEnvelope<{ items: QueueItem[] }> = await res.json();
   272→   272→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   273→   273→      setItems(json.data.items);
   274→   274→    } catch (e) {
   275→   275→      setErr(e instanceof Error ? e.message : "Network error");
   276→   276→    } finally {
   277→   277→      setLoading(false);
   278→   278→    }
   279→   279→  }, []);
   280→   280→
   281→   281→  useEffect(() => { fetchQ(); }, [fetchQ]);
   282→   282→
   283→   283→  const handleClaim = useCallback(async (item: QueueItem) => {
   284→   284→    setClaiming(item.employeeId);
   285→   285→    // Fake audit-event write — there's no API yet, so we just simulate the receipt.
   286→   286→    await new Promise((r) => setTimeout(r, 320));
   287→   287→    toast.success("Case claimed by Fatima — employee notified", {
   288→   288→      description: `${item.employeeName} · ${item.employeeId}`,
   289→   289→    });
   290→   290→    setClaiming(null);
   291→   291→  }, []);
   292→   292→
   293→   293→  if (loading) return <QueueSkeleton />;
   294→   294→  if (err) return <ErrorBanner message={err} onRetry={fetchQ} />;
   295→   295→  if (!items) return null;
   296→   296→
   297→   297→  const active = items.filter((i) => bandSortIndex(i.band) >= 2); // friction + blocked
   298→   298→  const rest = items.filter((i) => bandSortIndex(i.band) < 2);
   299→   299→
   300→   300→  return (
   301→   301→    <div>
   302→   302→      <div className="flex items-center justify-between mb-4 flex-wrap gap-3">
   303→   303→        <div>
   304→   304→          <h2 className="text-[17px] font-semibold" style={{ color: "var(--text-hi)" }}>
   305→   305→            {locale === "ar" ? "قائمة أول من يحتاج مساعدة" : "Who needs help first"}
   306→   306→          </h2>
   307→   307→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   308→   308→            {locale === "ar"
   309→   309→              ? `${active.length} موظف في احتكاك أو أعلى اليوم.`
   310→   310→              : `${active.length} employee${active.length === 1 ? "" : "s"} in Friction or above today.`}
   311→   311→          </p>
   312→   312→        </div>
   313→   313→        <button
   314→   314→          onClick={fetchQ}
   315→   315→          className="ast-focus inline-flex items-center gap-2 h-9 px-3 rounded-full text-[12px] font-medium"
   316→   316→          style={{ background: "var(--surface-2)", border: "1px solid var(--border)", color: "var(--text-mid)" }}
   317→   317→        >
   318→   318→          <RefreshCw className="w-3.5 h-3.5" strokeWidth={1.75} />
   319→   319→          {locale === "ar" ? "تحديث" : "Refresh"}
   320→   320→        </button>
   321→   321→      </div>
   322→   322→
   323→   323→      {active.length === 0 ? (
   324→   324→        <div className="ast-card p-10 flex flex-col items-center text-center">
   325→   325→          <div
   326→   326→            className="flex items-center justify-center w-14 h-14 rounded-full mb-4"
   327→   327→            style={{ background: "rgba(14,159,110,0.10)", color: "var(--status-steady)" }}
   328→   328→          >
   329→   329→            <CheckCircle2 className="w-6 h-6" strokeWidth={1.75} />
   330→   330→          </div>
   331→   331→          <p className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   332→   332→            {locale === "ar" ? "لا أحد في احتكاك أو أعلى اليوم." : "No one in Friction or above today."}
   333→   333→          </p>
   334→   334→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   335→   335→            {locale === "ar" ? "تحقق لاحقاً، أو تصفّح بقية القائمة أدناه." : "Check back later, or browse the watch list below."}
   336→   336→          </p>
   337→   337→        </div>
   338→   338→      ) : (
   339→   339→        <ul className="space-y-3 mb-8">
   340→   340→          {active.map((item, i) => (
   341→   341→            <QueueRow
   342→   342→              key={item.employeeId}
   343→   343→              item={item}
   344→   344→              rank={i + 1}
   345→   345→              claiming={claiming === item.employeeId}
   346→   346→              onClaim={() => handleClaim(item)}
   347→   347→              onOpen={() => onOpenPacket(item.employeeId)}
   348→   348→            />
   349→   349→          ))}
   350→   350→        </ul>
   351→   351→      )}
   352→   352→
   353→   353→      {/* Watch + steady (collapsed by default-ish) */}
   354→   354→      {rest.length > 0 && (
   355→   355→        <details className="ast-card p-4" style={{ borderColor: "var(--border-soft)" }}>
   356→   356→          <summary className="cursor-pointer text-[13px] font-medium flex items-center gap-2" style={{ color: "var(--text-mid)" }}>
   357→   357→            <ChevronRight className="w-3.5 h-3.5" strokeWidth={1.75} />
   358→   358→            {locale === "ar"
   359→   359→              ? `${rest.length} موظف تحت المراقبة أو مستقر`
   360→   360→              : `${rest.length} on watch or steady`}
   361→   361→          </summary>
   362→   362→          <ul className="mt-3 space-y-2">
   363→   363→            {rest.map((item, i) => (
   364→   364→              <QueueRow
   365→   365→                key={item.employeeId}
   366→   366→                item={item}
   367→   367→                rank={active.length + i + 1}
   368→   368→                claiming={claiming === item.employeeId}
   369→   369→                onClaim={() => handleClaim(item)}
   370→   370→                onOpen={() => onOpenPacket(item.employeeId)}
   371→   371→                compact
   372→   372→              />
   373→   373→            ))}
   374→   374→          </ul>
   375→   375→        </details>
   376→   376→      )}
   377→   377→    </div>
   378→   378→  );
   379→   379→}
   380→   380→
   381→   381→function QueueRow({
   382→   382→  item,
   383→   383→  rank,
   384→   384→  claiming,
   385→   385→  onClaim,
   386→   386→  onOpen,
   387→   387→  compact = false,
   388→   388→}: {
   389→   389→  item: QueueItem;
   390→   390→  rank: number;
   391→   391→  claiming: boolean;
   392→   392→  onClaim: () => void;
   393→   393→  onOpen: () => void;
   394→   394→  compact?: boolean;
   395→   395→}) {
   396→   396→  const locale = useAppStore((s) => s.locale);
   397→   397→  const leadDays = Math.max(0, Math.round(item.leadTimeHours / 24));
   398→   398→  return (
   399→   399→    <motion.li
   400→   400→      initial={false}
   401→   401→      className={cn("ast-card p-4 sm:p-5", compact && "p-3")}
   402→   402→    >
   403→   403→      <div className="flex items-start gap-3 sm:gap-4 flex-wrap">
   404→   404→        {/* Rank + band */}
   405→   405→        <div className="flex items-center gap-3 shrink-0">
   406→   406→          <span className="text-[12px] tabular font-mono" style={{ color: "var(--text-lo)" }}>
   407→   407→            #{rank}
   408→   408→          </span>
   409→   409→          <BandChip band={item.band} score={item.score} locale={locale} size="lg" />
   410→   410→        </div>
   411→   411→
   412→   412→        {/* Name + stage */}
   413→   413→        <div className="flex-1 min-w-0">
   414→   414→          <p className="text-[15px] font-medium truncate" style={{ color: "var(--text-hi)" }}>
   415→   415→            {item.employeeName}
   416→   416→          </p>
   417→   417→          <p className="text-[12px] mt-0.5" style={{ color: "var(--text-mid)" }}>
   418→   418→            {stageLabel(item.stage, locale)}
   419→   419→            <span className="mx-1.5">·</span>
   420→   420→            <span className="tabular">
   421→   421→              {locale === "ar"
   422→   422→                ? `أول إشارة قبل ${leadDays} يوم`
   423→   423→                : `first signal ${leadDays} day${leadDays === 1 ? "" : "s"} ago`}
   424→   424→            </span>
   425→   425→          </p>
   426→   426→
   427→   427→          {/* Top signals */}
   428→   428→          {item.topSignals.length > 0 && (
   429→   429→            <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   430→   430→              {item.topSignals.map((s, i) => (
   431→   431→                <SignalBadge
   432→   432→                  key={`${s.code}-${i}`}
   433→   433→                  code={s.code}
   434→   434→                  weight={s.weight}
   435→   435→                  hardTrip={s.code.startsWith("DP") || s.code.startsWith("MB")}
   436→   436→                />
   437→   437→              ))}
   438→   438→            </div>
   439→   439→          )}
   440→   440→        </div>
