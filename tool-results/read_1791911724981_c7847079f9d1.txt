   145→function namedOwnerForGroup(group: string): { name: string; nameAr?: string; role: string; group: string } | null {
   146→  const o = ALL_OWNERS.find((x) => x.group === group);

   151→function stageLabel(stageKey: string, journeyType: string, locale: "en" | "ar"): string {
   152→  const st = ALL_STAGES.find((s) => s.key === stageKey && s.journeyType === journeyType);
   153→  if (!st) return stageKey;
   154→  return locale === "ar" ? st.labelAr : st.label;
   155→}

   157→function stageShort(stageKey: string, journeyType: string, locale: "en" | "ar"): string {
   158→  const full = stageLabel(stageKey, journeyType, locale);
   159→  // Compact rail labels: take first 1–2 words
   160→  if (locale === "ar") return full;
   161→  const parts = full.split(" ");
   162→  return parts.length > 2 ? parts.slice(0, 2).join(" ") : full;
   163→}

   165→function fmtDateAndDistance(iso: string | undefined, locale: "en" | "ar"): string {
   166→  if (!iso) return locale === "ar" ? "في أي وقت" : "any time";
   167→  const d = new Date(iso);
   168→  const datePart = format(d, "EEEE d MMM");
   169→  const distance = formatDistanceToNow(d, { addSuffix: false });
   170→  return locale === "ar" ? `${datePart} · ${distance}` : `${datePart} · ${distance}`;
   171→}

   173→function lateCopy(step: Step, locale: "en" | "ar"): string {
   174→  if (step.breachLevel === "hard") {
   175→    const days = Math.max(1, Math.floor((step.elapsedHours - step.targetHours) / 24));
   176→    return locale === "ar"
   177→      ? `متأخر بـ ${days} أيام — خرق جوهري`
   178→      : `${days} day${days === 1 ? "" : "s"} late — hard breach`;
   179→  }
   180→  if (step.breachLevel === "breached") {
   181→    const days = Math.max(1, Math.floor((step.elapsedHours - step.targetHours) / 24));
   182→    return locale === "ar" ? `متأخر بـ ${days} أيام` : `${days} day${days === 1 ? "" : "s"} late`;
   183→  }
   184→  if (step.status === "blocked") {
   185→    return locale === "ar"
   186→      ? "محظور بانتظار خطوات أخرى"
   187→      : "Held up — waiting on another step";
   188→  }
   189→  return "";
   190→}

   192→function ownerLine(step: Step, locale: "en" | "ar"): string {
   193→  const grp = OWNER_GROUP_LABEL[step.ownerGroup];
   194→  const named = namedOwnerForGroup(step.ownerGroup);
   195→  if (!grp) return step.ownerGroup;
   196→  if (named) {
   197→    return locale === "ar" && named.nameAr
   198→      ? `مع ${named.nameAr} في ${grp.ar}`
   199→      : `with ${named.name} in ${grp.en}`;
   200→  }
   201→  return locale === "ar" ? `مع ${grp.ar}` : `with ${grp.en}`;
   202→}

   204→function upcomingTargetCopy(step: Step, locale: "en" | "ar"): string {

   208→    ? `يُتوقع خلال نحو ${days} يوم`
   209→    : `about ${days} day${days === 1 ? "" : "s"} from start`;
   210→}

   489→/* Parse "docSlug#section" → CitationChip shape */
   490→function parsePolicyRef(ref: string): { docSlug: string; title: string; section: string; effectiveDate: string } {
   491→  const [docSlug, section] = ref.split("#");
   492→  const titleMap: Record<string, string> = {
   493→    "relocation-policy": "Relocation Policy",
   494→    "immigration-policy": "Immigration Policy",
   495→    "it-policy": "IT Provisioning Policy",
   496→    "onboarding-policy": "Onboarding Policy",
   497→    "learning-policy": "Learning Policy",
   498→    "mobility-policy": "Internal Mobility Policy",
   499→    "payroll-policy": "Payroll Policy",
   500→    "facilities-policy": "Facilities Policy",
   501→  };
   502→  return {
   503→    docSlug,
   504→    title: titleMap[docSlug] ?? docSlug,
   505→    section: section ?? "—",
   506→    effectiveDate: "2024-01-01",
   507→  };

   623→/* ---------- Stage Rail ---------- */
   624→
   625→function StageRail({ data }: { data: JourneyMeData }) {
   626→  const { locale } = useAppStore();
   627→  const journeyType = data.employee.journeyType;
   628→  const journeyStages = useMemo(
   629→    () => ALL_STAGES.filter((s) => s.journeyType === journeyType).sort((a, b) => a.order - b.order),
   630→    [journeyType]
   631→  );
   632→
   633→  const segState = (stageKey: string): "done" | "active" | "blocked" | "pending" => {
   634→    const stepsInStage = data.steps.filter((s) => s.stage === stageKey);
   635→    if (stepsInStage.length === 0) return "pending";
   636→    const allDone = stepsInStage.every((s) => s.status === "done" || s.status === "skipped");
   637→    if (allDone) return "done";
   638→    const anyBlocked = stepsInStage.some(
   639→      (s) => s.status === "blocked" || s.breachLevel === "hard"
   640→    );
   641→    if (anyBlocked) return "blocked";
   642→    if (stageKey === data.stage) return "active";
   643→    // Stage before current — if not all done, mark done visually (history)
   644→    const stageOrder = journeyStages.find((s) => s.key === stageKey)?.order ?? 0;
   645→    const currentOrder = journeyStages.find((s) => s.key === data.stage)?.order ?? 0;
   646→    if (stageOrder < currentOrder) return "done";
   647→    return "pending";
   648→  };
   649→
   650→  return (
   651→    <section className="ast-card p-5 sm:p-6" aria-label="Stage rail">
   652→      <div className="flex items-center gap-2 mb-4">
   653→        <Compass className="w-4 h-4" strokeWidth={1.75} style={{ color: "var(--brand-400)" }} />
   654→        <h3 className="text-[14px] font-semibold uppercase tracking-[0.12em]" style={{ color: "var(--text-hi)" }}>
   655→          {locale === "ar" ? "مسار الرحلة" : "Journey rail"}
   656→        </h3>
   657→      </div>
   658→      <ol className="grid gap-2" style={{ gridTemplateColumns: `repeat(${journeyStages.length}, minmax(0, 1fr))` }}>
   659→        {journeyStages.map((st) => {
   660→          const state = segState(st.key);
   661→          const isActive = st.key === data.stage;
   662→          const label = locale === "ar" ? st.labelAr : st.label;
   663→          return (
   664→            <li key={st.key} className="flex flex-col gap-2 min-w-0">
   665→              <div
   666→                className="ast-rail-seg"
   667→                data-state={state === "pending" ? undefined : state}
   668→                aria-label={`${label}: ${state}`}
   669→                style={state === "pending" ? { background: "var(--surface-3)" } : undefined}
   670→              />
   671→              <div className="min-w-0">
   672→                <p
   673→                  className="text-[11px] sm:text-[12px] font-medium leading-tight truncate"
   674→                  style={{
   675→                    color: isActive ? "var(--accent-500)" : state === "done" ? "var(--brand-400)" : state === "blocked" ? "var(--status-blocked)" : "var(--text-lo)",
   676→                  }}
   677→                >
   678→                  {stageShort(st.key, journeyType, locale)}
   679→                </p>
   680→                {isActive && (
   681→                  <p className="text-[10px] tabular mt-0.5" style={{ color: "var(--text-lo)" }}>

   998→        return;
   999→      }
  1000→      const idempotencyKey = `idm-${emp.id}-${requestType}-${Date.now().toString(36)}`;
  1001→      try {
  1002→        const res = await fetch("/api/v1/requests", {
  1003→          method: "POST",
  1004→          headers: { "Content-Type": "application/json" },
  1005→          body: JSON.stringify({
  1006→            employeeId: emp.id,
  1007→            type: requestType,
  1008→            payload: { stepKey: nbs.stepKey, label: nbs.label },
  1009→            idempotencyKey,
  1010→          }),
  1011→        });
  1012→        const json: ApiEnvelope<{ ref: string; owner?: { name: string } }> = await res.json();
  1013→        if (json.ok && json.data) {
  1014→          const ownerName = json.data.owner?.name ?? "the team";
  1015→          toast.success(locale === "ar" ? "تم إرسال الطلب" : "Request sent", {
  1016→            description: locale === "ar"
  1017→              ? `${nbs.labelAr} — ${ownerName} سيتولى الأمر`
  1018→              : `${nbs.label} — ${ownerName} will handle it`,
  1019→          });
  1020→        } else {
  1021→          toast.error(locale === "ar" ? "تعذّر الإرسال" : "Couldn't send", {
  1022→            description: json.error?.message ?? "Please try again.",
  1023→          });
  1024→        }
  1025→      } catch {
  1026→        toast.error(locale === "ar" ? "خطأ في الشبكة" : "Network error", {
  1027→          description: locale === "ar" ? "حاول مرة أخرى." : "Please retry.",
  1028→        });
  1029→      }
  1030→    },
  1031→    [data, locale]
  1032→  );
  1033→
  1034→  const handleGrantConsent = useCallback(async () => {
  1035→    try {
  1036→      const res = await fetch("/api/v1/consent", {
  1037→        method: "POST",
  1038→        headers: { "Content-Type": "application/json" },
  1039→        body: JSON.stringify({
  1040→          employeeId,
  1041→          purpose: "pulse_capture",
  1042→          granted: true,
  1043→          version: "v1",
  1044→        }),
  1045→      });
  1046→      const json: ApiEnvelope<{ receiptId: string }> = await res.json();
  1047→      if (json.ok) {
  1048→        setPulseGranted(true);
  1049→        toast.success(locale === "ar" ? "تم تفعيل التقدير" : "Pulse capture enabled", {

  1051→            ? `رقم الإيصال: ${json.data?.receiptId ?? ""}`
  1052→            : `Receipt: ${json.data?.receiptId ?? ""}`,
  1053→        });
  1054→      } else {
  1055→        toast.error(json.error?.message ?? "Failed");
  1056→      }
  1057→    } catch {
  1058→      toast.error(locale === "ar" ? "خطأ في الشبكة" : "Network error");
  1059→    }
  1060→  }, [employeeId, locale]);
  1061→
  1062→  const handleSubmitPulse = useCallback(
  1063→    async (score: number) => {
  1064→      if (!data) return;
  1065→      try {
  1066→        const res = await fetch("/api/v1/pulse", {
  1067→          method: "POST",
  1068→          headers: { "Content-Type": "application/json" },
  1069→          body: JSON.stringify({
  1070→            employeeId,
  1071→            stage: data.stage,
  1072→            score,
  1073→            consentId: `CN-pulse-${employeeId}`,
  1074→          }),
  1075→        });
  1076→        const json: ApiEnvelope<{ id: string }> = await res.json();
  1077→        if (json.ok) {
  1078→          toast.success(locale === "ar" ? "تم الإرسال" : "Pulse recorded", {
  1079→            description: locale === "ar" ? "شكراً لمشاركتك." : "Thank you for sharing.",
  1080→          });
  1081→        } else if (json.error?.code === "E_CONSENT_REQUIRED") {
  1082→          setPulseGranted(false);
  1083→          toast.error(locale === "ar" ? "يلزم الموافقة" : "Consent required", {
  1084→            description: json.error.hint ?? "",
  1085→          });
  1086→        } else {
  1087→          toast.error(json.error?.message ?? "Failed");
  1088→        }
  1089→      } catch {
  1090→        toast.error(locale === "ar" ? "خطأ في الشبكة" : "Network error");
  1091→      }
  1092→    },
  1093→    [data, employeeId, locale]
  1094→  );
  1095→
  1096→  /* Render ---------------------------------------------------- */
  1097→
  1098→  return (
  1099→    <div
  1100→      dir={dir}
  1101→      className="min-h-screen flex flex-col"
  1102→      style={{ background: "var(--surface-0)" }}
  1103→    >
  1104→      <main className="flex-1 w-full max-w-[1180px] mx-auto px-4 sm:px-6 lg:px-8 py-6 sm:py-8 space-y-6">
  1105→        {loading ? (
  1106→          <TodaySkeleton />
  1107→        ) : error ? (
  1108→          <ErrorBanner
  1109→            message={error.message}
  1110→            requestId={error.requestId}
  1111→            onRetry={() => void load()}
  1112→          />
  1113→        ) : data ? (
  1114→          <>
  1115→            <HeroStrip data={data} />
  1116→
  1117→            <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">

  1121→
  1122→            <StageRail data={data} />
  1123→
  1124→            <ComingUp data={data} />
  1125→
  1126→            <PulseCard
  1127→              employeeId={employeeId}
  1128→              stage={data.stage}
  1129→              pulseGranted={pulseGranted}
  1130→              onGrantConsent={handleGrantConsent}

  1135→            {/* Version footer */}

  1140→              {data.engineVersion} · {data.mapVersion} · {data.corpusVersion}
  1141→            </p>
