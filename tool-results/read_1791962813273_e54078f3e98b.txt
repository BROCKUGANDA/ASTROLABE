   268→   268→function QueueView({ onOpenPacket }: { onOpenPacket: (employeeId: string) => void }) {
   269→   269→  const locale = useAppStore((s) => s.locale);
   270→   270→  const [items, setItems] = useState<QueueItem[] | null>(null);
   271→   271→  const [err, setErr] = useState<string | null>(null);
   272→   272→  const [loading, setLoading] = useState(true);
   273→   273→  const [claiming, setClaiming] = useState<string | null>(null);
   274→   274→
   275→   275→  const fetchQ = useCallback(async () => {
   276→   276→    setLoading(true);
   277→   277→    setErr(null);
   278→   278→    try {
   279→   279→      const res = await fetch("/api/v1/hrbp/queue", { cache: "no-store" });
   280→   280→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   281→   281→      const json: ApiEnvelope<{ items: QueueItem[] }> = await res.json();
   282→   282→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   283→   283→      setItems(json.data.items);
   284→   284→    } catch (e) {
   285→   285→      setErr(e instanceof Error ? e.message : "Network error");
   286→   286→    } finally {
   287→   287→      setLoading(false);
   288→   288→    }
   289→   289→  }, []);
   290→   290→
   291→   291→  useEffect(() => { fetchQ(); }, [fetchQ]);
   292→   292→
   293→   293→  const handleClaim = useCallback(async (item: QueueItem) => {
   294→   294→    setClaiming(item.employeeId);
   295→   295→    // Fake audit-event write — there's no API yet, so we just simulate the receipt.
   296→   296→    await new Promise((r) => setTimeout(r, 320));
   297→   297→    toast.success("Case claimed by Fatima — employee notified", {
   298→   298→      description: `${item.employeeName} · ${item.employeeId}`,
   299→   299→    });
   300→   300→    setClaiming(null);
   301→   301→  }, []);
   302→   302→
   303→   303→  if (loading) return <QueueSkeleton />;
   304→   304→  if (err) return <ErrorBanner message={err} onRetry={fetchQ} />;
   305→   305→  if (!items) return null;
   306→   306→
   307→   307→  const active = items.filter((i) => bandSortIndex(i.band) >= 2); // friction + blocked
   308→   308→  const rest = items.filter((i) => bandSortIndex(i.band) < 2);
   309→   309→
   310→   310→  return (
   311→   311→    <div>
   312→   312→      <div className="flex items-center justify-between mb-4 flex-wrap gap-3">
   313→   313→        <div>
   314→   314→          <h2 className="text-[17px] font-semibold" style={{ color: "var(--text-hi)" }}>
   315→   315→            {locale === "ar" ? "قائمة أول من يحتاج مساعدة" : "Who needs help first"}
   316→   316→          </h2>
   317→   317→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   318→   318→            {locale === "ar"
   319→   319→              ? `${active.length} موظف في احتكاك أو أعلى اليوم.`
   320→   320→              : `${active.length} employee${active.length === 1 ? "" : "s"} in Friction or above today.`}
   321→   321→          </p>
   322→   322→        </div>
   323→   323→        <button
   324→   324→          onClick={fetchQ}
   325→   325→          className="ast-focus inline-flex items-center gap-2 h-9 px-3 rounded-full text-[12px] font-medium"
   326→   326→          style={{ background: "var(--surface-2)", border: "1px solid var(--border)", color: "var(--text-mid)" }}
   327→   327→        >
   328→   328→          <RefreshCw className="w-3.5 h-3.5" strokeWidth={1.75} />
   329→   329→          {locale === "ar" ? "تحديث" : "Refresh"}
   330→   330→        </button>
   331→   331→      </div>
   332→   332→
   333→   333→      {active.length === 0 ? (
   334→   334→        <div className="ast-card p-10 flex flex-col items-center text-center">
   335→   335→          <div
   336→   336→            className="flex items-center justify-center w-14 h-14 rounded-full mb-4"
   337→   337→            style={{ background: "rgba(14,159,110,0.10)", color: "var(--status-steady)" }}
   338→   338→          >
   339→   339→            <CheckCircle2 className="w-6 h-6" strokeWidth={1.75} />
   340→   340→          </div>
   341→   341→          <p className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   342→   342→            {locale === "ar" ? "لا أحد في احتكاك أو أعلى اليوم." : "No one in Friction or above today."}
   343→   343→          </p>
   344→   344→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   345→   345→            {locale === "ar" ? "تحقق لاحقاً، أو تصفّح بقية القائمة أدناه." : "Check back later, or browse the watch list below."}
   346→   346→          </p>
   347→   347→        </div>
   348→   348→      ) : (
   349→   349→        <ul className="space-y-3 mb-8">
   350→   350→          {active.map((item, i) => (
   351→   351→            <QueueRow
   352→   352→              key={item.employeeId}
   353→   353→              item={item}
   354→   354→              rank={i + 1}
   355→   355→              index={i}
   356→   356→              claiming={claiming === item.employeeId}
   357→   357→              onClaim={() => handleClaim(item)}
   358→   358→              onOpen={() => onOpenPacket(item.employeeId)}
   359→   359→            />
   360→   360→          ))}
   361→   361→        </ul>
   362→   362→      )}
   363→   363→
   364→   364→      {/* Watch + steady (collapsed by default-ish) */}
   365→   365→      {rest.length > 0 && (
   366→   366→        <details className="ast-card p-4" style={{ borderColor: "var(--border-soft)" }}>
   367→   367→          <summary className="cursor-pointer text-[13px] font-medium flex items-center gap-2" style={{ color: "var(--text-mid)" }}>
   368→   368→            <ChevronRight className="w-3.5 h-3.5" strokeWidth={1.75} />
   369→   369→            {locale === "ar"
   370→   370→              ? `${rest.length} موظف تحت المراقبة أو مستقر`
   371→   371→              : `${rest.length} on watch or steady`}
   372→   372→          </summary>
   373→   373→          <ul className="mt-3 space-y-2">
   374→   374→            {rest.map((item, i) => (
   375→   375→              <QueueRow
   376→   376→                key={item.employeeId}
   377→   377→                item={item}
   378→   378→                rank={active.length + i + 1}
   379→   379→                index={i}
   380→   380→                claiming={claiming === item.employeeId}
   381→   381→                onClaim={() => handleClaim(item)}
   382→   382→                onOpen={() => onOpenPacket(item.employeeId)}
   383→   383→                compact
   384→   384→              />
   385→   385→            ))}
   386→   386→          </ul>
   387→   387→        </details>
   388→   388→      )}
   389→   389→    </div>
   390→   390→  );
   391→   391→}

   393→   393→function QueueRow({
   394→   394→  item,
   395→   395→  rank,
   396→   396→  index,
   397→   397→  claiming,
   398→   398→  onClaim,
   399→   399→  onOpen,
   400→   400→  compact = false,
   401→   401→}: {
   402→   402→  item: QueueItem;
   403→   403→  rank: number;
   404→   404→  index: number;
   405→   405→  claiming: boolean;
   406→   406→  onClaim: () => void;
   407→   407→  onOpen: () => void;
   408→   408→  compact?: boolean;
   409→   409→}) {
   410→   410→  const locale = useAppStore((s) => s.locale);
   411→   411→  const reducedMotion = useAppStore((s) => s.reducedMotion);
   412→   412→  const leadDays = Math.max(0, Math.round(item.leadTimeHours / 24));
   413→   413→
   414→   414→  // Hover-expand state.
   415→   415→  // Desktop (pointer: fine) → hover reveals preview.
   416→   416→  // Mobile  (pointer: coarse) → tap chevron toggles preview.
   417→   417→  const [isHover, setIsHover] = useState(false);
   418→   418→  const [isTapped, setIsTapped] = useState(false);
   419→   419→
   420→   420→  // Band → accent colour (left bar + glow).
   421→   421→  const bandAccent: Record<BandT, string> = {
   422→   422→    steady: "var(--status-steady)",
   423→   423→    watch: "var(--status-watch)",
   424→   424→    friction: "var(--status-friction)",
   425→   425→    blocked: "var(--status-blocked)",
   426→   426→  };
   427→   427→  const accent = bandAccent[item.band];
   428→   428→
   429→   429→  // Spring drop-in stagger (40ms per item, capped at 0.3s = ~8 items).
   430→   430→  const springDelay = reducedMotion ? 0 : Math.min(index * 0.04, 0.3);
   431→   431→
   432→   432→  const isPreviewOpen = isHover || isTapped;
   433→   433→
   434→   434→  return (
   435→   435→    <motion.li
   436→   436→      initial={reducedMotion ? false : { opacity: 0, y: 12 }}
   437→   437→      animate={{ opacity: 1, y: 0 }}
   438→   438→      transition={
   439→   439→        reducedMotion
   440→   440→          ? { duration: 0 }
   441→   441→          : { type: "spring", stiffness: 220, damping: 26, delay: springDelay }
   442→   442→      }
   443→   443→    >
