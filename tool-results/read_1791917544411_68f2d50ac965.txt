   150→function namedOwnerForGroup(group: string): { name: string; nameAr?: string; role: string; group: string } | null {
   151→  const o = ALL_OWNERS.find((x) => x.group === group);
   152→  if (!o) return null;

   156→function stageLabel(stageKey: string, journeyType: string, locale: "en" | "ar"): string {
   157→  const st = ALL_STAGES.find((s) => s.key === stageKey && s.journeyType === journeyType);
   158→  if (!st) return stageKey;
   159→  return locale === "ar" ? st.labelAr : st.label;
   160→}

   162→function stageShort(stageKey: string, journeyType: string, locale: "en" | "ar"): string {
   163→  const full = stageLabel(stageKey, journeyType, locale);
   164→  // Compact rail labels: take first 1–2 words
   165→  if (locale === "ar") return full;
   166→  const parts = full.split(" ");
   167→  return parts.length > 2 ? parts.slice(0, 2).join(" ") : full;
   168→}

   170→function fmtDateAndDistance(iso: string | undefined, locale: "en" | "ar"): string {
   171→  if (!iso) return locale === "ar" ? "في أي وقت" : "any time";
   172→  const d = new Date(iso);
   173→  const datePart = format(d, "EEEE d MMM");
   174→  // Use synthetic NOW from pack for deterministic "in 3 days" copy
   175→  const distance = formatDistance(d, PACK_NOW, { addSuffix: false });
   176→  return locale === "ar" ? `${datePart} · ${distance}` : `${datePart} · ${distance}`;
   177→}

   179→function lateCopy(step: Step, locale: "en" | "ar"): string {
   180→  if (step.breachLevel === "hard") {
   181→    const days = Math.max(1, Math.floor((step.elapsedHours - step.targetHours) / 24));
   182→    return locale === "ar"
   183→      ? `متأخر بـ ${days} أيام — خرق جوهري`
   184→      : `${days} day${days === 1 ? "" : "s"} late — hard breach`;
   185→  }
   186→  if (step.breachLevel === "breached") {
   187→    const days = Math.max(1, Math.floor((step.elapsedHours - step.targetHours) / 24));
   188→    return locale === "ar" ? `متأخر بـ ${days} أيام` : `${days} day${days === 1 ? "" : "s"} late`;
   189→  }
   190→  if (step.status === "blocked") {
   191→    return locale === "ar"
   192→      ? "محظور بانتظار خطوات أخرى"
   193→      : "Held up — waiting on another step";
   194→  }
   195→  return "";
   196→}

   198→function ownerLine(step: Step, locale: "en" | "ar"): string {
   199→  const grp = OWNER_GROUP_LABEL[step.ownerGroup];
   200→  const named = namedOwnerForGroup(step.ownerGroup);
   201→  if (!grp) return step.ownerGroup;
   202→  if (named) {
   203→    return locale === "ar" && named.nameAr
   204→      ? `مع ${named.nameAr} في ${grp.ar}`
   205→      : `with ${named.name} in ${grp.en}`;
   206→  }
   207→  return locale === "ar" ? `مع ${grp.ar}` : `with ${grp.en}`;
   208→}

   497→/* Parse "docSlug#section" → CitationChip shape */
   498→function parsePolicyRef(ref: string): { docSlug: string; title: string; section: string; effectiveDate: string } {
   499→  const [docSlug, section] = ref.split("#");
   500→  const titleMap: Record<string, string> = {
   501→    "relocation-policy": "Relocation Policy",
   502→    "immigration-policy": "Immigration Policy",
   503→    "it-policy": "IT Provisioning Policy",
   504→    "onboarding-policy": "Onboarding Policy",
   505→    "learning-policy": "Learning Policy",
   506→    "mobility-policy": "Internal Mobility Policy",
   507→    "payroll-policy": "Payroll Policy",
   508→    "facilities-policy": "Facilities Policy",

   631→/* ---------- Stage Rail ---------- */

   633→function StageRail({ data }: { data: JourneyMeData }) {
   634→  const { locale } = useAppStore();
   635→  const journeyType = data.employee.journeyType;
   636→  const journeyStages = useMemo(
   637→    () => ALL_STAGES.filter((s) => s.journeyType === journeyType).sort((a, b) => a.order - b.order),
   638→    [journeyType]
   639→  );
   640→
   641→  const segState = (stageKey: string): "done" | "active" | "blocked" | "pending" => {
   642→    const stepsInStage = data.steps.filter((s) => s.stage === stageKey);
   643→    if (stepsInStage.length === 0) return "pending";
   644→    const allDone = stepsInStage.every((s) => s.status === "done" || s.status === "skipped");
   645→    if (allDone) return "done";
   646→    const anyBlocked = stepsInStage.some(
   647→      (s) => s.status === "blocked" || s.breachLevel === "hard"
   648→    );
   649→    if (anyBlocked) return "blocked";
   650→    if (stageKey === data.stage) return "active";
   651→    // Stage before current — if not all done, mark done visually (history)
   652→    const stageOrder = journeyStages.find((s) => s.key === stageKey)?.order ?? 0;
   653→    const currentOrder = journeyStages.find((s) => s.key === data.stage)?.order ?? 0;
   654→    if (stageOrder < currentOrder) return "done";
   655→    return "pending";
   656→  };
   657→
   658→  return (
   659→    <section className="ast-card p-5 sm:p-6" aria-label="Stage rail">
   660→      <div className="flex items-center gap-2 mb-4">
   661→        <Compass className="w-4 h-4" strokeWidth={1.75} style={{ color: "var(--brand-400)" }} />
   662→        <h3 className="text-[14px] font-semibold uppercase tracking-[0.12em]" style={{ color: "var(--text-hi)" }}>
   663→          {locale === "ar" ? "مسار الرحلة" : "Journey rail"}
   664→        </h3>
   665→      </div>
   666→      <ol className="grid gap-2" style={{ gridTemplateColumns: `repeat(${journeyStages.length}, minmax(0, 1fr))` }}>
   667→        {journeyStages.map((st) => {
   668→          const state = segState(st.key);
   669→          const isActive = st.key === data.stage;
   670→          const label = locale === "ar" ? st.labelAr : st.label;
   671→          return (
   672→            <li key={st.key} className="flex flex-col gap-2 min-w-0">
   673→              <div
   674→                className="ast-rail-seg"
   675→                data-state={state === "pending" ? undefined : state}
   676→                aria-label={`${label}: ${state}`}
   677→                style={state === "pending" ? { background: "var(--surface-3)" } : undefined}
   678→              />
   679→              <div className="min-w-0">
   680→                <p
   681→                  className="text-[11px] sm:text-[12px] font-medium leading-tight truncate"
   682→                  style={{
   683→                    color: isActive ? "var(--accent-500)" : state === "done" ? "var(--brand-400)" : state === "blocked" ? "var(--status-blocked)" : "var(--text-lo)",
   684→                  }}
   685→                >
   686→                  {stageShort(st.key, journeyType, locale)}

  1002→        toast.success(locale === "ar" ? "تم تسجيل البدء" : "Step started", {
  1003→          description: locale === "ar" ? nbs.labelAr : nbs.label,
  1004→        });
  1005→        return;
  1006→      }
  1007→      const idempotencyKey = `idm-${emp.id}-${requestType}-${Date.now().toString(36)}`;
  1008→      try {
  1009→        const res = await fetch("/api/v1/requests", {
  1010→          method: "POST",
  1011→          headers: { "Content-Type": "application/json" },
  1012→          body: JSON.stringify({
  1013→            employeeId: emp.id,
  1014→            type: requestType,
  1015→            payload: { stepKey: nbs.stepKey, label: nbs.label },
  1016→            idempotencyKey,
  1017→          }),
  1018→        });
  1019→        const json: ApiEnvelope<{ ref: string; owner?: { name: string } }> = await res.json();
  1020→        if (json.ok && json.data) {
  1021→          const ownerName = json.data.owner?.name ?? "the team";
  1022→          toast.success(locale === "ar" ? "تم إرسال الطلب" : "Request sent", {
  1023→            description: locale === "ar"
  1024→              ? `${nbs.labelAr} — ${ownerName} سيتولى الأمر`
  1025→              : `${nbs.label} — ${ownerName} will handle it`,
  1026→          });
  1027→        } else {
  1028→          toast.error(locale === "ar" ? "تعذّر الإرسال" : "Couldn't send", {
  1029→            description: json.error?.message ?? "Please try again.",
  1030→          });
  1031→        }
  1032→      } catch {
  1033→        toast.error(locale === "ar" ? "خطأ في الشبكة" : "Network error", {
  1034→          description: locale === "ar" ? "حاول مرة أخرى." : "Please retry.",
  1035→        });
  1036→      }
  1037→    },
  1038→    [data, locale]
  1039→  );
  1040→
  1041→  const handleGrantConsent = useCallback(async () => {
  1042→    try {
  1043→      const res = await fetch("/api/v1/consent", {
  1044→        method: "POST",
  1045→        headers: { "Content-Type": "application/json" },
  1046→        body: JSON.stringify({
  1047→          employeeId,
  1048→          purpose: "pulse_capture",
  1049→          granted: true,
  1050→          version: "v1",
  1051→        }),
  1052→      });
  1053→      const json: ApiEnvelope<{ receiptId: string }> = await res.json();
  1054→      if (json.ok) {
  1055→        setPulseGranted(true);
  1056→        toast.success(locale === "ar" ? "تم تفعيل التقدير" : "Pulse capture enabled", {
  1057→          description: locale === "ar"
  1058→            ? `رقم الإيصال: ${json.data?.receiptId ?? ""}`
  1059→            : `Receipt: ${json.data?.receiptId ?? ""}`,
  1060→        });
  1061→      } else {
  1062→        toast.error(json.error?.message ?? "Failed");
  1063→      }
  1064→    } catch {
  1065→      toast.error(locale === "ar" ? "خطأ في الشبكة" : "Network error");
  1066→    }
  1067→  }, [employeeId, locale]);
  1068→
  1069→  const handleSubmitPulse = useCallback(
  1070→    async (score: number) => {
  1071→      if (!data) return;
  1072→      try {
  1073→        const res = await fetch("/api/v1/pulse", {
  1074→          method: "POST",
  1075→          headers: { "Content-Type": "application/json" },
  1076→          body: JSON.stringify({
  1077→            employeeId,
  1078→            stage: data.stage,
  1079→            score,
  1080→            consentId: `CN-pulse-${employeeId}`,
  1081→          }),
  1082→        });
  1083→        const json: ApiEnvelope<{ id: string }> = await res.json();
  1084→        if (json.ok) {
  1085→          toast.success(locale === "ar" ? "تم الإرسال" : "Pulse recorded", {
  1086→            description: locale === "ar" ? "شكراً لمشاركتك." : "Thank you for sharing.",
  1087→          });
  1088→        } else if (json.error?.code === "E_CONSENT_REQUIRED") {
  1089→          setPulseGranted(false);
  1090→          toast.error(locale === "ar" ? "يلزم الموافقة" : "Consent required", {
  1091→            description: json.error.hint ?? "",
  1092→          });
  1093→        } else {
  1094→          toast.error(json.error?.message ?? "Failed");
  1095→        }
  1096→      } catch {
  1097→        toast.error(locale === "ar" ? "خطأ في الشبكة" : "Network error");
  1098→      }
  1099→    },
  1100→    [data, employeeId, locale]
  1101→  );
  1102→
  1103→  /* Render ---------------------------------------------------- */
  1104→
  1105→  return (
  1106→    <div
  1107→      dir={dir}
  1108→      className="min-h-screen flex flex-col"
  1109→      style={{ background: "var(--surface-0)" }}
  1110→    >
  1111→      <main className="flex-1 w-full max-w-[1180px] mx-auto px-4 sm:px-6 lg:px-8 py-6 sm:py-8 space-y-6">
  1112→        {loading ? (
  1113→          <TodaySkeleton />
  1114→        ) : error ? (
  1115→          <ErrorBanner
  1116→            message={error.message}
  1117→            requestId={error.requestId}
  1118→            onRetry={() => void load()}
  1119→          />
  1120→        ) : data ? (
  1121→          <>
  1122→            <HeroStrip data={data} />
  1123→
  1124→            <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
  1125→              <Tilt3D maxDeg={3} className="lg:col-span-7">
  1126→                <NextBestStepCard data={data} onStart={handleStartStep} />
  1127→              </Tilt3D>
  1128→              <LateItemsPanel data={data} />
  1129→            </div>
  1130→
  1131→            <StageRail data={data} />
  1132→
  1133→            <ComingUp data={data} />
  1134→
  1135→            <PulseCard
  1136→              employeeId={employeeId}
  1137→              stage={data.stage}
  1138→              pulseGranted={pulseGranted}
  1139→              onGrantConsent={handleGrantConsent}
  1140→              onSubmit={handleSubmitPulse}
  1141→              locale={locale}
  1142→            />
  1143→
  1144→            {/* Version footer */}
  1145→            <p
  1146→              className="text-[11px] font-mono tabular text-center pt-2 pb-4"
  1147→              style={{ color: "var(--text-lo)" }}
  1148→            >
  1149→              {data.engineVersion} · {data.mapVersion} · {data.corpusVersion}
  1150→            </p>
  1151→          </>
  1152→        ) : null}
  1153→      </main>
  1154→    </div>
  1155→  );
  1156→}
