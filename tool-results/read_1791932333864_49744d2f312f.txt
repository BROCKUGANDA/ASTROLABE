   164→
   165→function computeBreach(
   166→  elapsedHours: number,
   167→  targetHours: number,
   168→): "none" | "due_soon" | "breached" | "hard" {
   169→  if (elapsedHours >= targetHours * 2) return "hard";
   170→  if (elapsedHours >= targetHours) return "breached";
   171→  if (targetHours > 0 && elapsedHours / targetHours >= 0.75) return "due_soon";
   172→  return "none";
   173→}
   174→
   175→function namedAssignee(assigneeId: string): { name: string; role: string; email: string; group: string } | null {
   176→  const o = getOwner(assigneeId);
   177→  if (!o) return null;
   178→  return { name: o.name, role: o.role, email: o.email, group: o.group };
   179→}
   180→
   181→function resolveOwnerForType(type: RequestType): { name: string; role: string; email: string; group: string } | null {
   182→  const schema = INTAKE_SCHEMAS[type];
   183→  const o = ALL_OWNERS.find((x) => x.group === schema.ownerGroup);
   184→  if (!o) return null;
   185→  return { name: o.name, role: o.role, email: o.email, group: o.group };
   186→}

   188→function parsePolicyRef(ref: string): { docSlug: string; title: string; section: string; effectiveDate: string } {
   189→  const [docSlug, section] = ref.split("#");
   190→  const titleMap: Record<string, string> = {
   191→    "relocation-policy": "Relocation Policy",
   192→    "immigration-policy": "Immigration Policy",
   193→    "it-policy": "IT Provisioning Policy",
   194→    "onboarding-policy": "Onboarding Policy",
   195→    "learning-policy": "Learning Policy",
   196→    "mobility-policy": "Internal Mobility Policy",
   197→    "payroll-policy": "Payroll Policy",
   198→    "facilities-policy": "Facilities Policy",
   199→  };
   200→  return {
   201→    docSlug,
   202→    title: titleMap[docSlug] ?? docSlug,
   203→    section: section ?? "—",
   204→    effectiveDate: "2024-01-01",
   205→  };
   206→}

   212→export function RequestsScreen() {
   213→  const locale = useAppStore((s) => s.locale);
   214→  const currentEmployeeId = useAppStore((s) => s.currentEmployeeId);
   215→  const activeRequestType = useAppStore((s) => s.activeRequestType);
   216→  const setActiveRequestType = useAppStore((s) => s.setActiveRequestType);
   217→
   218→  const [requests, setRequests] = useState<ServiceRequest[]>([]);
   219→  const [loading, setLoading] = useState(true);
   220→  const [error, setError] = useState<string | null>(null);
   221→  const [expandedId, setExpandedId] = useState<string | null>(null);
   222→
   223→  // Wizard state
   224→  const [wizardOpen, setWizardOpen] = useState(false);
   225→  const [wizardStep, setWizardStep] = useState<0 | 1 | 2>(0);
   226→  const [wizardType, setWizardType] = useState<RequestType | null>(null);
   227→  const [wizardPrefill, setWizardPrefill] = useState<Record<string, string>>({});
   228→  const [duplicateNotice, setDuplicateNotice] = useState<string | null>(null);
   229→
   230→  // Employee record for prefill
   231→  const [employee, setEmployee] = useState<Employee | null>(null);
   232→  const [journeyStage, setJourneyStage] = useState<string>("");
   233→
   234→  const fetchRequests = useCallback(async () => {
   235→    setLoading(true);
   236→    setError(null);
   237→    try {
   238→      const res = await fetch(
   239→        `/api/v1/requests?employeeId=${encodeURIComponent(currentEmployeeId)}`,
   240→      );
   241→      const json: ApiEnvelope<{ items: ServiceRequest[]; nextCursor: string | null }> =
   242→        await res.json();
   243→      if (!json.ok || !json.data) {
   244→        throw new Error(json.error?.message ?? "Failed to load requests");
   245→      }
   246→      setRequests(json.data.items);
   247→    } catch (err) {
   248→      setError(err instanceof Error ? err.message : "Unknown error");
   249→    } finally {
   250→      setLoading(false);
   251→    }
   252→  }, [currentEmployeeId]);
   253→
   254→  // Fetch employee record for prefill (from /api/v1/journey/me)
   255→  useEffect(() => {
   256→    let cancelled = false;
   257→    (async () => {
   258→      try {
   259→        const res = await fetch(
   260→          `/api/v1/journey/me?employeeId=${encodeURIComponent(currentEmployeeId)}&asOf=${encodeURIComponent(PACK_NOW.toISOString())}`,
   261→        );
   262→        const json: ApiEnvelope<JourneyMeData> = await res.json();
   263→        if (cancelled) return;
   264→        if (json.ok && json.data) {
   265→          setEmployee(json.data.employee);
   266→          setJourneyStage(json.data.stage);
   267→        }
   268→      } catch {
   269→        // Non-fatal — prefill just won't be available
   270→      }
   271→    })();
   272→    return () => {
   273→      cancelled = true;
   274→    };
   275→  }, [currentEmployeeId]);
   276→
   277→  useEffect(() => {
   278→    fetchRequests();
   279→  }, [fetchRequests]);
   280→
   281→  // Pick up activeRequestType from store (e.g. when user clicked
   282→  // "Start this request" from Ask or Today screen) → jump to step 2.
   283→  useEffect(() => {
   284→    if (activeRequestType && INTAKE_SCHEMAS[activeRequestType as RequestType]) {
   285→      setWizardType(activeRequestType as RequestType);
   286→      setWizardStep(1);
   287→      setWizardOpen(true);
   288→      // Pre-populate prefill map from employee record (only once per type)
   289→      if (employee) {
   290→        const pre: Record<string, string> = {};
   291→        const schema = INTAKE_SCHEMAS[activeRequestType as RequestType];
   292→        for (const field of schema.fields) {
   293→          if (!field.prefillFrom) continue;
   294→          if (field.prefillFrom === "employee.location") pre[field.key] = employee.location;
   295→          else if (field.prefillFrom === "employee.stage") pre[field.key] = journeyStage;
   296→          else if (field.prefillFrom === "employee.startDate") pre[field.key] = employee.startDate.slice(0, 10);
   297→          else if (field.prefillFrom === "employee.role") pre[field.key] = employee.role;
   298→        }
   299→        setWizardPrefill(pre);
   300→      }
   301→      // Clear the store flag so we don't re-open on next mount
   302→      setActiveRequestType(undefined);
   303→    }
   304→  }, [activeRequestType, employee, journeyStage, setActiveRequestType]);
   305→
   306→  const openWizard = () => {
   307→    setWizardType(null);
   308→    setWizardStep(0);
   309→    setWizardPrefill({});
   310→    setDuplicateNotice(null);
   311→    setWizardOpen(true);
   312→  };
   313→
   314→  const handlePickType = (type: RequestType) => {
   315→    setWizardType(type);
   316→    // Build prefill map fresh from employee record
   317→    const pre: Record<string, string> = {};
   318→    if (employee) {
   319→      const schema = INTAKE_SCHEMAS[type];
   320→      for (const field of schema.fields) {
   321→        if (!field.prefillFrom) continue;
   322→        if (field.prefillFrom === "employee.location") pre[field.key] = employee.location;
   323→        else if (field.prefillFrom === "employee.stage") pre[field.key] = journeyStage;
   324→        else if (field.prefillFrom === "employee.startDate") pre[field.key] = employee.startDate.slice(0, 10);
   325→        else if (field.prefillFrom === "employee.role") pre[field.key] = employee.role;
   326→      }
   327→    }
   328→    setWizardPrefill(pre);
   329→    setWizardStep(1);
   330→  };
   331→
   332→  type SubmitResult =
   333→    | { ok: true; id: string; duplicate: boolean }
   334→    | { ok: false; code: string; message: string };
   335→
   336→  const handleSubmit = async (
   337→    payload: Record<string, unknown>,
   338→  ): Promise<SubmitResult> => {
   339→    if (!wizardType) {
   340→      return { ok: false, code: "E_NO_TYPE", message: "No request type selected" };
   341→    }
   342→    const idempotencyKey = `idm-${currentEmployeeId}-${wizardType}-${Date.now().toString(36)}`;
   343→    try {
   344→      const res = await fetch("/api/v1/requests", {
   345→        method: "POST",
   346→        headers: { "Content-Type": "application/json" },
   347→        body: JSON.stringify({
   348→          employeeId: currentEmployeeId,
   349→          type: wizardType,
   350→          payload,
   351→          idempotencyKey,
   352→        }),
   353→      });
   354→      const json: ApiEnvelope<ServiceRequest & { servedFrom?: string }> = await res.json();
   355→      if (!json.ok || !json.data) {
   356→        const code = json.error?.code ?? "E_UNKNOWN";
   357→        const msg = json.error?.message ?? "Failed to create request";
   358→        if (json.error?.hint) {
   359→          toast.error(`${msg}`, { description: json.error.hint });
   360→        } else {
   361→          toast.error(msg);
   362→        }
   363→        return { ok: false, code, message: msg };
   364→      }
   365→
   366→      // Duplicate detection — idempotency replay returns servedFrom: "cache"
   367→      const isDuplicate = json.data.servedFrom === "cache";
   368→      await fetchRequests();
   369→      setWizardOpen(false);
   370→
   371→      if (isDuplicate) {
   372→        setDuplicateNotice(json.data.id);
   373→        setExpandedId(json.data.id);
   374→        toast.info(
   375→          locale === "ar"
   376→            ? "هذا الطلب موجود بالفعل — نعرض الأصلي."
   377→            : "This request already exists — showing the original.",
   378→        );
   379→      } else {
   380→        setExpandedId(json.data.id);
   381→        // Confetti-style toast
   382→        toast.success(
   383→          locale === "ar"
   384→            ? `تم إنشاء الطلب ${json.data.ref} ✨`
   385→            : `Request ${json.data.ref} created ✨`,
   386→          {
   387→            description:
   388→              locale === "ar"
   389→                ? "سيتواصل معك الفريق المالك قريباً."
   390→                : "The owning team will reach out shortly.",
   391→          },
   392→        );
   393→      }
   394→      return { ok: true, id: json.data.id, duplicate: isDuplicate };
   395→    } catch (err) {
   396→      const msg = err instanceof Error ? err.message : "Network error";
   397→      toast.error(msg);
   398→      return { ok: false, code: "E_NETWORK", message: msg };
   399→    }
   400→  };
   401→
   402→  // PRD §11-212 — when an employee answers a structured "needs more info"
   403→  // question and clicks Resubmit, transition the request back to `submitted`
   404→  // (which clears the active returnQuestion server-side) and refetch the
   405→  // list so the card disappears.
   406→  const handleResubmit = useCallback(
   407→    async (reqId: string, answer: string): Promise<void> => {
   408→      try {
   409→        const res = await fetch(`/api/v1/requests/${reqId}/transition`, {
   410→          method: "POST",
   411→          headers: { "Content-Type": "application/json" },
   412→          body: JSON.stringify({
   413→            to: "submitted",
   414→            idempotencyKey: `${reqId}-resubmit-${Date.now()}`,
   415→            answer,
   416→          }),
   417→        });
   418→        const json: ApiEnvelope<ServiceRequest> = await res.json();
   419→        if (!res.ok || !json.ok) {
   420→          throw new Error(json.error?.message ?? `HTTP ${res.status}`);
   421→        }
   422→        await fetchRequests();
   423→        toast.success(
   424→          locale === "ar"
   425→            ? "أُرسلت إلى IT — سيتولّونها الآن."
   426→            : "Sent back to IT — they'll pick it up now.",
   427→        );
   428→      } catch (err) {
   429→        toast.error(
   430→          locale === "ar" ? "تعذّر إعادة الإرسال" : "Couldn't resubmit",
   431→          { description: err instanceof Error ? err.message : "Network error" },
   432→        );
   433→      }
   434→    },
   435→    [fetchRequests, locale],
   436→  );
   437→
   438→  return (
   439→    <div className="max-w-[1180px] mx-auto px-4 sm:px-6 lg:px-8 py-6">
   440→      {/* Header */}
   441→      <div className="mb-5 flex items-center justify-between gap-3">
   442→        <div className="flex items-center gap-3 min-w-0">
   443→          <div
   444→            className="flex items-center justify-center w-10 h-10 rounded-full shrink-0"
   445→            style={{ background: "var(--g-veil)", color: "var(--brand-400)" }}
   446→            aria-hidden="true"
   447→          >
   448→            <Inbox className="w-5 h-5" strokeWidth={1.75} />
   449→          </div>
   450→          <div className="min-w-0">
   451→            <h1
   452→              className="text-[20px] sm:text-[22px] font-semibold leading-tight"
   453→              style={{ color: "var(--text-hi)" }}
   454→            >
   455→              {locale === "ar" ? "طلباتي" : "My requests"}
   456→            </h1>
   457→            <p className="text-[13px] mt-0.5" style={{ color: "var(--text-mid)" }}>
   458→              {locale === "ar"
   459→                ? "تتبّع طلباتك المفتوحة وابدأ طلباً جديداً."
   460→                : "Track your open requests and start a new one."}
   461→            </p>
   462→          </div>
   463→        </div>
   464→        <button
   465→          onClick={openWizard}
   466→          className="ast-focus inline-flex items-center justify-center gap-2 h-11 px-4 sm:px-5 rounded-[10px] text-[13px] sm:text-[14px] font-semibold shrink-0 transition-transform hover:-translate-y-0.5"
   467→          style={{
   468→            background: "var(--brand-500)",
   469→            color: "var(--surface-0)",
   470→            boxShadow: "0 4px 14px rgba(22,154,165,0.28)",
   471→          }}
   472→        >
   473→          <Plus className="w-4 h-4" strokeWidth={2} />
   474→          <span className="hidden sm:inline">
   475→            {locale === "ar" ? "طلب جديد" : "New request"}
   476→          </span>
   477→          <span className="sm:hidden">{locale === "ar" ? "جديد" : "New"}</span>
   478→        </button>
   479→      </div>
   480→
   481→      {/* Duplicate notice */}
   482→      {duplicateNotice && (
   483→        <div
   484→          className="ast-card p-3 mb-4 flex items-center gap-2"
   485→          style={{
   486→            borderColor: "rgba(22,154,165,0.30)",
   487→            background: "rgba(22,154,165,0.06)",
   488→          }}
   489→        >
   490→          <CheckCircle2
   491→            className="w-4 h-4 shrink-0"
   492→            strokeWidth={1.75}
   493→            style={{ color: "var(--brand-400)" }}
   494→          />
   495→          <p className="text-[13px]" style={{ color: "var(--text-hi)" }}>
   496→            {locale === "ar"
   497→              ? `هذا الطلب موجود بالفعل — نعرض الأصلي (${duplicateNotice}).`
   498→              : `This request already exists — showing the original (${duplicateNotice}).`}
   499→          </p>
   500→          <button
   501→            onClick={() => setDuplicateNotice(null)}
   502→            className="ast-focus ml-auto shrink-0"
   503→            aria-label={locale === "ar" ? "إغلاق" : "Dismiss"}
   504→          >
   505→            <X className="w-4 h-4" strokeWidth={1.75} style={{ color: "var(--text-lo)" }} />
   506→          </button>
   507→        </div>
   508→      )}
   509→
   510→      {/* Layout: list (left col-7) + wizard (right col-5) */}
   511→      <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
   512→        {/* List */}
   513→        <section
   514→          className="lg:col-span-7 order-2 lg:order-1"
   515→          aria-label={locale === "ar" ? "قائمة الطلبات" : "Request list"}
   516→        >
   517→          {loading ? (
   518→            <RequestListSkeleton />
   519→          ) : error ? (
   520→            <ErrorBanner
   521→              message={error}
   522→              onRetry={fetchRequests}
   523→              locale={locale}
   524→            />
   525→          ) : requests.length === 0 ? (
   526→            <EmptyState
   527→              onPickType={handlePickType}
   528→              locale={locale}
   529→            />
   530→          ) : (
   531→            <ul className="space-y-3">
   532→              <AnimatePresence initial={false}>
   533→                {requests.map((req, i) => (
   534→                  <RequestCard
   535→                    key={req.id}
   536→                    req={req}
   537→                    expanded={expandedId === req.id}
   538→                    onToggle={() =>
   539→                      setExpandedId((cur) => (cur === req.id ? null : req.id))

   541→                    onResubmit={handleResubmit}
   542→                    locale={locale}

   546→              </AnimatePresence>
   547→            </ul>
   548→          )}
   549→        </section>
   550→
   551→        {/* Wizard — desktop right column; mobile modal */}

   874→function RequestCard({
   875→  req,
   876→  expanded,
   877→  onToggle,
   878→  onResubmit,
   879→  locale,
   880→  index,
   881→}: {
   882→  req: ServiceRequest;
   883→  expanded: boolean;
   884→  onToggle: () => void;
   885→  onResubmit: (reqId: string, answer: string) => Promise<void>;
   886→  locale: "en" | "ar";
   887→  index: number;
   888→}) {
   889→  const [copied, setCopied] = useState(false);
   890→  const schema = INTAKE_SCHEMAS[req.type];
   891→  const typeLabel = schema ? (locale === "ar" ? schema.labelAr : schema.label) : req.type;
   892→  const TypeIcon = TYPE_ICONS[req.type] ?? Inbox;
   893→
   894→  // Compute elapsed from openedAt vs slaHours — use PACK_NOW as deterministic asOf
   895→  const elapsedHours = Math.max(
   896→    0,
   897→    (PACK_NOW.getTime() - new Date(req.openedAt).getTime()) / 3_600_000,
   898→  );
   899→  const breach = computeBreach(elapsedHours, req.slaHours);
   900→
   901→  const assignee = namedAssignee(req.assignee);
   902→  const ownerGroupLabel = OWNER_GROUP_LABEL[req.ownerGroup];
   903→  const ownerGroupText = ownerGroupLabel
   904→    ? locale === "ar"
   905→      ? ownerGroupLabel.ar
   906→      : ownerGroupLabel.en
   907→    : req.ownerGroup;
   908→
   909→  const handleCopyRef = async (e?: React.MouseEvent) => {
   910→    e?.stopPropagation();
   911→    try {
   912→      await navigator.clipboard.writeText(req.ref);
   913→      setCopied(true);
   914→      setTimeout(() => setCopied(false), 1600);
   915→      toast.success(locale === "ar" ? "تم نسخ المرجع" : "Reference copied");
   916→    } catch {
   917→      toast.error(locale === "ar" ? "تعذّر النسخ" : "Couldn't copy");
   918→    }
   919→  };
   920→
   921→  // PRD §11-212 — when the request has been returned WITH a structured
   922→  // question, show the prominent "We need one more detail" banner ABOVE the
   923→  // normal request details. Sand-gold accent — this is a "your next move"
   924→  // trigger, NOT an error.
   925→  const showReturnQuestion =
   926→    req.status === "returned" && !!req.returnQuestion;
   927→
   928→  return (
   929→    <motion.li
   930→      initial={{ opacity: 0, y: 8 }}
   931→      animate={{ opacity: 1, y: 0 }}
   932→      exit={{ opacity: 0, y: -8 }}
   933→      transition={{ duration: 0.18, delay: Math.min(index * 0.04, 0.2) }}
   934→      className="ast-card overflow-hidden"
   935→    >
   936→      {showReturnQuestion && req.returnQuestion && (
   937→        <ReturnQuestionBanner
   938→          req={req}
   939→          ownerGroupText={ownerGroupText}
   940→          onResubmit={onResubmit}
   941→          locale={locale}
   942→        />
   943→      )}
   944→      {/* Outer clickable area uses role=button on a div to avoid nested <button> hydration errors */}
   945→      <div
   946→        onClick={onToggle}
   947→        onKeyDown={(e) => {
   948→          if (e.key === "Enter" || e.key === " ") {
   949→            e.preventDefault();
   950→            onToggle();
   951→          }
   952→        }}
   953→        role="button"
   954→        tabIndex={0}
   955→        aria-expanded={expanded}
   956→        className="ast-focus w-full text-left p-4 sm:p-5 cursor-pointer"
   957→      >
   958→        <div className="flex items-start justify-between gap-3">
   959→          <div className="flex items-start gap-3 min-w-0 flex-1">
   960→            <div
   961→              className="flex items-center justify-center w-9 h-9 rounded-full shrink-0"
   962→              style={{
   963→                background: "rgba(22,154,165,0.10)",
   964→                color: "var(--brand-400)",
   965→              }}
   966→              aria-hidden="true"
   967→            >
   968→              <TypeIcon className="w-4 h-4" strokeWidth={1.75} />
   969→            </div>
   970→            <div className="min-w-0 flex-1">
   971→              <p
   972→                className="text-[15px] font-semibold leading-snug truncate"
   973→                style={{ color: "var(--text-hi)" }}
   974→              >
   975→                {typeLabel}
   976→              </p>
   977→              <div className="mt-1.5 flex items-center flex-wrap gap-2">
   978→                <button
   979→                  type="button"
   980→                  onClick={(e) => {
   981→                    e.stopPropagation();
   982→                    handleCopyRef();
   983→                  }}
   984→                  className="ast-focus inline-flex items-center gap-1 h-6 px-2 rounded-md text-[10px] font-mono tabular transition-colors hover:bg-[var(--surface-2)]"
   985→                  style={{ color: "var(--text-lo)" }}

  1217→/* ============================================================
  1218→   Wizard (3 steps)

  1220→
  1221→function Wizard({
  1222→  step,
  1223→  type,
  1224→  prefill,
  1225→  onStepChange,
  1226→  onPickType,
  1227→  onSubmit,
  1228→  onReset,
  1229→  locale,
  1230→  employeeId,
  1231→}: {
  1232→  step: 0 | 1 | 2;
  1233→  type: RequestType | null;
  1234→  prefill: Record<string, string>;
  1235→  onStepChange: (s: 0 | 1 | 2) => void;
  1236→  onPickType: (t: RequestType) => void;
  1237→  onSubmit: (payload: Record<string, unknown>) => Promise<{ ok: boolean; code?: string; message?: string; id?: string; duplicate?: boolean }>;
  1238→  onReset: () => void;
  1239→  locale: "en" | "ar";
  1240→  employeeId: string;
  1241→}) {
  1242→  const schema: IntakeSchema | null = type ? INTAKE_SCHEMAS[type] : null;
  1243→  // The Wizard is keyed by `type` in the parent so it remounts
  1244→  // whenever the type changes, ensuring this initializer re-runs.
  1245→  const [payload, setPayload] = useState<Record<string, unknown>>(() => {
  1246→    if (!type) return {};
  1247→    const s = INTAKE_SCHEMAS[type];
  1248→    const initial: Record<string, unknown> = {};
  1249→    for (const field of s.fields) {
  1250→      if (prefill[field.key] !== undefined) {
  1251→        initial[field.key] = prefill[field.key];
  1252→      } else if (field.type === "checkbox") {
  1253→        initial[field.key] = false;
  1254→      } else {
  1255→        initial[field.key] = "";
  1256→      }
  1257→    }
  1258→    return initial;
  1259→  });
  1260→  const [errors, setErrors] = useState<Record<string, string>>({});
  1261→  const [submitError, setSubmitError] = useState<string | null>(null);
  1262→  const [submitting, setSubmitting] = useState(false);
  1263→
  1264→  const validateField = (field: IntakeField, value: unknown): string => {
  1265→    if (field.required) {
  1266→      if (value === undefined || value === null || value === "" || value === false) {
  1267→        return locale === "ar"
  1268→          ? `${locale === "ar" ? field.labelAr : field.label} مطلوب`
  1269→          : `${field.label} is required`;
  1270→      }
  1271→    }
  1272→    if (field.type === "number" && value !== "" && value !== undefined) {
  1273→      const n = Number(value);
  1274→      if (Number.isNaN(n)) {
  1275→        return locale === "ar" ? "يجب أن يكون رقماً" : "Must be a number";
  1276→      }
  1277→      if (n < 0) return locale === "ar" ? "لا يمكن أن يكون سالباً" : "Cannot be negative";
  1278→    }
  1279→    if (field.type === "date" && value && field.key === "effective_date") {
  1280→      const d = new Date(value as string);
  1281→      if (Number.isNaN(d.getTime())) {
  1282→        return locale === "ar" ? "تاريخ غير صالح" : "Invalid date";
  1283→      }
  1284→    }
  1285→    return "";
  1286→  };
  1287→
  1288→  const validateAll = (): boolean => {
  1289→    if (!schema) return false;
  1290→    const next: Record<string, string> = {};
  1291→    for (const field of schema.fields) {
  1292→      const err = validateField(field, payload[field.key]);
  1293→      if (err) next[field.key] = err;
  1294→    }
  1295→    setErrors(next);
  1296→    return Object.keys(next).length === 0;
  1297→  };
  1298→
  1299→  const handleFieldChange = (key: string, value: unknown) => {
  1300→    setPayload((prev) => ({ ...prev, [key]: value }));
  1301→    // Inline validation as you type — clear error on change, re-validate if it had one
  1302→    if (errors[key]) {
  1303→      const field = schema?.fields.find((f) => f.key === key);
  1304→      if (field) {
  1305→        const err = validateField(field, value);
  1306→        setErrors((prev) => {
  1307→          const next = { ...prev };
  1308→          if (err) next[key] = err;
  1309→          else delete next[key];
  1310→          return next;
  1311→        });
  1312→      }
  1313→    }
  1314→  };
  1315→
  1316→  const handleSubmit = async () => {
  1317→    if (!schema) return;
  1318→    if (!validateAll()) {
  1319→      toast.error(
  1320→        locale === "ar"
  1321→          ? "يرجى تصحيح الحقول المطلوبة"
  1322→          : "Please fix the required fields",
  1323→      );
  1324→      return;
  1325→    }
  1326→    setSubmitting(true);
  1327→    setSubmitError(null);
  1328→    const result = await onSubmit(payload);
  1329→    setSubmitting(false);
  1330→    if (!result.ok) {
  1331→      setSubmitError(result.message ?? "Failed to submit");
  1332→    }
  1333→  };
  1334→
  1335→  return (
  1336→    <div>
  1337→      {/* Step indicator */}
  1338→      <div className="flex items-center gap-1.5 mb-4">
  1339→        {[0, 1, 2].map((i) => (
  1340→          <div
  1341→            key={i}
  1342→            className="flex-1 h-1 rounded-full"
  1343→            style={{
  1344→              background:
  1345→                i <= step ? "var(--brand-500)" : "var(--surface-3)",
  1346→            }}
  1347→          />
  1348→        ))}
  1349→      </div>
  1350→
  1351→      <AnimatePresence mode="wait">
  1352→        {step === 0 && (
  1353→          <motion.div
  1354→            key="step0"
  1355→            initial={{ opacity: 0, x: 8 }}
  1356→            animate={{ opacity: 1, x: 0 }}
  1357→            exit={{ opacity: 0, x: -8 }}
  1358→            transition={{ duration: 0.18 }}
  1359→          >
  1360→            <Step1TypeGrid onPick={onPickType} locale={locale} />
  1361→          </motion.div>
  1362→        )}
  1363→        {step === 1 && schema && (
  1364→          <motion.div
  1365→            key="step1"
  1366→            initial={{ opacity: 0, x: 8 }}
  1367→            animate={{ opacity: 1, x: 0 }}
  1368→            exit={{ opacity: 0, x: -8 }}
  1369→            transition={{ duration: 0.18 }}
  1370→          >
  1371→            <Step2Form
  1372→              schema={schema}
  1373→              payload={payload}
  1374→              errors={errors}
  1375→              prefill={prefill}
  1376→              onChange={handleFieldChange}
  1377→              onBack={() => {
  1378→                onReset();
  1379→                onStepChange(0);
  1380→              }}
  1381→              onNext={() => {
  1382→                if (validateAll()) onStepChange(2);
  1383→                else
  1384→                  toast.error(
  1385→                    locale === "ar"
  1386→                      ? "يرجى تصحيح الحقول المطلوبة"
  1387→                      : "Please fix the required fields",
  1388→                  );
  1389→              }}
  1390→              locale={locale}
  1391→            />
  1392→          </motion.div>
  1393→        )}
  1394→        {step === 2 && schema && (
  1395→          <motion.div
  1396→            key="step2"
  1397→            initial={{ opacity: 0, x: 8 }}
  1398→            animate={{ opacity: 1, x: 0 }}
  1399→            exit={{ opacity: 0, x: -8 }}
  1400→            transition={{ duration: 0.18 }}
  1401→          >
  1402→            <Step3Preview
  1403→              schema={schema}
  1404→              payload={payload}
  1405→              onBack={() => onStepChange(1)}
  1406→              onSubmit={handleSubmit}
  1407→              submitting={submitting}
  1408→              submitError={submitError}
  1409→              locale={locale}
  1410→              employeeId={employeeId}
  1411→            />
  1412→          </motion.div>
  1413→        )}
  1414→      </AnimatePresence>
  1415→    </div>
  1416→  );
  1417→}
