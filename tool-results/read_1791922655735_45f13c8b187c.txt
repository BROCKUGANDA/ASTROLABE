   495→/* ============================================================
   496→   HERO PARALLAX — 3D astrolabe + hero text
   497→   - Astrolabe canvas shifts max 8px opposite the cursor
   498→   - Hero text shifts max 4px with the cursor
   499→   - 60fps via framer-motion useMotionValue + useTransform
   500→   - Disabled on touch devices and when prefers-reduced-motion
   501→   ============================================================ */
   502→
   503→function usePrefersReducedMotion(): boolean {
   504→  const [reduce, setReduce] = useState(false);
   505→  useEffect(() => {
   506→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   507→    const update = () => setReduce(mq.matches);
   508→    update();
   509→    mq.addEventListener("change", update);
   510→    return () => mq.removeEventListener("change", update);
   511→  }, []);
   512→  return reduce;
   513→}
   514→
   515→function useFinePointer(): boolean {
   516→  const [fine, setFine] = useState(false);
   517→  useEffect(() => {
   518→    const mq = window.matchMedia("(pointer: fine)");
   519→    const update = () => setFine(mq.matches);
   520→    update();
   521→    mq.addEventListener("change", update);
   522→    return () => mq.removeEventListener("change", update);
   523→  }, []);
   524→  return fine;
   525→}

   527→function HeroParallax() {
   528→  const locale = useAppStore((s) => s.locale);
   529→  const setView = useAppStore((s) => s.setView);
   530→  const reduce = usePrefersReducedMotion();
   531→  const fine = useFinePointer();
   532→  const enabled = !reduce && fine;
   533→
   534→  // Normalised cursor offset from the centre of the hero, in [-0.5, 0.5].
   535→  const mx = useMotionValue(0);
   536→  const my = useMotionValue(0);
   537→
   538→  // Astrolabe: 8px in the opposite direction of the cursor.
   539→  const astX: MotionValue<number> = useTransform(mx, [-0.5, 0.5], [8, -8]);

   541→  // Hero text: 4px in the same direction as the cursor.
   542→  const textX: MotionValue<number> = useTransform(mx, [-0.5, 0.5], [-4, 4]);
   543→  const textY: MotionValue<number> = useTransform(my, [-0.5, 0.5], [-4, 4]);
   544→
   545→  const handleMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
   546→    if (!enabled) return;
   547→    const rect = e.currentTarget.getBoundingClientRect();
   548→    mx.set((e.clientX - rect.left) / rect.width - 0.5);
   549→    my.set((e.clientY - rect.top) / rect.height - 0.5);
   550→  };
   551→
   552→  const handleMouseLeave = () => {
   553→    mx.set(0);
   554→    my.set(0);
