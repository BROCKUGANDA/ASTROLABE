   157→   157→function computeBreach(
   158→   158→  elapsedHours: number,
   159→   159→  targetHours: number,
   160→   160→): "none" | "due_soon" | "breached" | "hard" {
   161→   161→  if (elapsedHours >= targetHours * 2) return "hard";
   162→   162→  if (elapsedHours >= targetHours) return "breached";
   163→   163→  if (targetHours > 0 && elapsedHours / targetHours >= 0.75) return "due_soon";
   164→   164→  return "none";
   165→   165→}

   197→   197→/**
   198→   198→ * Build a synthetic payload view from the schema + employee.
   199→   199→ * In production this would come from the request row itself; here we
   200→   200→ * synthesise a plausible payload using prefillFrom hints so the
   201→   201→ * service owner sees the same shape of data they would in production.
   202→   202→ */
   203→   203→function buildPayloadView(schema: IntakeSchema, employee: Employee): Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> {
   204→   204→  const out: Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> = {};
   205→   205→  for (const field of schema.fields) {
   206→   206→    let value = "";
   207→   207→    let prefilled = false;
   208→   208→    if (field.prefillFrom === "employee.location") {
   209→   209→      value = employee.location;
   210→   210→      prefilled = true;
   211→   211→    } else if (field.prefillFrom === "employee.stage") {
   212→   212→      value = employee.stage;
   213→   213→      prefilled = true;
   214→   214→    } else if (field.prefillFrom === "employee.startDate") {
   215→   215→      value = employee.startDate.slice(0, 10);
   216→   216→      prefilled = true;
   217→   217→    } else if (field.prefillFrom === "employee.role") {
   218→   218→      value = employee.role;
   219→   219→      prefilled = true;
   220→   220→    } else if (field.prefillFrom === "step.key") {
   221→   221→      value = "—";
   222→   222→      prefilled = true;
   223→   223→    } else if (field.prefillFrom === "step.dueAt") {
   224→   224→      value = "—";
   225→   225→      prefilled = true;
   226→   226→    } else {
   227→   227→      // Synthesise plausible free-text values
   228→   228→      if (field.type === "checkbox") {
   229→   229→        value = field.required ? "Yes" : "—";
   230→   230→      } else if (field.type === "select" && field.options && field.options.length > 0) {
   231→   231→        value = field.options[0].label;
   232→   232→      } else if (field.type === "number") {
   233→   233→        value = "1";
   234→   234→      } else if (field.type === "textarea") {
   235→   235→        value = "—";
   236→   236→      } else if (field.type === "date") {
   237→   237→        value = "2024-10-15";
   238→   238→      } else {
   239→   239→        value = "—";
   240→   240→      }
   241→   241→    }
   242→   242→    out[field.key] = { label: field.label, labelAr: field.labelAr, value, prefilled };
   243→   243→  }
   244→   244→  return out;
   245→   245→}

   247→   247→function fieldTypeDef(t: IntakeField["type"]): string {

   251→   251→    case "checkbox": return "checkbox";
   252→   252→    case "textarea": return "textarea";
   253→   253→    case "number": return "number";
   254→   254→    default: return "text";
   255→   255→  }
