   202→/* ---------- Helpers ---------- */
   203→
   204→function latestConsent(records: ConsentRecord[], purpose: Purpose): ConsentRecord | undefined {
   205→  return records
   206→    .filter((r) => r.purpose === purpose)
   207→    .sort((a, b) => b.grantedAt.localeCompare(a.grantedAt))[0];
   208→}
   209→
   210→function purposeGranted(
   211→  records: ConsentRecord[],
   212→  overrides: Record<string, boolean>,
   213→  meta: PurposeMeta
   214→): boolean {
   215→  if (typeof overrides[meta.key] === "boolean") return overrides[meta.key];
   216→  const latest = latestConsent(records, meta.key);
   217→  if (latest) return latest.granted;
   218→  return meta.defaultOn;
   219→}

   310→      const res = await fetch(`/api/v1/me/data?employeeId=${encodeURIComponent(employeeId)}`, { cache: "no-store" });
   311→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   312→      const json: ApiEnvelope<MeData> = await res.json();
   313→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   314→      setData(json.data);
   315→    } catch (e) {
   316→      setErr(e instanceof Error ? e.message : "Network error");
   317→    } finally {
   318→      setLoading(false);
   319→    }
   320→  }, [employeeId]);

   332→    async (meta: PurposeMeta, next: boolean) => {
   333→      if (meta.required) return;
   334→      setPendingPurpose(meta.key);
   335→      // Optimistic update via the store
   336→      setConsent(meta.key, next);
   337→      try {
   338→        const res = await fetch("/api/v1/consent", {
   339→          method: "POST",
   340→          headers: { "Content-Type": "application/json" },
   341→          body: JSON.stringify({ employeeId, purpose: meta.key, granted: next, version: "v1" }),
   342→        });
   343→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   344→        const json: ApiEnvelope<{ receiptId: string; grantedAt: string }> = await res.json();
   345→        if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   346→        const stamp = format(new Date(json.data.grantedAt), "HH:mm");
   347→        if (next) {
   348→          toast.success(`Consent recorded · ${json.data.receiptId} at ${stamp}`, {
   349→            description: meta.labelEn,
   350→          });
   351→        } else {
   352→          toast(`Revoked · ${json.data.receiptId}`, {
   353→            description: meta.labelEn,
   354→          });
   355→        }
   356→        // Refetch so the snapshot reflects the server
   357→        await fetchData();
   358→      } catch (e) {
   359→        // Roll back
   360→        setConsent(meta.key, !next);
   361→        toast.error("Could not update consent", {

   364→      } finally {

   380→            headers: { "Content-Type": "application/json" },
   381→            body: JSON.stringify({ employeeId, purpose: p.key, granted: false, version: "v1" }),

   383→        } catch {

   385→        }
