   127→/** Strip inline [^n] refs and footnote definitions; return clean markdown. */
   128→function stripCitations(md: string): string {
   129→  return md
   130→    // Strip footnote definition lines FIRST (before removing the [^n] markers they start with)
   131→    .replace(/^\[\^\d+\]:.*$/gm, "")
   132→    // Then strip inline [^n] references
   133→    .replace(/\[\^\d+\]/g, "")
   134→    .replace(/\n{3,}/g, "\n\n")
   135→    .trim();
   136→}
   137→
   138→/**
   139→ * Extract the first "decisive" sentence from a markdown answer.
   140→ * Used by the "Say it simply" toggle (PRD §01-016) so an employee can
   141→ * collapse a long answer to just the headline + citations.
   142→ *
   143→ * Strategy: walk the text looking for the first sentence-ending
   144→ * punctuation (`.`, `!`, `?`) that is followed by whitespace or end of
   145→ * string. If none found, return the whole text. Markdown emphasis
   146→ * markers (`*`, `_`, `**`) are preserved so ReactMarkdown still renders
   147→ * them — we only slice on punctuation.
   148→ */
   149→function firstSentence(md: string): string {
   150→  const text = md.trim();
   151→  if (!text) return "";
   152→  // Match up to and including the first . ! or ? that is followed by
   153→  // whitespace, end-of-string, or a newline.
   154→  const match = text.match(/^[^.!?\n]*[.!?](?=\s|$)/);
   155→  if (match) return match[0].trim();
   156→  // Fallback: first line, no terminal punctuation.
   157→  const firstLine = text.split(/\n/)[0];
   158→  return firstLine.trim();
   159→}
   160→
   161→function fmtMs(ms: number): string {
   162→  if (ms < 1000) return `${ms}ms`;
   163→  return `${(ms / 1000).toFixed(1)}s`;
   164→}

   333→              ? "أسألني أي شيء عن رحلتك أو سياساتنا. أُجيب من الوثائق الرسمية فقط — وإذا لم أكن متأكداً، أحيلك إلى الشخص المناسب."
   334→              : "Ask me anything about your journey or our policies. I answer from official documents only — and if I'm not sure, I hand you to the right person."}
   335→          </p>
   336→        </div>
   337→        {/* PRD §18-347 — "Search policies" trigger button. Opens the global
   338→            policy search modal. Sits next to the heading for parity with the

   403→      {/* Three-stage progress (only while loading) */}

   422→      </AnimatePresence>
   423→
   424→      {/* Answers / abstentions / errors */}

   427→          <StartersRow starters={starters} onPick={sendQuestion} locale={locale} />

   430→          {[...turns].reverse().map((turn) => (

   447→      {/* Sticky ask input area */}

   514→/* ============================================================
   515→   Three-stage progress — instrument style with pulse dots
   516→   connected by hairlines.
   517→   ============================================================ */
   518→
   519→function ThreeStageProgress({
   520→  activeStage,
   521→  stageElapsed,
   522→  locale,
   523→}: {
   524→  activeStage: number;
   525→  stageElapsed: number[];
   526→  locale: "en" | "ar";
   527→}) {
   528→  const labels =
   529→    locale === "ar"
   530→      ? ["يفحص رحلتك", "يقرأ السياسة", "يصيغ الإجابة"]
   531→      : ["checking your journey", "reading policy", "drafting"];
   532→  return (
   533→    <div>
   534→      <ol className="flex items-center gap-2">
   535→        {labels.map((label, i) => {
   536→          const isPast = i < activeStage;
   537→          const isActive = i === activeStage;
   538→          const isPending = i > activeStage;
   539→          return (
   540→            <li key={label} className="flex items-center gap-2 flex-1 min-w-0">
   541→              <span
   542→                className={cn(
   543→                  "inline-flex items-center justify-center w-3 h-3 rounded-full shrink-0",
   544→                  isActive && "ast-pulse-ring",
   545→                )}
   546→                style={{
   547→                  background: isPast
   548→                    ? "var(--brand-500)"
   549→                    : isActive
   550→                      ? "var(--brand-400)"
   551→                      : "var(--surface-3)",
   552→                  border: `1px solid ${
   553→                    isPast || isActive ? "var(--brand-500)" : "var(--border)"
   554→                  }`,
   555→                }}
   556→                aria-hidden="true"
   557→              />
   558→              <div className="flex-1 min-w-0">
   559→                <p
   560→                  className="text-[12px] font-medium leading-tight truncate"
   561→                  style={{
   562→                    color: isPast
   563→                      ? "var(--brand-400)"
   564→                      : isActive
   565→                        ? "var(--text-hi)"
   566→                        : "var(--text-lo)",
   567→                  }}
   568→                >
   569→                  {label}
   570→                </p>

   762→  const [copied, setCopied] = useState(false);
   763→  const [wrongReported, setWrongReported] = useState(false);
   764→  // "Say it simply" preference — per-question (local state, no persistence
   765→  // needed beyond the life of this card which is keyed by turn.id).
   766→  const [simple, setSimple] = useState(false);

   769→    () => (answer.answerMarkdown ? stripCitations(answer.answerMarkdown) : ""),
   770→    [answer.answerMarkdown],
   771→  );

   774→  const simpleMd = useMemo(() => firstSentence(cleanMd), [cleanMd]);
   775→

   839→    );

   841→  };
   842→
   843→  const handleWrong = () => {

   867→          }}

   872→            {locale === "ar"
   873→              ? "بحث السياسة أعاد وثيقة قديمة — يرجى التأكيد مع المالك."
   874→              : "Policy search returned a stale document — please confirm with the owner."}
   875→          </span>
   876→        </div>
   877→      )}
   878→
   879→      {/* "Say it simply" toggle (PRD §01-016) — pill button + Simple badge */}
