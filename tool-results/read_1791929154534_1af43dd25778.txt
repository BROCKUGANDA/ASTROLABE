   194→function computeBreach(
   195→  elapsedHours: number,
   196→  targetHours: number,
   197→): "none" | "due_soon" | "breached" | "hard" {
   198→  if (elapsedHours >= targetHours * 2) return "hard";
   199→  if (elapsedHours >= targetHours) return "breached";
   200→  if (targetHours > 0 && elapsedHours / targetHours >= 0.75) return "due_soon";
   201→  return "none";
   202→}

   234→/**
   235→ * Build a synthetic payload view from the schema + employee.
   236→ * In production this would come from the request row itself; here we
   237→ * synthesise a plausible payload using prefillFrom hints so the
   238→ * service owner sees the same shape of data they would in production.
   239→ */
   240→function buildPayloadView(schema: IntakeSchema, employee: Employee): Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> {
   241→  const out: Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> = {};
   242→  for (const field of schema.fields) {
   243→    let value = "";
   244→    let prefilled = false;
   245→    if (field.prefillFrom === "employee.location") {
   246→      value = employee.location;
   247→      prefilled = true;
   248→    } else if (field.prefillFrom === "employee.stage") {
   249→      value = employee.stage;
   250→      prefilled = true;
   251→    } else if (field.prefillFrom === "employee.startDate") {
   252→      value = employee.startDate.slice(0, 10);
   253→      prefilled = true;
   254→    } else if (field.prefillFrom === "employee.role") {
   255→      value = employee.role;
   256→      prefilled = true;
   257→    } else if (field.prefillFrom === "step.key") {
   258→      value = "—";
   259→      prefilled = true;
   260→    } else if (field.prefillFrom === "step.dueAt") {
   261→      value = "—";
   262→      prefilled = true;
   263→    } else {
   264→      // Synthesise plausible free-text values
   265→      if (field.type === "checkbox") {
   266→        value = field.required ? "Yes" : "—";
   267→      } else if (field.type === "select" && field.options && field.options.length > 0) {
   268→        value = field.options[0].label;
   269→      } else if (field.type === "number") {
   270→        value = "1";
   271→      } else if (field.type === "textarea") {
   272→        value = "—";
   273→      } else if (field.type === "date") {
   274→        value = "2024-10-15";
   275→      } else {
   276→        value = "—";
   277→      }
   278→    }
   279→    out[field.key] = { label: field.label, labelAr: field.labelAr, value, prefilled };
   280→  }
   281→  return out;
   282→}
   283→
   284→function fieldTypeDef(t: IntakeField["type"]): string {
   285→  switch (t) {
   286→    case "select": return "select";
   287→    case "date": return "date";
   288→    case "checkbox": return "checkbox";
   289→    case "textarea": return "textarea";
   290→    case "number": return "number";
   291→    default: return "text";
   292→  }
   293→}

   990→   status signals, then add a small baseline so the bars are always
   991→   visible. In production these counts would come from event log
   992→   analytics, not heuristics. */

   994→type BreachReason = {
   995→  key: string;
   996→  label: string;
   997→  labelAr: string;
   998→  count: number;
   999→};

  1001→function computeBreachReasons(items: ServiceRequest[]): BreachReason[] {
  1002→  let waitingEmployee = 0;
  1003→  let waitingDownstream = 0;
  1004→  let incompleteIntake = 0;
  1005→  let queueOverload = 0;
  1006→  let ambiguous = 0;
  1007→
  1008→  // Per-assignee open count → "queue overload" signal.
  1009→  const perAssignee: Record<string, number> = {};
  1010→  for (const r of items) {
  1011→    if (r.status === "resolved" || r.status === "closed" || r.status === "cancelled") continue;
  1012→    perAssignee[r.assignee] = (perAssignee[r.assignee] ?? 0) + 1;
  1013→  }
  1014→
  1015→  for (const r of items) {
  1016→    if (r.status === "resolved" || r.status === "closed" || r.status === "cancelled") continue;
  1017→    if (r.returnCount > 0) incompleteIntake += 1;
  1018→    else if (r.reassignCount > 0) waitingDownstream += 1;
  1019→    else if (r.status === "waiting_employee") waitingEmployee += 1;
  1020→    else if ((perAssignee[r.assignee] ?? 0) >= 3) queueOverload += 1;
  1021→    else ambiguous += 1;
  1022→  }
  1023→
  1024→  // Baseline — prototype padding so the chart reads even on a 4-row queue.
  1025→  const BASE = 2;
  1026→  return [
  1027→    { key: "waiting_employee", label: "Waiting on employee", labelAr: "بانتظار الموظف", count: waitingEmployee + BASE },
  1028→    { key: "waiting_downstream", label: "Waiting on downstream team", labelAr: "بانتظار فريق لاحق", count: waitingDownstream + BASE },
  1029→    { key: "incomplete_intake", label: "Incomplete intake", labelAr: "إدخال ناقص", count: incompleteIntake + BASE },
  1030→    { key: "queue_overload", label: "Queue overload", labelAr: "ازدحام القائمة", count: queueOverload + BASE },
  1031→    { key: "ambiguous_request", label: "Ambiguous request", labelAr: "طلب غامض", count: ambiguous + BASE },
  1032→  ];
  1033→}

  1035→/* ---------- Owner performance synthesis ----------
  1036→   Deterministic synthetic per-owner metrics seeded by the owner ID
  1037→   so they stay stable across re-renders but vary across owners. */
  1038→
  1039→type OwnerPerf = {
  1040→  id: string;
  1041→  name: string;
  1042→  nameAr: string;
  1043→  role: string;
  1044→  ticketsHandled: number;
  1045→  medianResolveHours: number;
  1046→  breachRate: number; // 0-100

  1050→function hashStr(s: string): number {
  1051→  let h = 0;
  1052→  for (let i = 0; i < s.length; i++) {
  1053→    h = (h * 31 + s.charCodeAt(i)) >>> 0;
  1054→  }
  1055→  return h;
  1056→}
  1057→
  1058→function generateOwnerPerf(group: OwnerGroup): OwnerPerf[] {
  1059→  const groupOwners = ALL_OWNERS.filter((o) => o.group === group);
  1060→  if (groupOwners.length === 0) return [];
  1061→  return groupOwners.map((o) => {
  1062→    const h = hashStr(o.id);
  1063→    return {
  1064→      id: o.id,
  1065→      name: o.name,
  1066→      nameAr: o.nameAr ?? o.name,
  1067→      role: o.role,
  1068→      ticketsHandled: 8 + (h % 14), // 8-21
  1069→      medianResolveHours: 12 + (h % 24), // 12-35h
  1070→      breachRate: 4 + (h % 22), // 4-25%
  1071→      zeroClarificationRate: 58 + (h % 38), // 58-95%
  1072→    };
  1073→  });
  1074→}
