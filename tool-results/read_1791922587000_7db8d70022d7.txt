   153→function computeBreach(
   154→  elapsedHours: number,
   155→  targetHours: number,
   156→): "none" | "due_soon" | "breached" | "hard" {
   157→  if (elapsedHours >= targetHours * 2) return "hard";
   158→  if (elapsedHours >= targetHours) return "breached";
   159→  if (targetHours > 0 && elapsedHours / targetHours >= 0.75) return "due_soon";
   160→  return "none";
   161→}
   162→
   163→function namedAssignee(assigneeId: string): { name: string; role: string; email: string; group: string } | null {
   164→  const o = getOwner(assigneeId);
   165→  if (!o) return null;
   166→  return { name: o.name, role: o.role, email: o.email, group: o.group };
   167→}
   168→
   169→function resolveOwnerForType(type: RequestType): { name: string; role: string; email: string; group: string } | null {
   170→  const schema = INTAKE_SCHEMAS[type];
   171→  const o = ALL_OWNERS.find((x) => x.group === schema.ownerGroup);
   172→  if (!o) return null;
   173→  return { name: o.name, role: o.role, email: o.email, group: o.group };
   174→}

   176→function parsePolicyRef(ref: string): { docSlug: string; title: string; section: string; effectiveDate: string } {
   177→  const [docSlug, section] = ref.split("#");
   178→  const titleMap: Record<string, string> = {
   179→    "relocation-policy": "Relocation Policy",
   180→    "immigration-policy": "Immigration Policy",
   181→    "it-policy": "IT Provisioning Policy",
   182→    "onboarding-policy": "Onboarding Policy",
   183→    "learning-policy": "Learning Policy",
   184→    "mobility-policy": "Internal Mobility Policy",
   185→    "payroll-policy": "Payroll Policy",
   186→    "facilities-policy": "Facilities Policy",
   187→  };
   188→  return {
   189→    docSlug,
   190→    title: titleMap[docSlug] ?? docSlug,
   191→    section: section ?? "—",
   192→    effectiveDate: "2024-01-01",
   193→  };
   194→}

   200→export function RequestsScreen() {
   201→  const locale = useAppStore((s) => s.locale);
   202→  const currentEmployeeId = useAppStore((s) => s.currentEmployeeId);
   203→  const activeRequestType = useAppStore((s) => s.activeRequestType);
   204→  const setActiveRequestType = useAppStore((s) => s.setActiveRequestType);
   205→
   206→  const [requests, setRequests] = useState<ServiceRequest[]>([]);
   207→  const [loading, setLoading] = useState(true);
   208→  const [error, setError] = useState<string | null>(null);
   209→  const [expandedId, setExpandedId] = useState<string | null>(null);
   210→
   211→  // Wizard state
   212→  const [wizardOpen, setWizardOpen] = useState(false);
   213→  const [wizardStep, setWizardStep] = useState<0 | 1 | 2>(0);
   214→  const [wizardType, setWizardType] = useState<RequestType | null>(null);
   215→  const [wizardPrefill, setWizardPrefill] = useState<Record<string, string>>({});
   216→  const [duplicateNotice, setDuplicateNotice] = useState<string | null>(null);
   217→
   218→  // Employee record for prefill
   219→  const [employee, setEmployee] = useState<Employee | null>(null);
   220→  const [journeyStage, setJourneyStage] = useState<string>("");
   221→
   222→  const fetchRequests = useCallback(async () => {
   223→    setLoading(true);
   224→    setError(null);
   225→    try {
   226→      const res = await fetch(
   227→        `/api/v1/requests?employeeId=${encodeURIComponent(currentEmployeeId)}`,
   228→      );
   229→      const json: ApiEnvelope<{ items: ServiceRequest[]; nextCursor: string | null }> =
   230→        await res.json();
   231→      if (!json.ok || !json.data) {
   232→        throw new Error(json.error?.message ?? "Failed to load requests");
   233→      }
   234→      setRequests(json.data.items);
   235→    } catch (err) {
   236→      setError(err instanceof Error ? err.message : "Unknown error");
   237→    } finally {
   238→      setLoading(false);
   239→    }
   240→  }, [currentEmployeeId]);
   241→
   242→  // Fetch employee record for prefill (from /api/v1/journey/me)
   243→  useEffect(() => {
   244→    let cancelled = false;
   245→    (async () => {
   246→      try {
   247→        const res = await fetch(
   248→          `/api/v1/journey/me?employeeId=${encodeURIComponent(currentEmployeeId)}&asOf=${encodeURIComponent(PACK_NOW.toISOString())}`,
   249→        );
   250→        const json: ApiEnvelope<JourneyMeData> = await res.json();
   251→        if (cancelled) return;
   252→        if (json.ok && json.data) {
   253→          setEmployee(json.data.employee);
   254→          setJourneyStage(json.data.stage);
   255→        }
   256→      } catch {
   257→        // Non-fatal — prefill just won't be available
   258→      }
   259→    })();
   260→    return () => {
   261→      cancelled = true;
   262→    };
   263→  }, [currentEmployeeId]);
   264→
   265→  useEffect(() => {
   266→    fetchRequests();
   267→  }, [fetchRequests]);
   268→
   269→  // Pick up activeRequestType from store (e.g. when user clicked
   270→  // "Start this request" from Ask or Today screen) → jump to step 2.
   271→  useEffect(() => {
   272→    if (activeRequestType && INTAKE_SCHEMAS[activeRequestType as RequestType]) {
   273→      setWizardType(activeRequestType as RequestType);
   274→      setWizardStep(1);
   275→      setWizardOpen(true);
   276→      // Pre-populate prefill map from employee record (only once per type)
   277→      if (employee) {
   278→        const pre: Record<string, string> = {};
   279→        const schema = INTAKE_SCHEMAS[activeRequestType as RequestType];
   280→        for (const field of schema.fields) {
   281→          if (!field.prefillFrom) continue;
   282→          if (field.prefillFrom === "employee.location") pre[field.key] = employee.location;
   283→          else if (field.prefillFrom === "employee.stage") pre[field.key] = journeyStage;
   284→          else if (field.prefillFrom === "employee.startDate") pre[field.key] = employee.startDate.slice(0, 10);
   285→          else if (field.prefillFrom === "employee.role") pre[field.key] = employee.role;
   286→        }
   287→        setWizardPrefill(pre);
   288→      }
   289→      // Clear the store flag so we don't re-open on next mount
   290→      setActiveRequestType(undefined);
   291→    }
   292→  }, [activeRequestType, employee, journeyStage, setActiveRequestType]);
   293→
   294→  const openWizard = () => {
   295→    setWizardType(null);
   296→    setWizardStep(0);
   297→    setWizardPrefill({});
   298→    setDuplicateNotice(null);
   299→    setWizardOpen(true);
   300→  };
   301→
   302→  const handlePickType = (type: RequestType) => {
   303→    setWizardType(type);
   304→    // Build prefill map fresh from employee record
   305→    const pre: Record<string, string> = {};
   306→    if (employee) {
   307→      const schema = INTAKE_SCHEMAS[type];
   308→      for (const field of schema.fields) {
   309→        if (!field.prefillFrom) continue;
   310→        if (field.prefillFrom === "employee.location") pre[field.key] = employee.location;
   311→        else if (field.prefillFrom === "employee.stage") pre[field.key] = journeyStage;
   312→        else if (field.prefillFrom === "employee.startDate") pre[field.key] = employee.startDate.slice(0, 10);
   313→        else if (field.prefillFrom === "employee.role") pre[field.key] = employee.role;
   314→      }
   315→    }
   316→    setWizardPrefill(pre);
   317→    setWizardStep(1);
   318→  };
   319→
   320→  type SubmitResult =
   321→    | { ok: true; id: string; duplicate: boolean }
   322→    | { ok: false; code: string; message: string };
   323→
   324→  const handleSubmit = async (
   325→    payload: Record<string, unknown>,
   326→  ): Promise<SubmitResult> => {
   327→    if (!wizardType) {
   328→      return { ok: false, code: "E_NO_TYPE", message: "No request type selected" };
   329→    }
   330→    const idempotencyKey = `idm-${currentEmployeeId}-${wizardType}-${Date.now().toString(36)}`;
   331→    try {
   332→      const res = await fetch("/api/v1/requests", {
   333→        method: "POST",
   334→        headers: { "Content-Type": "application/json" },
   335→        body: JSON.stringify({
   336→          employeeId: currentEmployeeId,
   337→          type: wizardType,
   338→          payload,
   339→          idempotencyKey,
   340→        }),
   341→      });
   342→      const json: ApiEnvelope<ServiceRequest & { servedFrom?: string }> = await res.json();
   343→      if (!json.ok || !json.data) {
   344→        const code = json.error?.code ?? "E_UNKNOWN";
   345→        const msg = json.error?.message ?? "Failed to create request";
   346→        if (json.error?.hint) {
   347→          toast.error(`${msg}`, { description: json.error.hint });
   348→        } else {
   349→          toast.error(msg);
   350→        }
   351→        return { ok: false, code, message: msg };
   352→      }
   353→
   354→      // Duplicate detection — idempotency replay returns servedFrom: "cache"
   355→      const isDuplicate = json.data.servedFrom === "cache";
   356→      await fetchRequests();
   357→      setWizardOpen(false);
   358→
   359→      if (isDuplicate) {
   360→        setDuplicateNotice(json.data.id);
   361→        setExpandedId(json.data.id);
   362→        toast.info(
   363→          locale === "ar"
   364→            ? "هذا الطلب موجود بالفعل — نعرض الأصلي."
   365→            : "This request already exists — showing the original.",
   366→        );
   367→      } else {
   368→        setExpandedId(json.data.id);
   369→        // Confetti-style toast
   370→        toast.success(
   371→          locale === "ar"
   372→            ? `تم إنشاء الطلب ${json.data.ref} ✨`
   373→            : `Request ${json.data.ref} created ✨`,
   374→          {
   375→            description:
   376→              locale === "ar"
   377→                ? "سيتواصل معك الفريق المالك قريباً."
   378→                : "The owning team will reach out shortly.",
   379→          },
   380→        );
   381→      }
   382→      return { ok: true, id: json.data.id, duplicate: isDuplicate };
   383→    } catch (err) {
   384→      const msg = err instanceof Error ? err.message : "Network error";
   385→      toast.error(msg);
   386→      return { ok: false, code: "E_NETWORK", message: msg };
   387→    }
   388→  };
   389→
   390→  return (
   391→    <div className="max-w-[1180px] mx-auto px-4 sm:px-6 lg:px-8 py-6">
   392→      {/* Header */}
   393→      <div className="mb-5 flex items-center justify-between gap-3">
   394→        <div className="flex items-center gap-3 min-w-0">
   395→          <div
   396→            className="flex items-center justify-center w-10 h-10 rounded-full shrink-0"
   397→            style={{ background: "var(--g-veil)", color: "var(--brand-400)" }}
   398→            aria-hidden="true"
   399→          >
   400→            <Inbox className="w-5 h-5" strokeWidth={1.75} />
   401→          </div>
   402→          <div className="min-w-0">
   403→            <h1
   404→              className="text-[20px] sm:text-[22px] font-semibold leading-tight"
   405→              style={{ color: "var(--text-hi)" }}
   406→            >
   407→              {locale === "ar" ? "طلباتي" : "My requests"}
   408→            </h1>
   409→            <p className="text-[13px] mt-0.5" style={{ color: "var(--text-mid)" }}>
   410→              {locale === "ar"
   411→                ? "تتبّع طلباتك المفتوحة وابدأ طلباً جديداً."
   412→                : "Track your open requests and start a new one."}
   413→            </p>
   414→          </div>
   415→        </div>
   416→        <button
   417→          onClick={openWizard}
   418→          className="ast-focus inline-flex items-center justify-center gap-2 h-11 px-4 sm:px-5 rounded-[10px] text-[13px] sm:text-[14px] font-semibold shrink-0 transition-transform hover:-translate-y-0.5"
   419→          style={{
   420→            background: "var(--brand-500)",
   421→            color: "var(--surface-0)",
   422→            boxShadow: "0 4px 14px rgba(22,154,165,0.28)",
   423→          }}
   424→        >
   425→          <Plus className="w-4 h-4" strokeWidth={2} />
   426→          <span className="hidden sm:inline">
   427→            {locale === "ar" ? "طلب جديد" : "New request"}

   430→        </button>
   431→      </div>
   432→
   433→      {/* Duplicate notice */}
   434→      {duplicateNotice && (
   435→        <div
   436→          className="ast-card p-3 mb-4 flex items-center gap-2"
   437→          style={{
   438→            borderColor: "rgba(22,154,165,0.30)",
   439→            background: "rgba(22,154,165,0.06)",
   440→          }}
   441→        >
   442→          <CheckCircle2
   443→            className="w-4 h-4 shrink-0"
   444→            strokeWidth={1.75}
   445→            style={{ color: "var(--brand-400)" }}
   446→          />
   447→          <p className="text-[13px]" style={{ color: "var(--text-hi)" }}>
   448→            {locale === "ar"

   457→            <X className="w-4 h-4" strokeWidth={1.75} style={{ color: "var(--text-lo)" }} />

   461→
   462→      {/* Layout: list (left col-7) + wizard (right col-5) */}
   463→      <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
   464→        {/* List */}
   465→        <section
   466→          className="lg:col-span-7 order-2 lg:order-1"

   468→        >
   469→          {loading ? (

   471→          ) : error ? (

   502→        {/* Wizard — desktop right column; mobile modal */}

   908→/* ============================================================
   909→   Wizard (3 steps)
   910→   ============================================================ */
   911→
   912→function Wizard({
   913→  step,
   914→  type,
   915→  prefill,
   916→  onStepChange,
   917→  onPickType,
   918→  onSubmit,
   919→  onReset,
   920→  locale,
   921→  employeeId,
   922→}: {
   923→  step: 0 | 1 | 2;
   924→  type: RequestType | null;
   925→  prefill: Record<string, string>;
   926→  onStepChange: (s: 0 | 1 | 2) => void;
   927→  onPickType: (t: RequestType) => void;
   928→  onSubmit: (payload: Record<string, unknown>) => Promise<{ ok: boolean; code?: string; message?: string; id?: string; duplicate?: boolean }>;
   929→  onReset: () => void;
   930→  locale: "en" | "ar";
   931→  employeeId: string;
   932→}) {
   933→  const schema: IntakeSchema | null = type ? INTAKE_SCHEMAS[type] : null;
   934→  // The Wizard is keyed by `type` in the parent so it remounts
   935→  // whenever the type changes, ensuring this initializer re-runs.
   936→  const [payload, setPayload] = useState<Record<string, unknown>>(() => {
   937→    if (!type) return {};
   938→    const s = INTAKE_SCHEMAS[type];
   939→    const initial: Record<string, unknown> = {};
   940→    for (const field of s.fields) {
   941→      if (prefill[field.key] !== undefined) {
   942→        initial[field.key] = prefill[field.key];
   943→      } else if (field.type === "checkbox") {
   944→        initial[field.key] = false;
   945→      } else {
   946→        initial[field.key] = "";
   947→      }
   948→    }
   949→    return initial;
   950→  });
   951→  const [errors, setErrors] = useState<Record<string, string>>({});
   952→  const [submitError, setSubmitError] = useState<string | null>(null);
   953→  const [submitting, setSubmitting] = useState(false);
   954→
   955→  const validateField = (field: IntakeField, value: unknown): string => {
   956→    if (field.required) {
   957→      if (value === undefined || value === null || value === "" || value === false) {
   958→        return locale === "ar"
   959→          ? `${locale === "ar" ? field.labelAr : field.label} مطلوب`
   960→          : `${field.label} is required`;
   961→      }
   962→    }
   963→    if (field.type === "number" && value !== "" && value !== undefined) {
   964→      const n = Number(value);
   965→      if (Number.isNaN(n)) {
   966→        return locale === "ar" ? "يجب أن يكون رقماً" : "Must be a number";
   967→      }
   968→      if (n < 0) return locale === "ar" ? "لا يمكن أن يكون سالباً" : "Cannot be negative";
   969→    }
   970→    if (field.type === "date" && value && field.key === "effective_date") {
   971→      const d = new Date(value as string);
   972→      if (Number.isNaN(d.getTime())) {
   973→        return locale === "ar" ? "تاريخ غير صالح" : "Invalid date";
   974→      }
   975→    }
   976→    return "";
   977→  };
   978→
   979→  const validateAll = (): boolean => {
   980→    if (!schema) return false;
   981→    const next: Record<string, string> = {};
   982→    for (const field of schema.fields) {
   983→      const err = validateField(field, payload[field.key]);
   984→      if (err) next[field.key] = err;
   985→    }
   986→    setErrors(next);
   987→    return Object.keys(next).length === 0;
   988→  };
   989→
   990→  const handleFieldChange = (key: string, value: unknown) => {
   991→    setPayload((prev) => ({ ...prev, [key]: value }));
   992→    // Inline validation as you type — clear error on change, re-validate if it had one
   993→    if (errors[key]) {
   994→      const field = schema?.fields.find((f) => f.key === key);
   995→      if (field) {
   996→        const err = validateField(field, value);
   997→        setErrors((prev) => {
   998→          const next = { ...prev };
   999→          if (err) next[key] = err;
  1000→          else delete next[key];
  1001→          return next;
  1002→        });
  1003→      }
  1004→    }
  1005→  };
  1006→
  1007→  const handleSubmit = async () => {
  1008→    if (!schema) return;
  1009→    if (!validateAll()) {
  1010→      toast.error(
  1011→        locale === "ar"
  1012→          ? "يرجى تصحيح الحقول المطلوبة"
  1013→          : "Please fix the required fields",
  1014→      );
  1015→      return;
  1016→    }
  1017→    setSubmitting(true);
  1018→    setSubmitError(null);
  1019→    const result = await onSubmit(payload);
  1020→    setSubmitting(false);
  1021→    if (!result.ok) {
  1022→      setSubmitError(result.message ?? "Failed to submit");
  1023→    }
  1024→  };
  1025→
  1026→  return (
  1027→    <div>
  1028→      {/* Step indicator */}
  1029→      <div className="flex items-center gap-1.5 mb-4">
  1030→        {[0, 1, 2].map((i) => (
  1031→          <div
  1032→            key={i}
  1033→            className="flex-1 h-1 rounded-full"
  1034→            style={{
  1035→              background:
  1036→                i <= step ? "var(--brand-500)" : "var(--surface-3)",
  1037→            }}
  1038→          />
  1039→        ))}
  1040→      </div>
  1041→
  1042→      <AnimatePresence mode="wait">
  1043→        {step === 0 && (
  1044→          <motion.div
  1045→            key="step0"
  1046→            initial={{ opacity: 0, x: 8 }}
  1047→            animate={{ opacity: 1, x: 0 }}
  1048→            exit={{ opacity: 0, x: -8 }}
  1049→            transition={{ duration: 0.18 }}
  1050→          >
  1051→            <Step1TypeGrid onPick={onPickType} locale={locale} />
  1052→          </motion.div>
  1053→        )}
  1054→        {step === 1 && schema && (
  1055→          <motion.div
  1056→            key="step1"
  1057→            initial={{ opacity: 0, x: 8 }}
  1058→            animate={{ opacity: 1, x: 0 }}
  1059→            exit={{ opacity: 0, x: -8 }}
  1060→            transition={{ duration: 0.18 }}
  1061→          >
  1062→            <Step2Form
  1063→              schema={schema}
  1064→              payload={payload}
  1065→              errors={errors}
  1066→              prefill={prefill}
  1067→              onChange={handleFieldChange}
  1068→              onBack={() => {
  1069→                onReset();
  1070→                onStepChange(0);
  1071→              }}
  1072→              onNext={() => {
  1073→                if (validateAll()) onStepChange(2);
  1074→                else
  1075→                  toast.error(
  1076→                    locale === "ar"
  1077→                      ? "يرجى تصحيح الحقول المطلوبة"
  1078→                      : "Please fix the required fields",
  1079→                  );
  1080→              }}
  1081→              locale={locale}
  1082→            />
  1083→          </motion.div>
  1084→        )}
  1085→        {step === 2 && schema && (
  1086→          <motion.div
  1087→            key="step2"
  1088→            initial={{ opacity: 0, x: 8 }}
  1089→            animate={{ opacity: 1, x: 0 }}
  1090→            exit={{ opacity: 0, x: -8 }}
  1091→            transition={{ duration: 0.18 }}
  1092→          >
  1093→            <Step3Preview
  1094→              schema={schema}
  1095→              payload={payload}
  1096→              onBack={() => onStepChange(1)}
  1097→              onSubmit={handleSubmit}
  1098→              submitting={submitting}
  1099→              submitError={submitError}
  1100→              locale={locale}
  1101→              employeeId={employeeId}
  1102→            />
  1103→          </motion.div>
  1104→        )}
  1105→      </AnimatePresence>
  1106→    </div>
  1107→  );
  1108→}
