   164→function truncateHash(h: string): string {
   165→  // "0x" + first 8 chars total + "…" + last 4 chars.
   166→  // 0xabc4129f7e8d2c1b → 0xabc412…2c1b
   167→  if (h.length <= 14) return h;
   168→  return `${h.slice(0, 8)}…${h.slice(-4)}`;
   169→}

   171→function shortDate(iso: string, locale: "en" | "ar"): string {
   172→  try {
   173→    const d = new Date(iso);
   174→    return format(d, "d MMM yyyy · HH:mm") + (locale === "ar" ? " ت" : " UTC");
   175→  } catch {
   176→    return iso;
   177→  }
   178→}

   190→interface VerifyResult {
   191→  ok: boolean;
   192→  verifiedCount: number;
   193→  totalCount: number;
   194→  tamperedEventId?: string;
   195→  expectedPrev?: string;
   196→  foundPrev?: string;
   197→  lastVerifiedId?: string;
   198→}
   199→
   200→/** Walk the chain and verify every prev-hash links to the prior event's hash. */
   201→function verifyChain(chain: AuditEventRow[]): VerifyResult {
   202→  if (chain.length === 0) {
   203→    return { ok: true, verifiedCount: 0, totalCount: 0 };
   204→  }
   205→  // Genesis event: its prevHash must be the zero marker "0x0000000000000000".
   206→  const ZERO = "0x0000000000000000";
   207→  if (chain[0].prevHash !== ZERO) {
   208→    return {
   209→      ok: false,
   210→      verifiedCount: 0,
   211→      totalCount: chain.length,
   212→      tamperedEventId: chain[0].id,
   213→      expectedPrev: ZERO,
   214→      foundPrev: chain[0].prevHash,
   215→      lastVerifiedId: undefined,
   216→    };
   217→  }
   218→  let lastVerifiedId: string | undefined = chain[0].id;
   219→  for (let i = 1; i < chain.length; i += 1) {
   220→    const prev = chain[i - 1];
   221→    const cur = chain[i];
   222→    if (cur.prevHash !== prev.hash) {
   223→      return {
   224→        ok: false,
   225→        verifiedCount: i,
   226→        totalCount: chain.length,
   227→        tamperedEventId: cur.id,
   228→        expectedPrev: prev.hash,
   229→        foundPrev: cur.prevHash,
   230→        lastVerifiedId,
   231→      };
   232→    }
   233→    lastVerifiedId = cur.id;
   234→  }
   235→  return {
   236→    ok: true,
   237→    verifiedCount: chain.length,
   238→    totalCount: chain.length,
   239→  };
   240→}

   242→function cloneChain(src: AuditEventRow[]): AuditEventRow[] {
   243→  return src.map((e) => ({ ...e }));
   244→}

   246→/** Apply a tamper to a fresh copy of the chain. Returns the tampered copy and the set of affected ids. */
   247→function applyTamper(type: TamperType): { chain: AuditEventRow[]; tamperedIds: string[] } {
   248→  const chain = cloneChain(AUDIT_EVENTS);
   249→  const tamperedIds: string[] = [];
   250→  if (type === "alter_hash") {
   251→    // ae-003's prevHash is rewritten to a fake value, breaking the link
   252→    // at ae-003. (Matches the spec's example verification message.)
   253→    const target = chain.find((e) => e.id === "ae-003");
   254→    if (target) {
   255→      target.prevHash = "0xFAKEFAKEFAKEFAKE";
   256→      tamperedIds.push("ae-003");
   257→    }
   258→  } else if (type === "delete_event") {
   259→    // Remove ae-002 → ae-003's prevHash no longer links to the prior event (ae-001).
   260→    const idx = chain.findIndex((e) => e.id === "ae-002");
   261→    if (idx >= 0) {
   262→      chain.splice(idx, 1);
   263→      const orphan = chain.find((e) => e.id === "ae-003");
   264→      if (orphan) tamperedIds.push(orphan.id);
   265→    }
   266→  } else if (type === "reorder_events") {
   267→    // Swap ae-002 and ae-003 → both become orphans of the new ordering.
   268→    const a = chain.findIndex((e) => e.id === "ae-002");
   269→    const b = chain.findIndex((e) => e.id === "ae-003");
   270→    if (a >= 0 && b >= 0) {
   271→      const tmp = chain[a];
   272→      chain[a] = chain[b];
   273→      chain[b] = tmp;
   274→      tamperedIds.push("ae-002", "ae-003");
   275→    }
   276→  } else if (type === "inject_fake") {
   277→    // Insert a synthetic event between ae-002 and ae-003 with a wrong prev hash.
   278→    const idx = chain.findIndex((e) => e.id === "ae-003");
   279→    if (idx >= 0) {
   280→      const fake: AuditEventRow = {
   281→        id: "ae-fake",
   282→        actor: "system",
   283→        action: "ingestion",
   284→        subject: "pack",
   285→        subjectId: "synthetic-fake-v1",
   286→        hash: "0xDEADBEEFDEADBEEF",
   287→        prevHash: "0xWRONGPREV000000",
   288→        at: "2024-10-19T11:00:00Z",
   289→      };
   290→      chain.splice(idx, 0, fake);
   291→      tamperedIds.push("ae-fake");
   292→    }
   293→  }
   294→  return { chain, tamperedIds };
   295→}
