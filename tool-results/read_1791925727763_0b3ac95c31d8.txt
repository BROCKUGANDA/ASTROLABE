   253→   253→/* Walk the blockedBy chain to build a phrase in words. */
   254→   254→function blockedByChainWords(step: Step, allSteps: Step[], locale: "en" | "ar"): string {
   255→   255→  if (step.blockedBy.length === 0) return "";
   256→   256→  const chain: string[] = [];
   257→   257→  const visited = new Set<string>();
   258→   258→  let current: Step | undefined = step;
   259→   259→  while (current && current.blockedBy.length > 0) {
   260→   260→    const firstDepKey = current.blockedBy[0];
   261→   261→    if (visited.has(firstDepKey)) break;
   262→   262→    visited.add(firstDepKey);
   263→   263→    const dep = allSteps.find((s) => s.stepKey === firstDepKey);
   264→   264→    if (!dep) break;
   265→   265→    chain.push(locale === "ar" ? dep.labelAr : dep.label);
   266→   266→    current = dep;
   267→   267→  }
   268→   268→  if (chain.length === 0) return "";
   269→   269→  if (locale === "ar") {
   270→   270→    return `${chain.join(" ← ")} — يجب إكمال هذه الخطوات أولاً`;
   271→   271→  }
   272→   272→  if (chain.length === 1) return `${chain[0]} needs to finish first`;
   273→   273→  if (chain.length === 2) return `${chain[0]} waits on ${chain[1]}`;
   274→   274→  // chain.length >= 3 — join with "which waits on"
   275→   275→  const head = chain[0];
   276→   276→  const tail = chain.slice(1);
   277→   277→  return `${head} waits on ${tail.join(", which waits on ")}`;
   278→   278→}

   482→   482→    [journeyType]

   489→   489→  const [h, setH] = useState(64);
   490→   490→  useEffect(() => {
   491→   491→    const unsub = heightSpring.on("change", (v) => setH(v));

   519→   519→          {journeyStages.map((st, idx) => {

   523→   523→            return (

   525→   525→                {idx > 0 && (
   526→   526→                  <ChevronRight
   527→   527→                    className="w-3 h-3 shrink-0"
   528→   528→                    strokeWidth={1.75}
   529→   529→                    style={{ color: "var(--text-lo)" }}
   530→   530→                    aria-hidden="true"

   570→   570→function BlockedByGraph({ step, allSteps, locale }: { step: Step; allSteps: Step[]; locale: "en" | "ar" }) {
   571→   571→  const chain: Step[] = [];
   572→   572→  const visited = new Set<string>();
   573→   573→  let cur: Step | undefined = step;
   574→   574→  while (cur && cur.blockedBy.length > 0) {
   575→   575→    const firstKey = cur.blockedBy[0];
   576→   576→    if (visited.has(firstKey)) break;
   577→   577→    visited.add(firstKey);
   578→   578→    const dep = allSteps.find((s) => s.stepKey === firstKey);
   579→   579→    if (!dep) break;
   580→   580→    chain.unshift(dep);
   581→   581→    cur = dep;
   582→   582→  }
   583→   583→  if (chain.length === 0) return null;
   584→   584→
   585→   585→  return (
   586→   586→    <div className="ast-card-2 p-3 overflow-x-auto">
   587→   587→      <div className="flex items-center gap-2 min-w-max">

   599→   599→                {locale === "ar" ? s.labelAr : s.label}

   955→   955→/* ---------- Timeline (events with sticky week headers) ---------- */
   956→   956→
   957→   957→interface WeekGroup {
   958→   958→  weekKey: string;
   959→   959→  weekLabel: string;
   960→   960→  events: JourneyEvent[];
   961→   961→}
   962→   962→
   963→   963→function groupByWeek(events: JourneyEvent[], locale: "en" | "ar"): WeekGroup[] {
   964→   964→  const groups: WeekGroup[] = [];
   965→   965→  const seenWeeks = new Set<string>();
   966→   966→  // Sort newest first
   967→   967→  const sorted = [...events].sort((a, b) => b.occurredAt.localeCompare(a.occurredAt));
   968→   968→  for (const ev of sorted) {
   969→   969→    const d = new Date(ev.occurredAt);
   970→   970→    const weekStart = startOfWeek(d, { weekStartsOn: 1 });
   971→   971→    const weekKey = format(weekStart, "yyyy-MM-dd");
   972→   972→    let g = groups.find((x) => x.weekKey === weekKey);
   973→   973→    if (!g) {
   974→   974→      g = {
   975→   975→        weekKey,
   976→   976→        weekLabel: locale === "ar"
   977→   977→          ? `أسبوع ${format(weekStart, "d MMM")}`
   978→   978→          : `Week of ${format(weekStart, "d MMM")}`,
   979→   979→        events: [],
   980→   980→      };
   981→   981→      groups.push(g);
   982→   982→      seenWeeks.add(weekKey);
   983→   983→    }
   984→   984→    g.events.push(ev);
   985→   985→  }
   986→   986→  return groups;
   987→   987→}
