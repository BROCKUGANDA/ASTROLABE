   502→   - Astrolabe canvas shifts max 8px opposite the cursor
   503→   - Hero text shifts max 4px with the cursor
   504→   - 60fps via framer-motion useMotionValue + useTransform
   505→   - Disabled on touch devices and when prefers-reduced-motion

   507→
   508→function usePrefersReducedMotion(): boolean {
   509→  const [reduce, setReduce] = useState(false);
   510→  useEffect(() => {
   511→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   512→    const update = () => setReduce(mq.matches);
   513→    update();
   514→    mq.addEventListener("change", update);
   515→    return () => mq.removeEventListener("change", update);
   516→  }, []);
   517→  return reduce;
   518→}
   519→
   520→function useFinePointer(): boolean {
   521→  const [fine, setFine] = useState(false);
   522→  useEffect(() => {
   523→    const mq = window.matchMedia("(pointer: fine)");
   524→    const update = () => setFine(mq.matches);
   525→    update();
   526→    mq.addEventListener("change", update);
   527→    return () => mq.removeEventListener("change", update);
   528→  }, []);
   529→  return fine;
   530→}

   532→function HeroParallax() {
