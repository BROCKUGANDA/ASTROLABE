   245→function AsOfClock() {
   246→  const locale = useAppStore((s) => s.locale);
   247→  const offsetHours = useAppStore((s) => s.asOfOffsetHours);
   248→  const setOffset = useAppStore((s) => s.setAsOfOffset);
   249→
   250→  const asOf = useMemo(() => {
   251→    return new Date(PACK_NOW.getTime() + offsetHours * 3600 * 1000);
   252→  }, [offsetHours]);
   253→
   254→  const offsetDays = offsetHours / 24;
   255→
   256→  return (
   257→    <div className="ast-card p-5 sm:p-6">
   258→      <div className="flex items-center justify-between mb-4 flex-wrap gap-3">
   259→        <div>
   260→          <p className="text-[11px] uppercase tracking-[0.14em] font-medium" style={{ color: "var(--text-lo)" }}>

   262→          </p>
   263→          <p className="text-[28px] font-semibold tabular mt-1" style={{ color: "var(--text-hi)" }} dir="ltr">
   264→            {format(asOf, "EEE d MMM yyyy · HH:mm")}
   265→          </p>
   266→          <p className="text-[12px] mt-1 tabular" style={{ color: "var(--text-mid)" }}>
   267→            {offsetHours === 0
   268→              ? (locale === "ar" ? "الوقت المرجعي للنموذج" : "Pack reference time (now)")

   271→              : (locale === "ar" ? `قبل ${Math.abs(offsetDays).toFixed(1)} يوم` : `${Math.abs(offsetDays).toFixed(1)} days ago`)}

   273→        </div>
   274→        <div className="text-right">
   275→          <p className="text-[11px] uppercase tracking-[0.14em]" style={{ color: "var(--text-lo)" }}>

   292→      />

   294→        <span>{locale === "ar" ? "−10 أيام" : "−10 days"}</span>

   326→function ScenariosView({ onLoad }: { onLoad: (employeeId: string) => void }) {
   327→  const locale = useAppStore((s) => s.locale);
   328→  const setEmployee = useAppStore((s) => s.setEmployee);
   329→  const setOffset = useAppStore((s) => s.setAsOfOffset);
   330→  const [scenarios, setScenarios] = useState<Scenario[] | null>(null);
   331→  const [employees, setEmployees] = useState<Record<string, EmployeeLite> | null>(null);
   332→  const [previewAssess, setPreviewAssess] = useState<Record<string, Assessment | null>>({});
   333→  const [selectedId, setSelectedId] = useState<string | null>(null);
   334→  const [err, setErr] = useState<string | null>(null);
   335→  const [loading, setLoading] = useState(true);
   336→
   337→  const fetchAll = useCallback(async () => {
   338→    setLoading(true);
   339→    setErr(null);
   340→    try {
   341→      const [scRes, empRes] = await Promise.all([
   342→        fetch("/api/v1/scenarios", { cache: "no-store" }),
   343→        fetch("/api/v1/employees", { cache: "no-store" }),
   344→      ]);
   345→      if (!scRes.ok || !empRes.ok) throw new Error("HTTP error");
   346→      const scJson: ApiEnvelope<{ scenarios: Scenario[] }> = await scRes.json();
   347→      const empJson: ApiEnvelope<{ items: EmployeeLite[] }> = await empRes.json();
   348→      if (!scJson.ok || !scJson.data || !empJson.ok || !empJson.data) {
   349→        throw new Error(scJson.error?.message ?? empJson.error?.message ?? "Unknown error");
   350→      }
   351→      setScenarios(scJson.data.scenarios);
   352→      const map: Record<string, EmployeeLite> = {};
   353→      for (const e of empJson.data.items) map[e.id] = e;
   354→      setEmployees(map);
   355→    } catch (e) {
   356→      setErr(e instanceof Error ? e.message : "Network error");
   357→    } finally {
   358→      setLoading(false);
   359→    }
   360→  }, []);
   361→
   362→  useEffect(() => { fetchAll(); }, [fetchAll]);
   363→
   364→  // Live preview assessment for selected scenario (recomputes with asOf offset)
   365→  const offsetHours = useAppStore((s) => s.asOfOffsetHours);
   366→  useEffect(() => {
   367→    if (!selectedId || !scenarios) return;
   368→    const sc = scenarios.find((s) => s.id === selectedId);
   369→    if (!sc) return;
   370→    const asOf = new Date(PACK_NOW.getTime() + offsetHours * 3600 * 1000).toISOString();
   371→    let cancelled = false;
   372→    fetch(`/api/v1/journey/${encodeURIComponent(sc.employeeId)}/assessment?asOf=${encodeURIComponent(asOf)}`, { cache: "no-store" })
   373→      .then((r) => r.json())
   374→      .then((json: ApiEnvelope<{ assessment: Assessment; trace: EngineTrace }>) => {
   375→        if (cancelled) return;
   376→        if (json.ok && json.data) {
   377→          setPreviewAssess((prev) => ({ ...prev, [sc.id]: json.data!.assessment }));
   378→        }
   379→      })
   380→      .catch(() => {});
   381→    return () => { cancelled = true; };
   382→  }, [selectedId, scenarios, offsetHours]);
   383→
   384→  const handleLoad = useCallback((sc: Scenario) => {
   385→    setEmployee(sc.employeeId);
   386→    onLoad(sc.employeeId);
   387→  }, [onLoad, setEmployee]);
   388→
   389→  if (loading) {
   390→    return (
   391→      <div className="space-y-3" aria-hidden="true">
   392→        <div className="ast-card h-24 ast-shimmer rounded-[14px]" />
   393→        {Array.from({ length: 5 }).map((_, i) => (
   394→          <div key={i} className="ast-card p-4 h-24 ast-shimmer rounded-[12px]" />
   395→        ))}
   396→      </div>
   397→    );
   398→  }
   399→  if (err) return <ErrorBanner message={err} onRetry={fetchAll} />;
   400→
   401→  return (
   402→    <div className="space-y-5">
   403→      <div>
   404→        <h2 className="text-[17px] font-semibold" style={{ color: "var(--text-hi)" }}>
   405→          {locale === "ar" ? "السيناريوهات الخمسة" : "Five scenarios"}
   406→        </h2>
   407→        <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   408→          {locale === "ar"
   409→            ? "كل بطاقة تُحمّل موظفاً في الأداة وتعرض التصنيف المتوقع. حرّك الساعة لرؤية المحرك يعيد التقييم."
   410→            : "Each card loads an employee into the instrument and previews the expected band. Drag the clock to watch the engine re-assess."}
   411→        </p>
   412→      </div>
   413→
   414→      <AsOfClock />
   415→
   416→      <ul className="grid grid-cols-1 lg:grid-cols-2 gap-3">
   417→        {scenarios?.map((sc) => {
   418→          const emp = employees?.[sc.employeeId];
   419→          const a = previewAssess[sc.id];
   420→          const isSelected = selectedId === sc.id;
   421→          return (
   422→            <motion.li
   423→              key={sc.id}
   424→              initial={false}
   425→              whileHover={{ y: -2 }}
   426→              onClick={() => { setSelectedId(sc.id); setEmployee(sc.employeeId); }}
   427→              className={cn("ast-card p-4 sm:p-5 cursor-pointer ast-focus transition-colors")}
   428→              style={{
   429→                borderColor: isSelected ? "rgba(22,154,165,0.40)" : "var(--border)",
   430→                background: isSelected ? "rgba(22,154,165,0.04)" : "var(--surface-1)",
   431→              }}
   432→            >
   433→              <div className="flex items-baseline justify-between gap-3 mb-2">
   434→                <div className="flex items-center gap-2">
   435→                  <span
   436→                    className="font-mono text-[12px] tabular px-2 py-0.5 rounded"
   437→                    style={{ background: "var(--surface-3)", color: "var(--brand-400)" }}
   438→                  >
   439→                    {sc.id}
   440→                  </span>
   441→                  <h3 className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   442→                    {sc.title}
   443→                  </h3>
   444→                </div>
   445→                {a && <BandChip band={a.band} score={a.score} locale={locale} size="sm" />}
   446→              </div>
   447→              <p className="text-[12px] leading-relaxed" style={{ color: "var(--text-mid)" }}>
   448→                {sc.description}
   449→              </p>
   450→              <div className="flex items-center justify-between mt-3 flex-wrap gap-2">
   451→                <p className="text-[11px] font-mono tabular" style={{ color: "var(--text-lo)" }} dir="ltr">
   452→                  {sc.employeeId}{emp ? ` · ${emp.displayName}` : ""}
   453→                </p>
   454→                <button
   455→                  onClick={(e) => { e.stopPropagation(); handleLoad(sc); }}
   456→                  className="ast-focus inline-flex items-center gap-1.5 h-9 px-3 rounded-full text-[12px] font-medium"
   457→                  style={{
   458→                    background: "var(--brand-500)",
   459→                    color: "var(--surface-0)",
   460→                    border: "1px solid var(--brand-500)",
   461→                  }}
   462→                >
   463→                  <PlayCircle className="w-3.5 h-3.5" strokeWidth={1.75} />
   464→                  {locale === "ar" ? "حمّل السيناريو" : "Load scenario"}
   465→                </button>
   466→              </div>
   467→            </motion.li>
   468→          );
   469→        })}
   470→      </ul>
   471→    </div>
   472→  );
   473→}
