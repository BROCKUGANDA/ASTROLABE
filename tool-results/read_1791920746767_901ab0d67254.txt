   152→function truncateHash(h: string): string {
   153→  // Keep "0x" + first 8 hex chars
   154→  if (h.length <= 10) return h;
   155→  return `${h.slice(0, 10)}`;
   156→}

   158→function shortDate(iso: string, locale: "en" | "ar"): string {
   159→  try {
   160→    const d = new Date(iso);
   161→    return format(d, "d MMM yyyy · HH:mm") + (locale === "ar" ? " ت" : " UTC");
   162→  } catch {
   163→    return iso;
   164→  }
   165→}
