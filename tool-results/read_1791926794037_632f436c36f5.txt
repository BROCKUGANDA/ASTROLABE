   155→function truncateHash(h: string): string {
   156→  // Keep "0x" + first 8 hex chars
   157→  if (h.length <= 10) return h;
   158→  return `${h.slice(0, 10)}`;
   159→}

   161→function shortDate(iso: string, locale: "en" | "ar"): string {
   162→  try {
   163→    const d = new Date(iso);
   164→    return format(d, "d MMM yyyy · HH:mm") + (locale === "ar" ? " ت" : " UTC");
   165→  } catch {
   166→    return iso;
   167→  }
   168→}
