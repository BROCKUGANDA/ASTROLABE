   149→function computeBreach(
   150→  elapsedHours: number,
   151→  targetHours: number,
   152→): "none" | "due_soon" | "breached" | "hard" {
   153→  if (elapsedHours >= targetHours * 2) return "hard";
   154→  if (elapsedHours >= targetHours) return "breached";
   155→  if (targetHours > 0 && elapsedHours / targetHours >= 0.75) return "due_soon";
   156→  return "none";
   157→}

   189→/**
   190→ * Build a synthetic payload view from the schema + employee.
   191→ * In production this would come from the request row itself; here we
   192→ * synthesise a plausible payload using prefillFrom hints so the
   193→ * service owner sees the same shape of data they would in production.
   194→ */
   195→function buildPayloadView(schema: IntakeSchema, employee: Employee): Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> {
   196→  const out: Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> = {};
   197→  for (const field of schema.fields) {
   198→    let value = "";
   199→    let prefilled = false;
   200→    if (field.prefillFrom === "employee.location") {
   201→      value = employee.location;
   202→      prefilled = true;
   203→    } else if (field.prefillFrom === "employee.stage") {
   204→      value = employee.stage;
   205→      prefilled = true;
   206→    } else if (field.prefillFrom === "employee.startDate") {
   207→      value = employee.startDate.slice(0, 10);
   208→      prefilled = true;
   209→    } else if (field.prefillFrom === "employee.role") {
   210→      value = employee.role;
   211→      prefilled = true;
   212→    } else if (field.prefillFrom === "step.key") {
   213→      value = "—";
   214→      prefilled = true;
   215→    } else if (field.prefillFrom === "step.dueAt") {
   216→      value = "—";
   217→      prefilled = true;
   218→    } else {
   219→      // Synthesise plausible free-text values
   220→      if (field.type === "checkbox") {
   221→        value = field.required ? "Yes" : "—";
   222→      } else if (field.type === "select" && field.options && field.options.length > 0) {
   223→        value = field.options[0].label;
   224→      } else if (field.type === "number") {
   225→        value = "1";
   226→      } else if (field.type === "textarea") {
   227→        value = "—";
   228→      } else if (field.type === "date") {
   229→        value = "2024-10-15";
   230→      } else {
   231→        value = "—";
   232→      }
   233→    }
   234→    out[field.key] = { label: field.label, labelAr: field.labelAr, value, prefilled };
   235→  }
   236→  return out;
   237→}
   238→
   239→function fieldTypeDef(t: IntakeField["type"]): string {
   240→  switch (t) {
   241→    case "select": return "select";
   242→    case "date": return "date";
   243→    case "checkbox": return "checkbox";
   244→    case "textarea": return "textarea";
   245→    case "number": return "number";
   246→    default: return "text";
   247→  }
   248→}

  1031→export function ServiceOwnerScreen() {
  1032→  const locale = useAppStore((s) => s.locale);
  1033→  const dir = locale === "ar" ? "rtl" : "ltr";
  1034→
  1035→  const [group, setGroup] = useState<OwnerGroup>("IT");
  1036→  const [items, setItems] = useState<ServiceRequest[]>([]);
  1037→  const [loading, setLoading] = useState(true);
  1038→  const [err, setErr] = useState<string | null>(null);
  1039→
  1040→  const fetchQueue = useCallback(async () => {
  1041→    setLoading(true);
  1042→    setErr(null);
  1043→    try {
  1044→      const res = await fetch(`/api/v1/requests`, { cache: "no-store" });
  1045→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
  1046→      const json: ApiEnvelope<{ items: ServiceRequest[]; nextCursor: string | null }> = await res.json();
  1047→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
  1048→      // Merge with any locally-mutated seeds (so transitions persist in-session)
  1049→      // by combining server items with SEED_REQUESTS that may have been mutated.
  1050→      // The server returns the in-memory array which IS mutated; so we just use it.
  1051→      setItems(json.data.items);
  1052→    } catch (e) {
  1053→      setErr(e instanceof Error ? e.message : "Network error");
  1054→    } finally {
  1055→      setLoading(false);
  1056→    }
  1057→  }, []);
  1058→
  1059→  useEffect(() => {
  1060→    fetchQueue();
  1061→  }, [fetchQueue]);
  1062→
  1063→  const filtered = useMemo(() => {
  1064→    return items.filter((r) => r.ownerGroup === group);
  1065→  }, [items, group]);
  1066→
  1067→  // Zero-clarification rate = items with returnCount === 0 / total items
  1068→  const zeroClarificationCount = filtered.filter((r) => r.returnCount === 0).length;
  1069→  const totalFiltered = filtered.length;
  1070→  const zeroClarificationRate = totalFiltered > 0 ? (zeroClarificationCount / totalFiltered) * 100 : 100;
  1071→

  1073→    async (id: string, to: string, opts?: { note?: string; question?: string; detail?: string }) => {

  1076→          method: "POST",
  1077→          headers: { "Content-Type": "application/json" },
  1078→          body: JSON.stringify({ to, idempotencyKey: `${id}-${to}-${Date.now()}` }),
  1079→        });
  1080→        if (!res.ok) {
  1081→          const json = await res.json().catch(() => null);
  1082→          throw new Error(json?.error?.message ?? `HTTP ${res.status}`);
  1083→        }
  1084→        const json: ApiEnvelope<ServiceRequest> = await res.json();
  1085→        if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
  1086→
  1087→        // Optimistically update local state
  1088→        setItems((prev) => prev.map((r) => (r.id === id ? { ...r, status: to } : r)));
  1089→
  1090→        if (to === "returned") {
  1091→          // Also bump the return count locally
  1092→          setItems((prev) => prev.map((r) => (r.id === id ? { ...r, returnCount: r.returnCount + 1 } : r)));
  1093→          toast.success(
  1094→            locale === "ar"

  1097→            { description: opts?.question }
  1098→          );
  1099→        } else if (to === "resolved") {

  1101→            locale === "ar"
  1102→              ? "تم إشعار الموظف: تم حل طلبك — إليك ما تغيّر"
  1103→              : "Employee notified: Your request is resolved — here's what changed",
  1104→            { description: opts?.note }
  1105→          );
  1106→        } else if (to === "acknowledged") {
  1107→          toast.success(locale === "ar" ? "تم الاستلام" : "Acknowledged", { description: id });
  1108→        } else if (to === "in_progress") {
  1109→          toast.success(locale === "ar" ? "بدأ العمل" : "Started work", { description: id });

  1113→          description: e instanceof Error ? e.message : "Network error",

  1125→          <Inbox className="w-3.5 h-3.5" strokeWidth={1.75} />
  1126→          <span>{locale === "ar" ? "وحدة مالك الخدمة" : "Service owner console"}</span>
  1127→        </div>
  1128→        <h1 className="mt-2 text-[26px] sm:text-[30px] font-semibold tracking-tight" style={{ color: "var(--text-hi)" }}>
  1129→          {locale === "ar" ? "قائمة الانتظار" : "Your queue"}
  1130→        </h1>
