   186→function computeBreach(
   187→  elapsedHours: number,
   188→  targetHours: number,
   189→): "none" | "due_soon" | "breached" | "hard" {
   190→  if (elapsedHours >= targetHours * 2) return "hard";
   191→  if (elapsedHours >= targetHours) return "breached";
   192→  if (targetHours > 0 && elapsedHours / targetHours >= 0.75) return "due_soon";
   193→  return "none";
   194→}

   226→/**
   227→ * Build a synthetic payload view from the schema + employee.
   228→ * In production this would come from the request row itself; here we
   229→ * synthesise a plausible payload using prefillFrom hints so the
   230→ * service owner sees the same shape of data they would in production.
   231→ */
   232→function buildPayloadView(schema: IntakeSchema, employee: Employee): Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> {
   233→  const out: Record<string, { label: string; labelAr: string; value: string; prefilled: boolean }> = {};
   234→  for (const field of schema.fields) {
   235→    let value = "";
   236→    let prefilled = false;
   237→    if (field.prefillFrom === "employee.location") {
   238→      value = employee.location;
   239→      prefilled = true;
   240→    } else if (field.prefillFrom === "employee.stage") {
   241→      value = employee.stage;
   242→      prefilled = true;
   243→    } else if (field.prefillFrom === "employee.startDate") {
   244→      value = employee.startDate.slice(0, 10);
   245→      prefilled = true;
   246→    } else if (field.prefillFrom === "employee.role") {
   247→      value = employee.role;
   248→      prefilled = true;
   249→    } else if (field.prefillFrom === "step.key") {
   250→      value = "—";
   251→      prefilled = true;
   252→    } else if (field.prefillFrom === "step.dueAt") {
   253→      value = "—";
   254→      prefilled = true;
   255→    } else {
   256→      // Synthesise plausible free-text values
   257→      if (field.type === "checkbox") {
   258→        value = field.required ? "Yes" : "—";
   259→      } else if (field.type === "select" && field.options && field.options.length > 0) {
   260→        value = field.options[0].label;
   261→      } else if (field.type === "number") {
   262→        value = "1";
   263→      } else if (field.type === "textarea") {
   264→        value = "—";
   265→      } else if (field.type === "date") {
   266→        value = "2024-10-15";
   267→      } else {
   268→        value = "—";
   269→      }
   270→    }
   271→    out[field.key] = { label: field.label, labelAr: field.labelAr, value, prefilled };
   272→  }
   273→  return out;
   274→}
   275→
   276→function fieldTypeDef(t: IntakeField["type"]): string {
   277→  switch (t) {
   278→    case "select": return "select";
   279→    case "date": return "date";
   280→    case "checkbox": return "checkbox";
   281→    case "textarea": return "textarea";
   282→    case "number": return "number";
   283→    default: return "text";
   284→  }
   285→}

   979→/* ---------- Breach-reason synthesis ----------
   980→   For the prototype we classify each open request into one of five
   981→   reason buckets using its actual returnCount / reassignCount /

   986→type BreachReason = {
   987→  key: string;
   988→  label: string;
   989→  labelAr: string;
   990→  count: number;

   993→function computeBreachReasons(items: ServiceRequest[]): BreachReason[] {
   994→  let waitingEmployee = 0;
   995→  let waitingDownstream = 0;
   996→  let incompleteIntake = 0;
   997→  let queueOverload = 0;
   998→  let ambiguous = 0;
   999→
  1000→  // Per-assignee open count → "queue overload" signal.
  1001→  const perAssignee: Record<string, number> = {};
  1002→  for (const r of items) {
  1003→    if (r.status === "resolved" || r.status === "closed" || r.status === "cancelled") continue;
  1004→    perAssignee[r.assignee] = (perAssignee[r.assignee] ?? 0) + 1;
  1005→  }
  1006→
  1007→  for (const r of items) {
  1008→    if (r.status === "resolved" || r.status === "closed" || r.status === "cancelled") continue;
  1009→    if (r.returnCount > 0) incompleteIntake += 1;
  1010→    else if (r.reassignCount > 0) waitingDownstream += 1;
  1011→    else if (r.status === "waiting_employee") waitingEmployee += 1;
  1012→    else if ((perAssignee[r.assignee] ?? 0) >= 3) queueOverload += 1;
  1013→    else ambiguous += 1;
  1014→  }
  1015→
  1016→  // Baseline — prototype padding so the chart reads even on a 4-row queue.
  1017→  const BASE = 2;
  1018→  return [
  1019→    { key: "waiting_employee", label: "Waiting on employee", labelAr: "بانتظار الموظف", count: waitingEmployee + BASE },
  1020→    { key: "waiting_downstream", label: "Waiting on downstream team", labelAr: "بانتظار فريق لاحق", count: waitingDownstream + BASE },
  1021→    { key: "incomplete_intake", label: "Incomplete intake", labelAr: "إدخال ناقص", count: incompleteIntake + BASE },
  1022→    { key: "queue_overload", label: "Queue overload", labelAr: "ازدحام القائمة", count: queueOverload + BASE },
  1023→    { key: "ambiguous_request", label: "Ambiguous request", labelAr: "طلب غامض", count: ambiguous + BASE },
  1024→  ];
  1025→}

  1042→function hashStr(s: string): number {
  1043→  let h = 0;
  1044→  for (let i = 0; i < s.length; i++) {
  1045→    h = (h * 31 + s.charCodeAt(i)) >>> 0;
  1046→  }
  1047→  return h;
  1048→}

  1050→function generateOwnerPerf(group: OwnerGroup): OwnerPerf[] {

  1059→      role: o.role,
  1060→      ticketsHandled: 8 + (h % 14), // 8-21
  1061→      medianResolveHours: 12 + (h % 24), // 12-35h
