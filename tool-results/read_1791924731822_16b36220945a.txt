   266→function QueueView({ onOpenPacket }: { onOpenPacket: (employeeId: string) => void }) {
   267→  const locale = useAppStore((s) => s.locale);
   268→  const [items, setItems] = useState<QueueItem[] | null>(null);
   269→  const [err, setErr] = useState<string | null>(null);
   270→  const [loading, setLoading] = useState(true);
   271→  const [claiming, setClaiming] = useState<string | null>(null);
   272→
   273→  const fetchQ = useCallback(async () => {
   274→    setLoading(true);
   275→    setErr(null);
   276→    try {
   277→      const res = await fetch("/api/v1/hrbp/queue", { cache: "no-store" });
   278→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   279→      const json: ApiEnvelope<{ items: QueueItem[] }> = await res.json();
   280→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   281→      setItems(json.data.items);
   282→    } catch (e) {
   283→      setErr(e instanceof Error ? e.message : "Network error");
   284→    } finally {
   285→      setLoading(false);
   286→    }
   287→  }, []);
   288→
   289→  useEffect(() => { fetchQ(); }, [fetchQ]);
   290→
   291→  const handleClaim = useCallback(async (item: QueueItem) => {
   292→    setClaiming(item.employeeId);
   293→    // Fake audit-event write — there's no API yet, so we just simulate the receipt.
   294→    await new Promise((r) => setTimeout(r, 320));
   295→    toast.success("Case claimed by Fatima — employee notified", {
   296→      description: `${item.employeeName} · ${item.employeeId}`,
   297→    });
   298→    setClaiming(null);
   299→  }, []);
   300→
   301→  if (loading) return <QueueSkeleton />;
   302→  if (err) return <ErrorBanner message={err} onRetry={fetchQ} />;
   303→  if (!items) return null;
   304→
   305→  const active = items.filter((i) => bandSortIndex(i.band) >= 2); // friction + blocked
   306→  const rest = items.filter((i) => bandSortIndex(i.band) < 2);
   307→
   308→  return (
   309→    <div>
   310→      <div className="flex items-center justify-between mb-4 flex-wrap gap-3">
   311→        <div>
   312→          <h2 className="text-[17px] font-semibold" style={{ color: "var(--text-hi)" }}>
   313→            {locale === "ar" ? "قائمة أول من يحتاج مساعدة" : "Who needs help first"}
   314→          </h2>
   315→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   316→            {locale === "ar"
   317→              ? `${active.length} موظف في احتكاك أو أعلى اليوم.`
   318→              : `${active.length} employee${active.length === 1 ? "" : "s"} in Friction or above today.`}
   319→          </p>
   320→        </div>
   321→        <button
   322→          onClick={fetchQ}
   323→          className="ast-focus inline-flex items-center gap-2 h-9 px-3 rounded-full text-[12px] font-medium"
   324→          style={{ background: "var(--surface-2)", border: "1px solid var(--border)", color: "var(--text-mid)" }}
   325→        >
   326→          <RefreshCw className="w-3.5 h-3.5" strokeWidth={1.75} />
   327→          {locale === "ar" ? "تحديث" : "Refresh"}
   328→        </button>
   329→      </div>
   330→
   331→      {active.length === 0 ? (
   332→        <div className="ast-card p-10 flex flex-col items-center text-center">
   333→          <div
   334→            className="flex items-center justify-center w-14 h-14 rounded-full mb-4"
   335→            style={{ background: "rgba(14,159,110,0.10)", color: "var(--status-steady)" }}
   336→          >
   337→            <CheckCircle2 className="w-6 h-6" strokeWidth={1.75} />
   338→          </div>
   339→          <p className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   340→            {locale === "ar" ? "لا أحد في احتكاك أو أعلى اليوم." : "No one in Friction or above today."}
   341→          </p>
   342→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   343→            {locale === "ar" ? "تحقق لاحقاً، أو تصفّح بقية القائمة أدناه." : "Check back later, or browse the watch list below."}
   344→          </p>
   345→        </div>
   346→      ) : (
   347→        <ul className="space-y-3 mb-8">
   348→          {active.map((item, i) => (
   349→            <QueueRow
   350→              key={item.employeeId}
   351→              item={item}
   352→              rank={i + 1}
   353→              index={i}
   354→              claiming={claiming === item.employeeId}
   355→              onClaim={() => handleClaim(item)}
   356→              onOpen={() => onOpenPacket(item.employeeId)}
   357→            />
   358→          ))}
   359→        </ul>
   360→      )}
   361→
   362→      {/* Watch + steady (collapsed by default-ish) */}
   363→      {rest.length > 0 && (
   364→        <details className="ast-card p-4" style={{ borderColor: "var(--border-soft)" }}>
   365→          <summary className="cursor-pointer text-[13px] font-medium flex items-center gap-2" style={{ color: "var(--text-mid)" }}>
   366→            <ChevronRight className="w-3.5 h-3.5" strokeWidth={1.75} />
   367→            {locale === "ar"
   368→              ? `${rest.length} موظف تحت المراقبة أو مستقر`
   369→              : `${rest.length} on watch or steady`}
   370→          </summary>
   371→          <ul className="mt-3 space-y-2">
   372→            {rest.map((item, i) => (
   373→              <QueueRow
   374→                key={item.employeeId}
   375→                item={item}
   376→                rank={active.length + i + 1}
   377→                index={i}
   378→                claiming={claiming === item.employeeId}
   379→                onClaim={() => handleClaim(item)}
   380→                onOpen={() => onOpenPacket(item.employeeId)}
   381→                compact
   382→              />
   383→            ))}
   384→          </ul>
   385→        </details>
   386→      )}
   387→    </div>
   388→  );
   389→}

   391→function QueueRow({
   392→  item,
   393→  rank,
   394→  index,
   395→  claiming,
   396→  onClaim,
   397→  onOpen,
   398→  compact = false,
   399→}: {
   400→  item: QueueItem;
   401→  rank: number;
   402→  index: number;
   403→  claiming: boolean;
   404→  onClaim: () => void;
   405→  onOpen: () => void;
   406→  compact?: boolean;
   407→}) {
   408→  const locale = useAppStore((s) => s.locale);
   409→  const reducedMotion = useAppStore((s) => s.reducedMotion);
   410→  const leadDays = Math.max(0, Math.round(item.leadTimeHours / 24));
   411→
   412→  // Hover-expand state.
   413→  // Desktop (pointer: fine) → hover reveals preview.
   414→  // Mobile  (pointer: coarse) → tap chevron toggles preview.
   415→  const [isHover, setIsHover] = useState(false);
   416→  const [isTapped, setIsTapped] = useState(false);
   417→
   418→  // Band → accent colour (left bar + glow).
   419→  const bandAccent: Record<BandT, string> = {
   420→    steady: "var(--status-steady)",
   421→    watch: "var(--status-watch)",
   422→    friction: "var(--status-friction)",
   423→    blocked: "var(--status-blocked)",
   424→  };
   425→  const accent = bandAccent[item.band];
   426→
   427→  // Spring drop-in stagger (40ms per item, capped at 0.3s = ~8 items).
   428→  const springDelay = reducedMotion ? 0 : Math.min(index * 0.04, 0.3);
   429→
   430→  const isPreviewOpen = isHover || isTapped;
   431→
   432→  return (
   433→    <motion.li
   434→      initial={reducedMotion ? false : { opacity: 0, y: 12 }}
   435→      animate={{ opacity: 1, y: 0 }}
   436→      transition={
   437→        reducedMotion
   438→          ? { duration: 0 }
   439→          : { type: "spring", stiffness: 220, damping: 26, delay: springDelay }
   440→      }
   441→    >
   442→      <Tilt3D maxDeg={2}>
   443→        <div
   444→          className={cn(
   445→            "relative ast-card",
   446→            compact ? "p-3" : "p-4 sm:p-5",
   447→          )}
   448→          style={{
   449→            // Depth shadow on hover / preview-open (180ms).
   450→            boxShadow: isPreviewOpen ? "var(--e3)" : "var(--e2)",
   451→            borderColor: isPreviewOpen ? "rgba(22,154,165,0.35)" : undefined,
   452→            transition:
   453→              "box-shadow 180ms var(--ease-standard), border-color 180ms var(--ease-standard)",
   454→          }}
   455→          onMouseEnter={() => setIsHover(true)}
   456→          onMouseLeave={() => {
   457→            setIsHover(false);
   458→          }}
   459→        >
   460→          {/* Band-coloured left accent bar — at-a-glance scanning */}
   461→          <div
   462→            aria-hidden="true"
   463→            className="absolute left-0 top-0 bottom-0 rounded-l-[14px] pointer-events-none"
   464→            style={{
   465→              width: 3,
   466→              background: accent,
   467→              boxShadow: `0 0 10px ${accent}55`,
   468→            }}
   469→          />
   470→
   471→          <div className="flex items-start gap-3 sm:gap-4 flex-wrap pl-1">
   472→            {/* Rank + band + score */}
   473→            <div className="flex items-center gap-3 shrink-0">
   474→              <span
   475→                className="text-[12px] tabular font-mono"
   476→                style={{ color: "var(--text-lo)" }}
   477→              >
   478→                #{rank}
   479→              </span>
   480→              <BandChip
   481→                band={item.band}
   482→                score={item.score}
   483→                locale={locale}
   484→                size="lg"
   485→              />
   486→            </div>
   487→
   488→            {/* Name + stage */}
   489→            <div className="flex-1 min-w-0">
   490→              <p
   491→                className="text-[15px] font-medium truncate"
   492→                style={{ color: "var(--text-hi)" }}
   493→              >
   494→                {item.employeeName}
   495→              </p>
   496→              <p
   497→                className="text-[12px] mt-0.5"
   498→                style={{ color: "var(--text-mid)" }}
   499→              >
   500→                {stageLabel(item.stage, locale)}
   501→                <span className="mx-1.5">·</span>
   502→                <span className="tabular">
   503→                  {locale === "ar"
   504→                    ? `أول إشارة قبل ${leadDays} يوم`
   505→                    : `first signal ${leadDays} day${leadDays === 1 ? "" : "s"} ago`}
   506→                </span>
   507→                <span className="mx-1.5">·</span>
   508→                <span className="tabular" style={{ color: "var(--text-lo)" }}>
   509→                  {locale === "ar" ? "النتيجة " : "score "}
   510→                  <NumberTicker value={item.score} />
   511→                </span>
   512→              </p>
   513→
   514→              {/* Top signals */}
   515→              {item.topSignals.length > 0 && (
   516→                <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   517→                  {item.topSignals.map((s, i) => (
   518→                    <SignalBadge
   519→                      key={`${s.code}-${i}`}
   520→                      code={s.code}
   521→                      weight={s.weight}
   522→                      hardTrip={s.code.startsWith("DP") || s.code.startsWith("MB")}
   523→                    />
   524→                  ))}
   525→                </div>
   526→              )}
   527→            </div>
   528→
   529→            {/* Suggested owner + actions */}
   530→            <div className="flex items-center gap-2 shrink-0 ml-auto">
   531→              {item.suggestedOwner && (
   532→                <div className="hidden md:flex flex-col items-end text-right">
   533→                  <span
   534→                    className="text-[11px] uppercase tracking-[0.1em]"
   535→                    style={{ color: "var(--text-lo)" }}
