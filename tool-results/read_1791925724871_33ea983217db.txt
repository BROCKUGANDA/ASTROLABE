   253→/* Walk the blockedBy chain to build a phrase in words. */
   254→function blockedByChainWords(step: Step, allSteps: Step[], locale: "en" | "ar"): string {
   255→  if (step.blockedBy.length === 0) return "";
   256→  const chain: string[] = [];
   257→  const visited = new Set<string>();
   258→  let current: Step | undefined = step;
   259→  while (current && current.blockedBy.length > 0) {
   260→    const firstDepKey = current.blockedBy[0];
   261→    if (visited.has(firstDepKey)) break;
   262→    visited.add(firstDepKey);
   263→    const dep = allSteps.find((s) => s.stepKey === firstDepKey);
   264→    if (!dep) break;
   265→    chain.push(locale === "ar" ? dep.labelAr : dep.label);
   266→    current = dep;
   267→  }
   268→  if (chain.length === 0) return "";
   269→  if (locale === "ar") {
   270→    return `${chain.join(" ← ")} — يجب إكمال هذه الخطوات أولاً`;
   271→  }
   272→  if (chain.length === 1) return `${chain[0]} needs to finish first`;
   273→  if (chain.length === 2) return `${chain[0]} waits on ${chain[1]}`;
   274→  // chain.length >= 3 — join with "which waits on"
   275→  const head = chain[0];
   276→  const tail = chain.slice(1);
   277→  return `${head} waits on ${tail.join(", which waits on ")}`;
   278→}

   476→/* ---------- Sticky collapsing Stage Rail ---------- */
   477→
   478→function StickyStageRail({ data, locale }: { data: JourneyMeData; locale: "en" | "ar" }) {
   479→  const journeyType = data.employee.journeyType;
   480→  const journeyStages = useMemo(
   481→    () => ALL_STAGES.filter((s) => s.journeyType === journeyType).sort((a, b) => a.order - b.order),
   482→    [journeyType]
   483→  );
   484→
   485→  const { scrollY } = useScroll();
   486→  // Map scroll 0..120px → height 64..48px (height values apply per segment padding wrapper)
   487→  const raw = useTransform(scrollY, [0, 120], [64, 48]);
   488→  const heightSpring = useSpring(raw, { stiffness: 220, damping: 28 });
   489→  const [h, setH] = useState(64);
   490→  useEffect(() => {
   491→    const unsub = heightSpring.on("change", (v) => setH(v));
   492→    return () => unsub();
   493→  }, [heightSpring]);
   494→
   495→  const segState = (stageKey: string): "done" | "active" | "blocked" | "pending" => {
   496→    const stepsInStage = data.steps.filter((s) => s.stage === stageKey);
   497→    if (stepsInStage.length === 0) return "pending";
   498→    const allDone = stepsInStage.every((s) => s.status === "done" || s.status === "skipped");
   499→    if (allDone) return "done";
   500→    const anyBlocked = stepsInStage.some((s) => s.status === "blocked" || s.breachLevel === "hard");
   501→    if (anyBlocked) return "blocked";
   502→    if (stageKey === data.stage) return "active";

   570→function BlockedByGraph({ step, allSteps, locale }: { step: Step; allSteps: Step[]; locale: "en" | "ar" }) {
   571→  const chain: Step[] = [];
   572→  const visited = new Set<string>();
   573→  let cur: Step | undefined = step;
   574→  while (cur && cur.blockedBy.length > 0) {
   575→    const firstKey = cur.blockedBy[0];
   576→    if (visited.has(firstKey)) break;
   577→    visited.add(firstKey);
   578→    const dep = allSteps.find((s) => s.stepKey === firstKey);
   579→    if (!dep) break;
   580→    chain.unshift(dep);
   581→    cur = dep;
   582→  }
   583→  if (chain.length === 0) return null;
   584→
   585→  return (
   586→    <div className="ast-card-2 p-3 overflow-x-auto">

   599→                {locale === "ar" ? s.labelAr : s.label}

   963→function groupByWeek(events: JourneyEvent[], locale: "en" | "ar"): WeekGroup[] {
   964→  const groups: WeekGroup[] = [];
   965→  const seenWeeks = new Set<string>();
   966→  // Sort newest first
   967→  const sorted = [...events].sort((a, b) => b.occurredAt.localeCompare(a.occurredAt));
   968→  for (const ev of sorted) {
   969→    const d = new Date(ev.occurredAt);
   970→    const weekStart = startOfWeek(d, { weekStartsOn: 1 });
   971→    const weekKey = format(weekStart, "yyyy-MM-dd");
   972→    let g = groups.find((x) => x.weekKey === weekKey);
   973→    if (!g) {
   974→      g = {
   975→        weekKey,
   976→        weekLabel: locale === "ar"
   977→          ? `أسبوع ${format(weekStart, "d MMM")}`
   978→          : `Week of ${format(weekStart, "d MMM")}`,
   979→        events: [],
   980→      };
   981→      groups.push(g);
   982→      seenWeeks.add(weekKey);
   983→    }
   984→    g.events.push(ev);
   985→  }
   986→  return groups;
   987→}

  1077→export function JourneyScreen() {
  1078→  const employeeId = useAppStore((s) => s.currentEmployeeId);
  1079→  const locale = useAppStore((s) => s.locale);
  1080→
  1081→  const [data, setData] = useState<JourneyMeData | null>(null);
  1082→  const [events, setEvents] = useState<JourneyEvent[]>([]);
  1083→  const [loading, setLoading] = useState(true);
  1084→  const [error, setError] = useState<{ message: string; requestId?: string } | null>(null);
  1085→
  1086→  const load = useCallback(async () => {
  1087→    setLoading(true);
  1088→    setError(null);
  1089→    try {
  1090→      // Use the pack's deterministic NOW (2024-10-19) so elapsed-hour and
  1091→      // band reads match the seeded events.
  1092→      const asOf = PACK_NOW.toISOString();
  1093→      const [journeyRes, meRes] = await Promise.all([
  1094→        fetch(`/api/v1/journey/me?employeeId=${encodeURIComponent(employeeId)}&asOf=${encodeURIComponent(asOf)}`, {
  1095→          headers: { Accept: "application/json" },
  1096→        }),
  1097→        fetch(`/api/v1/me/data?employeeId=${encodeURIComponent(employeeId)}`, {
  1098→          headers: { Accept: "application/json" },
  1099→        }),
  1100→      ]);
  1101→      const journeyJson: ApiEnvelope<JourneyMeData> = await journeyRes.json();
  1102→      if (!journeyJson.ok || !journeyJson.data) {
  1103→        throw new Error(journeyJson.error?.message ?? "Failed to load journey");
  1104→      }
  1105→      setData(journeyJson.data);
  1106→
  1107→      try {
  1108→        const meJson: ApiEnvelope<MeData> = await meRes.json();
  1109→        if (meJson.ok && meJson.data) {
  1110→          setEvents(meJson.data.journeyEvents ?? []);
  1111→        }
  1112→      } catch {
  1113→        setEvents([]);
  1114→      }
  1115→    } catch (e) {
  1116→      const msg = e instanceof Error ? e.message : "Unknown error";
  1117→      setError({ message: msg });
  1118→    } finally {
  1119→      setLoading(false);
  1120→    }
  1121→  }, [employeeId]);
  1122→
  1123→  useEffect(() => {
  1124→    void load();
  1125→  }, [load]);
  1126→
  1127→  const dir = locale === "ar" ? "rtl" : "ltr";
  1128→
  1129→  const hasEvents = events.length > 0 || (data?.steps.some((s) => s.startedAt) ?? false);
  1130→
  1131→  return (
  1132→    <div
  1133→      dir={dir}
  1134→      className="min-h-screen flex flex-col"
  1135→      style={{ background: "var(--surface-0)" }}
  1136→    >
  1137→      <main className="flex-1 w-full max-w-[1180px] mx-auto px-4 sm:px-6 lg:px-8 py-6 sm:py-8 space-y-6">
  1138→        {loading ? (
  1139→          <JourneySkeleton />
  1140→        ) : error ? (
  1141→          <ErrorBanner
  1142→            message={error.message}
  1143→            requestId={error.requestId}
  1144→            onRetry={() => void load()}
  1145→            locale={locale}
  1146→          />
  1147→        ) : data ? (
  1148→          !hasEvents ? (
  1149→            <EmptyState employee={data.employee} locale={locale} onRetry={() => void load()} />
  1150→          ) : (
  1151→            <>
  1152→              <OrbitHeader data={data} locale={locale} />
  1153→              <StickyStageRail data={data} locale={locale} />
  1154→              <StepList data={data} events={events} locale={locale} />
  1155→              <Timeline events={events} locale={locale} />
  1156→
  1157→              <p
  1158→                className="text-[11px] font-mono tabular text-center pt-2 pb-4"
  1159→                style={{ color: "var(--text-lo)" }}
  1160→              >
  1161→                {data.engineVersion} · {data.mapVersion} · {data.corpusVersion}
  1162→              </p>
  1163→            </>
  1164→          )
  1165→        ) : null}
  1166→      </main>
  1167→    </div>
  1168→  );
  1169→}
