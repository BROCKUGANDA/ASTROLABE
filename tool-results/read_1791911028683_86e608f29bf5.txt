   137→/* ============================================================
   138→   SLA CLOCK — working calendar aware

   140→export interface WorkingCalendar {
   141→  weekendDays: number[]; // 0 = Sunday
   142→  holidays: string[]; // ISO dates
   143→}
   144→
   145→export const CALENDARS: Record<string, WorkingCalendar> = {
   146→  DXB: { weekendDays: [6, 5], holidays: ["2024-12-02", "2024-12-03"] }, // Fri/Sat in UAE
   147→  AUH: { weekendDays: [6, 5], holidays: ["2024-12-02", "2024-12-03"] },
   148→  REMOTE: { weekendDays: [0, 6], holidays: [] },
   149→};
   150→
   151→/**
   152→ * Add hours but skip non-working time (weekends + holidays).
   153→ * For the prototype we approximate: if target_hours < 24 we use wall clock.
   154→ * For longer SLAs we count only working hours per day (8-18, Mon-Fri in UAE).
   155→ */
   156→export function computeDueAt(startIso: string, targetHours: number, location: string): string {
   157→  const start = new Date(startIso);
   158→  // Short SLAs (<24h) — use wall clock from start
   159→  if (targetHours <= 24) {
   160→    return new Date(start.getTime() + targetHours * 3600 * 1000).toISOString();
   161→  }
   162→  // Longer SLAs — calendar working days
   163→  const cal = CALENDARS[location] ?? CALENDARS.DXB;
   164→  let remainingMs = targetHours * 3600 * 1000;
   165→  let cursor = new Date(start);
   166→  // For prototype: assume 8 working hours per working day
   167→  const workHoursPerDay = 8;
   168→  const safetyMax = 365 * 24; // days cap
   169→  let iter = 0;
   170→  while (remainingMs > 0 && iter < safetyMax) {
   171→    iter++;
   172→    const dayOfWeek = cursor.getUTCDay();
   173→    const isoDay = cursor.toISOString().slice(0, 10);
   174→    const isWeekend = cal.weekendDays.includes(dayOfWeek);
   175→    const isHoliday = cal.holidays.includes(isoDay);
   176→    if (isWeekend || isHoliday) {
   177→      // skip full day
   178→      cursor = new Date(cursor.getTime() + 24 * 3600 * 1000);
   179→      continue;
   180→    }
   181→    const dayWorkMs = workHoursPerDay * 3600 * 1000;
   182→    if (remainingMs <= dayWorkMs) {
   183→      cursor = new Date(cursor.getTime() + remainingMs);
   184→      remainingMs = 0;
   185→    } else {
   186→      remainingMs -= dayWorkMs;
   187→      cursor = new Date(cursor.getTime() + 24 * 3600 * 1000);
   188→    }
   189→  }
   190→  return cursor.toISOString();
   191→}

   193→/**
   194→ * Breach level from elapsed/total ratio.
   195→ * Clock pauses while waiting_employee (handled by caller).
   196→ */
   197→export function breachLevel(elapsedHours: number, targetHours: number): StepState["breachLevel"] {
   198→  if (targetHours === 0) return "none";
   199→  const ratio = elapsedHours / targetHours;
   200→  if (ratio >= 2.0) return "hard";
   201→  if (ratio >= 1.0) return "breached";
   202→  if (ratio >= 0.8) return "due_soon";
   203→  return "none";
   204→}

   206→/* ============================================================
   207→   STEP STATE — derived from events
   208→   ============================================================ */
   209→export function computeStepState(
   210→  employee: Employee,
   211→  step: JourneyStep,
   212→  employeeEvents: JourneyEvent[],
   213→  asOf: Date
   214→): StepState {
   215→  // Filter events for this step at or before asOf
   216→  const stepEvents = employeeEvents
   217→    .filter((e) => e.stepKey === step.key && new Date(e.occurredAt) <= asOf)
   218→    .sort((a, b) => a.occurredAt.localeCompare(b.occurredAt));
   219→
   220→  let status: StepStatus = "not_started";
   221→  let startedAt: string | undefined;
   222→  let completedAt: string | undefined;
   223→  let isWaitingEmployee = false;
   224→
   225→  for (const ev of stepEvents) {
   226→    if (ev.type === "completed") {
   227→      status = "done";
   228→      completedAt = ev.occurredAt;
   229→    } else if (ev.type === "started") {
   230→      if (status !== "done") {
   231→        status = "in_progress";
   232→        startedAt = ev.occurredAt;
   233→      }
   234→    } else if (ev.type === "waiting_employee") {
   235→      status = "waiting_employee";
   236→      isWaitingEmployee = true;
   237→      if (!startedAt) startedAt = ev.occurredAt;
   238→    } else if (ev.type === "waiting_owner") {
   239→      if (status !== "done") status = "waiting_owner";
   240→      if (!startedAt) startedAt = ev.occurredAt;
   241→    } else if (ev.type === "blocked") {
   242→      if (status !== "done") status = "blocked";
   243→    } else if (ev.type === "skipped") {
   244→      status = "skipped";
   245→    }
   246→  }
   247→
   248→  // If dependencies not done → blocked
   249→  const blockedBy: string[] = [];
   250→  if (status !== "done" && status !== "skipped") {
   251→    for (const depKey of step.dependsOn) {
   252→      const depStep = ALL_STEPS.find((s) => s.key === depKey);
   253→      if (!depStep) continue;
   254→      const depState = computeStepState(employee, depStep, employeeEvents, asOf);
   255→      if (depState.status !== "done" && depState.status !== "skipped") {
   256→        blockedBy.push(depKey);
   257→      }
   258→    }
   259→    if (blockedBy.length > 0 && status === "not_started") {
   260→      status = "blocked";
   261→    }
   262→  }
   263→
   264→  // Compute SLA
   265→  const dueAt = startedAt ? computeDueAt(startedAt, step.targetHours, employee.location) : undefined;
   266→  const elapsedHours = startedAt && status !== "done"
   267→    ? (asOf.getTime() - new Date(startedAt).getTime()) / 3600000
   268→    : 0;
   269→  // Pause clock while waiting_employee
   270→  const effectiveElapsed = isWaitingEmployee ? 0 : elapsedHours;
   271→  const remainingHours = dueAt && status !== "done"
   272→    ? Math.max(0, (new Date(dueAt).getTime() - asOf.getTime()) / 3600000)
   273→    : 0;
   274→
   275→  const bl = (status === "done" || status === "skipped" || !startedAt)
   276→    ? "none"
   277→    : breachLevel(effectiveElapsed, step.targetHours);
   278→
   279→  // Compute downstream-at-risk count
   280→  const downstreamAtRisk = computeDownstreamAtRisk(step.key);
   281→
   282→  return {
   283→    stepKey: step.key,
   284→    status,
   285→    startedAt,
   286→    dueAt,
   287→    completedAt,
   288→    blockedBy,
   289→    breachLevel: bl,
   290→    targetHours: step.targetHours,
   291→    elapsedHours: effectiveElapsed,
   292→    remainingHours,
   293→    ownerGroup: step.ownerGroup,
   294→    actor: step.actor,
   295→    policyRefs: step.policyRefs,
   296→    dependsOn: step.dependsOn,
   297→    stage: step.stage,
   298→    label: step.label,
   299→    labelAr: step.labelAr,
   300→    requestType: step.requestType,
   301→    journeyType: step.journeyType,
   302→    downstreamAtRisk,
   303→  };
   304→}
   305→
   306→function computeDownstreamAtRisk(stepKey: string): number {
   307→  // Count steps that depend (transitively) on this step
   308→  const visited = new Set<string>();
   309→  const queue = [stepKey];
   310→  let count = 0;
   311→  while (queue.length > 0) {
   312→    const cur = queue.shift()!;
   313→    for (const s of ALL_STEPS) {
   314→      if (s.dependsOn.includes(cur) && !visited.has(s.key)) {
   315→        visited.add(s.key);
   316→        count++;
   317→        queue.push(s.key);
   318→      }
   319→    }
   320→  }
   321→  return count;
   322→}

   324→/* ============================================================
   325→   STAGE INFERENCE — conservative
   326→   ============================================================ */
   327→export function inferStage(
   328→  employee: Employee,
   329→  employeeEvents: JourneyEvent[],
   330→  asOf: Date
   331→): { stage: string; dayInStage: number } {
   332→  const journeyStages = ALL_STAGES
   333→    .filter((s) => s.journeyType === employee.journeyType)
   334→    .sort((a, b) => a.order - b.order);
   335→
   336→  // Compute step states for all steps
   337→  const journeySteps = ALL_STEPS.filter((s) => s.journeyType === employee.journeyType);
   338→  const stepStates = journeySteps.map((s) => computeStepState(employee, s, employeeEvents, asOf));
   339→
   340→  // Stage = latest stage whose entry condition is satisfied and exit is not
   341→  // For prototype: a stage is "entered" when its first step is started or completed.
   342→  // A stage is "exited" when all its steps are done.
   343→  let currentStage = journeyStages[0];
   344→  for (const st of journeyStages) {
   345→    const stSteps = stepStates.filter((s) => s.stage === st.key);
   346→    if (stSteps.length === 0) continue;
   347→    const anyStarted = stSteps.some((s) => s.status === "in_progress" || s.status === "done" || s.status === "waiting_employee" || s.status === "waiting_owner" || s.status === "blocked");
   348→    const allDone = stSteps.every((s) => s.status === "done" || s.status === "skipped");
   349→    if (anyStarted && !allDone) {
   350→      currentStage = st;
   351→      break; // conservative: first non-exited stage wins
   352→    }
   353→    if (allDone) {
   354→      currentStage = st; // last fully-done stage; we'll fall through to next
   355→    }
   356→  }
   357→
   358→  // Day in stage: find first event in this stage
   359→  const stageStepKeys = journeySteps.filter((s) => s.stage === currentStage.key).map((s) => s.key);
   360→  const firstEventInStage = employeeEvents
   361→    .filter((e) => stageStepKeys.includes(e.stepKey) && new Date(e.occurredAt) <= asOf)
   362→    .sort((a, b) => a.occurredAt.localeCompare(b.occurredAt))[0];
   363→  const dayInStage = firstEventInStage
   364→    ? Math.max(1, Math.floor((asOf.getTime() - new Date(firstEventInStage.occurredAt).getTime()) / (24 * 3600 * 1000)) + 1)
   365→    : 1;
   366→
   367→  return { stage: currentStage.key, dayInStage };
   368→}

   370→/* ============================================================
   371→   42 SIGNALS — pure predicates
   372→   ============================================================ */
   373→export function computeSignals(
   374→  employee: Employee,
   375→  stepStates: StepState[],
   376→  employeeRequests: ServiceRequest[],
   377→  employeePulse: PulseResponse[],
   378→  asOf: Date
   379→): { fired: FrictionSignal[]; suppressed: SignalFamily[] } {
   380→  const fired: FrictionSignal[] = [];
   381→  const suppressed: SignalFamily[] = [];
   382→
   383→  // Consent gates
   384→  const pulseConsent = getConsent(employee.id, "pulse_capture");
   385→  if (!pulseConsent) suppressed.push("PS");
   386→
   387→  // ---- TL family (timing) ----
   388→  let tlFamilyTotal = 0;
   389→  for (const s of stepStates) {
   390→    if (s.status === "done" || s.status === "skipped" || s.status === "not_started") continue;
   391→    // TL-01: step past target time (3 per step, family-capped)
   392→    if (s.breachLevel === "breached" || s.breachLevel === "hard") {
   393→      fired.push({
   394→        code: "TL-01",
   395→        family: "TL",
   396→        weight: 3,
   397→        evidenceRefs: [`step:${s.stepKey}`, `due:${s.dueAt}`, `elapsed:${s.elapsedHours.toFixed(1)}h`],
   398→        description: `${s.label} is past its target time (${s.elapsedHours.toFixed(0)}h vs ${s.targetHours}h target).`,
   399→        descriptionAr: `${s.labelAr} متأخر عن الوقت المستهدف.`,
   400→      });
   401→      tlFamilyTotal += 3;
   402→    }
   403→    // TL-02: step past 200% of target (hard breach, weight 6)
   404→    if (s.breachLevel === "hard") {
   405→      fired.push({
   406→        code: "TL-02",
   407→        family: "TL",
   408→        weight: 6,
   409→        evidenceRefs: [`step:${s.stepKey}`, `ratio:${(s.elapsedHours / s.targetHours).toFixed(2)}`],
   410→        description: `${s.label} is past 200% of target time — a hard breach.`,
   411→        descriptionAr: `${s.labelAr} تجاوز 200% من الوقت المستهدف — خرق جوهري.`,
   412→      });
   413→      tlFamilyTotal += 6;
   414→    }
   415→    // TL-05: No event for 72h while stage is active (weight 3)
   416→    if (s.status === "in_progress" && s.startedAt) {
   417→      const sinceStart = (asOf.getTime() - new Date(s.startedAt).getTime()) / 3600000;
   418→      // Approximate: if no other event in 72h on any step in same stage
   419→      // For prototype we check this step's last event
   420→      if (sinceStart > 72) {
   421→        // Check no other event for this step in last 72h
   422→        // (simplified for prototype)
   423→      }
   424→    }
   425→  }
   426→
   427→  // TL-04: Start date in ≤5 days and pre-boarding completion < 60% (weight 4)
   428→  const daysToStart = (new Date(employee.startDate).getTime() - asOf.getTime()) / (24 * 3600 * 1000);
   429→  if (daysToStart >= 0 && daysToStart <= 5 && employee.journeyType === "onboarding") {
   430→    const preSteps = stepStates.filter((s) => s.stage === "pre_boarding");
   431→    const doneCount = preSteps.filter((s) => s.status === "done").length;
   432→    const pct = preSteps.length > 0 ? doneCount / preSteps.length : 0;
   433→    if (pct < 0.6) {
   434→      fired.push({
   435→        code: "TL-04",
   436→        family: "TL",
   437→        weight: 4,
   438→        evidenceRefs: [`startDate:${employee.startDate}`, `preBoardingPct:${pct.toFixed(2)}`],
   439→        description: `Start date is in ${Math.ceil(daysToStart)} day(s) but pre-boarding is only ${Math.round(pct * 100)}% complete.`,
   440→        descriptionAr: `تاريخ البدء بعد ${Math.ceil(daysToStart)} يوم/أيام لكن ما قبل الالتحاق مكتمل بنسبة ${Math.round(pct * 100)}% فقط.`,
   441→      });
   442→    }
   443→  }
   444→
   445→  // ---- DP family (dependencies) ----
   446→  for (const s of stepStates) {
   447→    if (s.status === "done" || s.status === "skipped") continue;
   448→    // DP-01: Incomplete step blocks ≥2 downstream steps (weight 5)
   449→    if (s.downstreamAtRisk >= 2 && (s.status === "in_progress" || s.status === "blocked")) {
   450→      fired.push({
   451→        code: "DP-01",
   452→        family: "DP",
   453→        weight: 5,
   454→        evidenceRefs: [`step:${s.stepKey}`, `downstream:${s.downstreamAtRisk}`],
   455→        description: `${s.label} is incomplete and blocks ${s.downstreamAtRisk} downstream steps.`,
   456→        descriptionAr: `${s.labelAr} غير مكتمل ويحجب ${s.downstreamAtRisk} خطوات لاحقة.`,
   457→      });
   458→    }
   459→    // DP-02: Visa / work-authorisation step stalled > 7 days (hard trip, weight 7)
   460→    if (
   461→      (s.stepKey === "ob_visa_application" || s.stepKey === "ob_visa_approval") &&
   462→      s.status === "in_progress" &&
   463→      s.startedAt
   464→    ) {
   465→      const stalled = (asOf.getTime() - new Date(s.startedAt).getTime()) / (24 * 3600 * 1000);
   466→      if (stalled > 7) {
   467→        fired.push({
   468→          code: "DP-02",
   469→          family: "DP",
   470→          weight: 7,
   471→          hardTrip: true,
   472→          forceBlocked: s.stepKey === "ob_visa_application",
   473→          evidenceRefs: [`step:${s.stepKey}`, `stalledDays:${stalled.toFixed(1)}`],
   474→          description: `Visa step ${s.label} has been stalled for ${stalled.toFixed(0)} days — hard trip.`,
   475→          descriptionAr: `خطوة التأشيرة ${s.labelAr} متوقفة منذ ${stalled.toFixed(0)} أيام — تعطيل جوهري.`,
   476→        });
   477→      }
   478→    }
   479→    // DP-03: Access provisioning blocked by missing ID or asset (weight 4)
   480→    if (s.stepKey.includes("access") && s.status === "blocked") {
   481→      fired.push({
   482→        code: "DP-03",
   483→        family: "DP",
   484→        weight: 4,
   485→        evidenceRefs: [`step:${s.stepKey}`, `blockedBy:${s.blockedBy.join(",")}`],
   486→        description: `Access provisioning is blocked by missing prerequisites.`,
   487→        descriptionAr: `توفير الوصول محجوب بسبب متطلبات مفقودة.`,
   488→      });
   489→    }
   490→    // DP-05: Transfer approved, releasing-manager handover absent > 5 days (weight 4)
   491→    if (s.stepKey === "mb_release_handover" && s.status === "in_progress" && s.startedAt) {
   492→      const stalled = (asOf.getTime() - new Date(s.startedAt).getTime()) / (24 * 3600 * 1000);
   493→      if (stalled > 5) {
   494→        fired.push({
   495→          code: "DP-05",
   496→          family: "DP",
   497→          weight: 4,
   498→          evidenceRefs: [`step:${s.stepKey}`, `stalledDays:${stalled.toFixed(1)}`],
   499→          description: `Releasing manager handover has been pending for ${stalled.toFixed(0)} days.`,
   500→          descriptionAr: `تسليم المدير السابق معلق منذ ${stalled.toFixed(0)} أيام.`,
   501→        });
   502→      }
   503→    }
   504→  }
   505→
   506→  // ---- RQ family (requests) ----
   507→  for (const r of employeeRequests) {
   508→    const ageHours = (asOf.getTime() - new Date(r.openedAt).getTime()) / 3600000;
   509→    // RQ-01: Request open beyond its type SLA (weight 3)
   510→    if (ageHours > r.slaHours) {
   511→      fired.push({
   512→        code: "RQ-01",
   513→        family: "RQ",
   514→        weight: 3,
   515→        evidenceRefs: [`request:${r.ref}`, `ageHours:${ageHours.toFixed(0)}`, `sla:${r.slaHours}`],
   516→        description: `Request ${r.ref} is open ${ageHours.toFixed(0)}h vs ${r.slaHours}h SLA.`,
   517→        descriptionAr: `الطلب ${r.ref} مفتوح منذ ${ageHours.toFixed(0)} ساعة مقابل ${r.slaHours} ساعة مستهدفة.`,
   518→      });
   519→    }
   520→    // RQ-02: Request reassigned ≥3 times (weight 4)
   521→    if (r.reassignCount >= 3) {
   522→      fired.push({
   523→        code: "RQ-02",
   524→        family: "RQ",
   525→        weight: 4,
   526→        evidenceRefs: [`request:${r.ref}`, `reassignCount:${r.reassignCount}`],
   527→        description: `Request ${r.ref} has been reassigned ${r.reassignCount} times (queue ping-pong).`,
   528→        descriptionAr: `الطلب ${r.ref} تمت إعادة تعيينه ${r.reassignCount} مرة (ترقّص بين الطوابير).`,
   529→      });
   530→    }
   531→    // RQ-03: Request returned as incomplete ≥2 times (weight 4)
   532→    if (r.returnCount >= 2) {
   533→      fired.push({
   534→        code: "RQ-03",
   535→        family: "RQ",
   536→        weight: 4,
   537→        evidenceRefs: [`request:${r.ref}`, `returnCount:${r.returnCount}`],
   538→        description: `Request ${r.ref} returned as incomplete ${r.returnCount} times.`,
   539→        descriptionAr: `الطلب ${r.ref} تم إرجاعه كغير مكتمل ${r.returnCount} مرة.`,
   540→      });
   541→    }
   542→  }
   543→
   544→  // ---- PS family (pulse, consent-gated) ----
   545→  if (pulseConsent && suppressed.indexOf("PS") === -1) {
   546→    for (const p of employeePulse) {
   547→      // PS-01: Opt-in pulse confidence ≤2/5 at a stage gate (weight 4)
   548→      if (p.score <= 2) {
   549→        fired.push({
   550→          code: "PS-01",
   551→          family: "PS",
   552→          weight: 4,
   553→          evidenceRefs: [`pulse:${p.id}`, `score:${p.score}`, `stage:${p.stage}`],
   554→          description: `Pulse score at ${p.stage} was ${p.score}/5 — low confidence.`,
   555→          descriptionAr: `نتيجة النبضة في ${p.stage} كانت ${p.score}/5 — ثقة منخفضة.`,
   556→        });
   557→      }
   558→      // PS-05: Explicit "I need help now" flag (weight 6, hard trip)
   559→      // Detected via comment lexicon (deterministic, model labels only)
   560→      if (p.commentText && /(stuck|nobody replied|i need help now|no one told me)/i.test(p.commentText)) {
   561→        fired.push({
   562→          code: "PS-05",
   563→          family: "PS",
   564→          weight: 6,
   565→          hardTrip: true,
   566→          forceBlocked: false,
   567→          evidenceRefs: [`pulse:${p.id}`, `commentRef:${p.commentRef}`, `lexicon:help_seeking`],
   568→          description: `Employee comment signals help-seeking intent.`,
   569→          descriptionAr: `تعليق الموظف يشير إلى طلب المساعدة.`,
   570→        });
   571→      }
   572→    }
   573→  } else {
   574→    // PS-04: Pulse consent absent → family suppressed, surfaced in explanation
   575→    if (suppressed.indexOf("PS") !== -1) {
   576→      // Do not add a signal; suppression is disclosed separately
   577→    }
   578→  }
   579→
   580→  // ---- MB family (mobility) ----
   581→  if (employee.journeyType === "mobility" && employee.effectiveDate) {
   582→    const daysSinceEffective = (asOf.getTime() - new Date(employee.effectiveDate).getTime()) / (24 * 3600 * 1000);
   583→    // MB-01: Effective date set but role access profile unchanged > 3 days (weight 4)
   584→    const roleAccessStep = stepStates.find((s) => s.stepKey === "mb_role_access");
   585→    if (roleAccessStep && roleAccessStep.status === "in_progress" && daysSinceEffective > 3) {
   586→      fired.push({
   587→        code: "MB-01",
   588→        family: "MB",
   589→        weight: 4,
   590→        evidenceRefs: [`step:mb_role_access`, `daysSinceEffective:${daysSinceEffective.toFixed(0)}`],
   591→        description: `Role access profile unchanged for ${daysSinceEffective.toFixed(0)} days after effective date.`,
   592→        descriptionAr: `ملف الوصول للدور لم يتغير منذ ${daysSinceEffective.toFixed(0)} أيام من تاريخ السريان.`,
   593→      });
   594→    }
   595→    // MB-04: Payroll/allowance change unconfirmed, cycle cutoff < 7 days (hard trip, weight 6)
   596→    const payrollStep = stepStates.find((s) => s.stepKey === "mb_payroll_change");
   597→    if (payrollStep && payrollStep.status === "in_progress") {
   598→      // Days to next payroll cutoff (25th of month)
   599→      const nextCutoff = new Date(asOf);
   600→      nextCutoff.setUTCDate(25);
   601→      if (nextCutoff <= asOf) nextCutoff.setUTCMonth(nextCutoff.getUTCMonth() + 1);
   602→      const daysToCutoff = (nextCutoff.getTime() - asOf.getTime()) / (24 * 3600 * 1000);
   603→      if (daysToCutoff < 7) {
   604→        fired.push({
   605→          code: "MB-04",
   606→          family: "MB",
   607→          weight: 6,
   608→          hardTrip: true,
   609→          forceBlocked: true,
   610→          evidenceRefs: [`step:mb_payroll_change`, `daysToCutoff:${daysToCutoff.toFixed(0)}`],
   611→          description: `Payroll change unconfirmed and cycle cutoff is in ${daysToCutoff.toFixed(0)} days — hard trip.`,
   612→          descriptionAr: `تغيير الراتب غير مؤكد وتقطع دورة الرواتب بعد ${daysToCutoff.toFixed(0)} أيام — تعطيل جوهري.`,
   613→        });
   614→      }
   615→    }
   616→    // MB-03: Dual reporting unresolved > 7 days after effective date (weight 3)
   617→    const dualStep = stepStates.find((s) => s.stepKey === "mb_dual_reporting");
   618→    if (dualStep && dualStep.status === "waiting_owner" && daysSinceEffective > 7) {
   619→      fired.push({
   620→        code: "MB-03",
   621→        family: "MB",
   622→        weight: 3,
   623→        evidenceRefs: [`step:mb_dual_reporting`, `daysSinceEffective:${daysSinceEffective.toFixed(0)}`],
   624→        description: `Dual reporting unresolved for ${daysSinceEffective.toFixed(0)} days after effective date.`,
   625→        descriptionAr: `التقارير المزدوجة غير محلولة منذ ${daysSinceEffective.toFixed(0)} أيام من تاريخ السريان.`,
   626→      });
   627→    }
   628→  }
   629→
   630→  // ---- DQ family (data quality) ----
   631→  // DQ-01: Manager or HRBP mapping missing (weight 4)
   632→  if (!employee.managerId || !employee.hrbpId) {
   633→    fired.push({
   634→      code: "DQ-01",
   635→      family: "DQ",
   636→      weight: 4,
   637→      evidenceRefs: [`employee:${employee.id}`],
   638→      description: `Manager or HRBP mapping missing on employee record.`,
   639→      descriptionAr: `تعيين المدير أو شريك الموارد البشرية مفقود في سجل الموظف.`,
   640→    });
   641→  }
   642→  // DQ-05: Step references a policy absent from the corpus
   643→  // (simplified — for prototype assume all policy refs resolve)
   644→
   645→  return { fired, suppressed };
   646→}

   648→/* ============================================================
   649→   FUSION — bands from capped signal weights + hard trips
   650→   ============================================================ */
   651→export function fuse(signals: FrictionSignal[], suppressed: SignalFamily[]): {
   652→  score: number;
   653→  band: Band;
   654→  hardTrips: string[];
   655→  forcedBlocked: boolean;
   656→} {
   657→  // Apply family caps
   658→  const familyTotals: Record<string, number> = {};
   659→  const hardTrips: string[] = [];
   660→  let forcedBlocked = false;
   661→
   662→  for (const sig of signals) {
   663→    const fam = sig.family;
   664→    const capped = Math.min(sig.weight, FAMILY_CAPS[fam]);
   665→    familyTotals[fam] = (familyTotals[fam] ?? 0) + capped;
   666→    if (sig.hardTrip) hardTrips.push(sig.code);
   667→    if (sig.forceBlocked) forcedBlocked = true;
   668→  }
   669→
   670→  let score = 0;
   671→  for (const [fam, total] of Object.entries(familyTotals)) {
   672→    score += Math.min(total, FAMILY_CAPS[fam as SignalFamily]);
   673→  }
   674→
   675→  let band: Band = "steady";
   676→  if (score >= 20) band = "blocked";
   677→  else if (score >= 12) band = "friction";
   678→  else if (score >= 6) band = "watch";
   679→
   680→  // Hard trips: any weight≥6 forces at least Friction
   681→  if (hardTrips.length > 0 && band === "steady") band = "watch";
   682→  if (hardTrips.length > 0 && (band === "steady" || band === "watch")) band = "friction";
   683→  // DP-02, PS-05, MB-04 force Blocked
   684→  if (hardTrips.some((c) => c === "DP-02" || c === "PS-05" || c === "MB-04") || forcedBlocked) {
   685→    band = "blocked";
   686→  }
   687→
   688→  return { score, band, hardTrips, forcedBlocked };
   689→}

   691→/* ============================================================
   692→   ASSESSMENT — top-level verdict
   693→   ============================================================ */
   694→export function assess(
   695→  employee: Employee,
   696→  asOf: Date = new Date()
   697→): Assessment {
   698→  const employeeEvents = eventsForEmployee(employee.id);
   699→  const employeeRequests = ALL_REQUESTS.filter((r) => r.employeeId === employee.id);
   700→  const employeePulse = ALL_PULSE.filter((p) => p.employeeId === employee.id);
   701→
   702→  const journeySteps = ALL_STEPS.filter((s) => s.journeyType === employee.journeyType);
   703→  const stepStates = journeySteps.map((s) => computeStepState(employee, s, employeeEvents, asOf));
   704→
   705→  const { fired, suppressed } = computeSignals(employee, stepStates, employeeRequests, employeePulse, asOf);
   706→  const fusion = fuse(fired, suppressed);
   707→
   708→  const { stage, dayInStage } = inferStage(employee, employeeEvents, asOf);
   709→
   710→  return {
   711→    employeeId: employee.id,
   712→    asOf: asOf.toISOString(),
   713→    band: fusion.band,
   714→    score: fusion.score,
   715→    signals: fired,
   716→    suppressedFamilies: suppressed,
   717→    mapVersion: "map-1.0.0",
   718→    corpusVersion: "policies-2024-q4-v1",
   719→    engineVersion: ENGINE_VERSION,
   720→    stage,
   721→    dayInStage,
   722→  };
   723→}

   729→/* ============================================================
   730→   NEXT BEST STEP — ranked
   731→   ============================================================ */
   732→export function nextBestStep(employee: Employee, asOf: Date = new Date()): NextBestStep | null {
   733→  const employeeEvents = eventsForEmployee(employee.id);
   734→  const journeySteps = ALL_STEPS.filter((s) => s.journeyType === employee.journeyType);
   735→  const stepStates = journeySteps.map((s) => computeStepState(employee, s, employeeEvents, asOf));
   736→
   737→  // Candidates: not done, dependencies satisfied, in current or next stage
   738→  const { stage } = inferStage(employee, employeeEvents, asOf);
   739→  const journeyStages = ALL_STAGES.filter((s) => s.journeyType === employee.journeyType).sort((a, b) => a.order - b.order);
   740→  const stageIdx = journeyStages.findIndex((s) => s.key === stage);
   741→  const allowedStages = journeyStages.slice(stageIdx, stageIdx + 2).map((s) => s.key);
   742→
   743→  const candidates = stepStates.filter((s) => {
   744→    if (s.status === "done" || s.status === "skipped") return false;
   745→    if (!allowedStages.includes(s.stage)) return false;
   746→    if (s.status === "blocked") return false;
   747→    return true;
   748→  });
   749→
   750→  if (candidates.length === 0) return null;
   751→
   752→  // Rank: hard breach > most downstream blocked > nearest due > employee-actionable > least effort
   753→  candidates.sort((a, b) => {
   754→    const aHard = a.breachLevel === "hard" ? 1 : 0;
   755→    const bHard = b.breachLevel === "hard" ? 1 : 0;
   756→    if (aHard !== bHard) return bHard - aHard;
   757→    if (b.downstreamAtRisk !== a.downstreamAtRisk) return b.downstreamAtRisk - a.downstreamAtRisk;
   758→    const aDue = a.dueAt ? new Date(a.dueAt).getTime() : Infinity;
   759→    const bDue = b.dueAt ? new Date(b.dueAt).getTime() : Infinity;
   760→    if (aDue !== bDue) return aDue - bDue;
   761→    const aEmp = a.actor === "employee" ? 0 : 1;
   762→    const bEmp = b.actor === "employee" ? 0 : 1;
   763→    if (aEmp !== bEmp) return aEmp - bEmp;
   764→    return a.targetHours - b.targetHours;
   765→  });
   766→
   767→  const top = candidates[0];
   768→  const step = journeySteps.find((s) => s.key === top.stepKey)!;
   769→  let reason = `This unblocks ${top.downstreamAtRisk} step(s) and is the most urgent item.`;
   770→  let reasonAr = `هذا يحرر ${top.downstreamAtRisk} خطوة/خطوات وهو العنصر الأكثر إلحاحاً.`;
   771→  if (top.breachLevel === "hard") {
   772→    reason = `${step.label} is past 200% of its target — this is the most urgent item.`;
   773→    reasonAr = `${step.labelAr} تجاوز 200% من المستهدف — هذا هو العنصر الأكثر إلحاحاً.`;
   774→  } else if (top.actor === "employee") {
   775→    reason = `${step.label} is yours to do — and it unblocks the next ${top.downstreamAtRisk} step(s).`;
   776→    reasonAr = `${step.labelAr} خطوتك للقيام بها — وتحرر ${top.downstreamAtRisk} خطوة/خطوات لاحقة.`;
   777→  }
   778→
   779→  return {
   780→    stepKey: top.stepKey,
   781→    label: step.label,
   782→    labelAr: step.labelAr,
   783→    ownerGroup: step.ownerGroup,
   784→    dueAt: top.dueAt,
   785→    policyRefs: step.policyRefs,
   786→    actor: step.actor,
   787→    reason,
   788→    reasonAr,
   789→    requestType: step.requestType,
   790→    primary: true,
   791→  };
   792→}

   794→/* ============================================================
   795→   ENGINE TRACE — judge mode visibility
   796→   ============================================================ */
   797→export function trace(employee: Employee, asOf: Date = new Date()): EngineTrace {
   798→  const assessment = assess(employee, asOf);
   799→  const employeeEvents = eventsForEmployee(employee.id);
   800→  const journeySteps = ALL_STEPS.filter((s) => s.journeyType === employee.journeyType);
   801→  const stepStates = journeySteps.map((s) => computeStepState(employee, s, employeeEvents, asOf));
   802→  const employeeRequests = ALL_REQUESTS.filter((r) => r.employeeId === employee.id);
   803→  const employeePulse = ALL_PULSE.filter((p) => p.employeeId === employee.id);
   804→  const { fired, suppressed } = computeSignals(employee, stepStates, employeeRequests, employeePulse, asOf);
   805→  const fusion = fuse(fired, suppressed);
   806→  const nbs = nextBestStep(employee, asOf);
   807→
   808→  return {
   809→    stageInference: {
   810→      stage: assessment.stage,
   811→      reason: `Latest stage with started activity and not all steps done (conservative tie-break).`,
   812→    },
   813→    signalsFired: fired.map((s) => ({
   814→      code: s.code,
   815→      weight: s.weight,
   816→      family: s.family,
   817→      evidence: s.evidenceRefs,
   818→    })),
   819→    signalsSuppressed: suppressed.map((family) => ({
   820→      family,
   821→      reason: family === "PS" ? "Pulse consent not granted — family suppressed, contributes 0." : "Unknown",
   822→    })),
   823→    band: {
   824→      score: fusion.score,
   825→      band: fusion.band,
   826→      hardTrips: fusion.hardTrips,
   827→      forcedBlocked: fusion.forcedBlocked,
   828→    },
   829→    nextBestStep: nbs ? { stepKey: nbs.stepKey, reason: nbs.reason } : null,
   830→    servedFrom: "live",
   831→    degraded: [],
   832→  };
   833→}
