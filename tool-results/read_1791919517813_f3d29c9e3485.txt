   492→/* ============================================================
   493→   HERO PARALLAX — 3D astrolabe + hero text
   494→   - Astrolabe canvas shifts max 8px opposite the cursor
   495→   - Hero text shifts max 4px with the cursor
   496→   - 60fps via framer-motion useMotionValue + useTransform
   497→   - Disabled on touch devices and when prefers-reduced-motion
   498→   ============================================================ */
   499→
   500→function usePrefersReducedMotion(): boolean {
   501→  const [reduce, setReduce] = useState(false);
   502→  useEffect(() => {
   503→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   504→    const update = () => setReduce(mq.matches);
   505→    update();
   506→    mq.addEventListener("change", update);
   507→    return () => mq.removeEventListener("change", update);
   508→  }, []);
   509→  return reduce;
   510→}
   511→
   512→function useFinePointer(): boolean {
   513→  const [fine, setFine] = useState(false);
   514→  useEffect(() => {
   515→    const mq = window.matchMedia("(pointer: fine)");
   516→    const update = () => setFine(mq.matches);
   517→    update();
   518→    mq.addEventListener("change", update);
   519→    return () => mq.removeEventListener("change", update);
   520→  }, []);
   521→  return fine;
   522→}

   524→function HeroParallax() {
   525→  const locale = useAppStore((s) => s.locale);
   526→  const setView = useAppStore((s) => s.setView);
   527→  const reduce = usePrefersReducedMotion();
   528→  const fine = useFinePointer();
   529→  const enabled = !reduce && fine;
   530→
   531→  // Normalised cursor offset from the centre of the hero, in [-0.5, 0.5].
   532→  const mx = useMotionValue(0);
   533→  const my = useMotionValue(0);
   534→
   535→  // Astrolabe: 8px in the opposite direction of the cursor.
   536→  const astX: MotionValue<number> = useTransform(mx, [-0.5, 0.5], [8, -8]);
   537→  const astY: MotionValue<number> = useTransform(my, [-0.5, 0.5], [8, -8]);
   538→  // Hero text: 4px in the same direction as the cursor.
   539→  const textX: MotionValue<number> = useTransform(mx, [-0.5, 0.5], [-4, 4]);
   540→  const textY: MotionValue<number> = useTransform(my, [-0.5, 0.5], [-4, 4]);
   541→
   542→  const handleMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
   543→    if (!enabled) return;
   544→    const rect = e.currentTarget.getBoundingClientRect();
   545→    mx.set((e.clientX - rect.left) / rect.width - 0.5);
   546→    my.set((e.clientY - rect.top) / rect.height - 0.5);
   547→  };
   548→
   549→  const handleMouseLeave = () => {
   550→    mx.set(0);
   551→    my.set(0);
   552→  };
