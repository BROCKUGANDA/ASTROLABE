   196→/* ---------- Helpers ---------- */
   197→
   198→function latestConsent(records: ConsentRecord[], purpose: Purpose): ConsentRecord | undefined {
   199→  return records
   200→    .filter((r) => r.purpose === purpose)
   201→    .sort((a, b) => b.grantedAt.localeCompare(a.grantedAt))[0];
   202→}
   203→
   204→function purposeGranted(
   205→  records: ConsentRecord[],
   206→  overrides: Record<string, boolean>,
   207→  meta: PurposeMeta
   208→): boolean {
   209→  if (typeof overrides[meta.key] === "boolean") return overrides[meta.key];
   210→  const latest = latestConsent(records, meta.key);
   211→  if (latest) return latest.granted;
   212→  return meta.defaultOn;
   213→}

   291→
   292→  const [data, setData] = useState<MeData | null>(null);
   293→  const [loading, setLoading] = useState(true);
   294→  const [err, setErr] = useState<string | null>(null);
   295→  const [pendingPurpose, setPendingPurpose] = useState<Purpose | null>(null);
   296→  const [expandedRow, setExpandedRow] = useState<string | null>(null);
   297→  const [revokeOpen, setRevokeOpen] = useState(false);
   298→
   299→  const fetchData = useCallback(async () => {
   300→    setLoading(true);
   301→    setErr(null);
   302→    try {
   303→      const res = await fetch(`/api/v1/me/data?employeeId=${encodeURIComponent(employeeId)}`, { cache: "no-store" });
   304→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   305→      const json: ApiEnvelope<MeData> = await res.json();
   306→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   307→      setData(json.data);
   308→    } catch (e) {
   309→      setErr(e instanceof Error ? e.message : "Network error");
   310→    } finally {
   311→      setLoading(false);
   312→    }
   313→  }, [employeeId]);
   314→
   315→  useEffect(() => {
   316→    fetchData();
   317→  }, [fetchData]);
   318→
   319→  const pulseOn = useMemo(() => {
   320→    if (!data) return false;
   321→    return purposeGranted(data.consentRecords, overrides, PURPOSES[1]);
   322→  }, [data, overrides]);
   323→
   324→  const handleToggle = useCallback(
   325→    async (meta: PurposeMeta, next: boolean) => {
   326→      if (meta.required) return;
   327→      setPendingPurpose(meta.key);
   328→      // Optimistic update via the store
   329→      setConsent(meta.key, next);
   330→      try {
   331→        const res = await fetch("/api/v1/consent", {
   332→          method: "POST",
   333→          headers: { "Content-Type": "application/json" },
   334→          body: JSON.stringify({ employeeId, purpose: meta.key, granted: next, version: "v1" }),
   335→        });
   336→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   337→        const json: ApiEnvelope<{ receiptId: string; grantedAt: string }> = await res.json();
   338→        if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   339→        const stamp = format(new Date(json.data.grantedAt), "HH:mm");
   340→        if (next) {
   341→          toast.success(`Consent recorded · ${json.data.receiptId} at ${stamp}`, {
   342→            description: meta.labelEn,
   343→          });
   344→        } else {
   345→          toast(`Revoked · ${json.data.receiptId}`, {
   346→            description: meta.labelEn,
   347→          });
   348→        }
   349→        // Refetch so the snapshot reflects the server
   350→        await fetchData();
   351→      } catch (e) {
   352→        // Roll back
   353→        setConsent(meta.key, !next);
   354→        toast.error("Could not update consent", {
   355→          description: e instanceof Error ? e.message : "Network error",
   356→        });
   357→      } finally {
   358→        setPendingPurpose(null);
   359→      }
   360→    },
   361→    [employeeId, fetchData, setConsent]
   362→  );
   363→
   364→  const handleRevokeAll = useCallback(async () => {
   365→    const optionalPurposes = PURPOSES.filter((p) => !p.required);
   366→    for (const p of optionalPurposes) {
   367→      const currentlyOn = data ? purposeGranted(data.consentRecords, overrides, p) : false;
   368→      if (currentlyOn) {
   369→        setConsent(p.key, false);
   370→        try {
   371→          await fetch("/api/v1/consent", {
   372→            method: "POST",
   373→            headers: { "Content-Type": "application/json" },
   374→            body: JSON.stringify({ employeeId, purpose: p.key, granted: false, version: "v1" }),
   375→          });
   376→        } catch {
   377→          /* ignore — already rolled back state */
   378→        }
   379→      }
   380→    }
   381→    setRevokeOpen(false);
   382→    toast("All optional consents revoked", {
   383→      description: "Pulse, sharing, comment retention, manager visibility, and analytics are off. Journey assistance remains on.",
   384→    });
   385→    await fetchData();
   386→  }, [data, employeeId, fetchData, overrides, setConsent]);

   395→    document.body.appendChild(a);
   396→    a.click();
   397→    document.body.removeChild(a);
   398→    URL.revokeObjectURL(url);
   399→    toast.success("Exported as JSON", { description: `${employeeId} · snapshot ${data.snapshotAt.slice(0, 10)}` });
   400→  }, [data, employeeId]);
