   623→export function getConsent(employeeId: string, purpose: ConsentRecord["purpose"]): boolean {
   624→  const records = consentRecords
   625→    .filter((r) => r.employeeId === employeeId && r.purpose === purpose)
   626→    .sort((a, b) => b.grantedAt.localeCompare(a.grantedAt));
   627→  if (records.length === 0) {
   628→    if (purpose === "journey_assistance") return true;
   629→    if (purpose === "manager_visibility") return true;
   630→    if (purpose === "aggregate_analytics") return true;
   631→    return false;
   632→  }
   633→  const latest = records[0];
   634→  return latest.granted && !latest.revokedAt;
   635→}
