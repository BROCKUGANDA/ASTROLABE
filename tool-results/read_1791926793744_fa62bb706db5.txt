   239→/* ============================================================
   240→   Component
   241→   ============================================================ */
   242→export function ManagerScreen() {
   243→  const locale = useAppStore((s) => s.locale);
   244→  const view = useAppStore((s) => s.managerView);
   245→  const dir = locale === "ar" ? "rtl" : "ltr";
   246→
   247→  const [items, setItems] = useState<CohortItem[] | null>(null);
   248→  const [err, setErr] = useState<string | null>(null);
   249→  const [loading, setLoading] = useState(true);
   250→
   251→  const fetchCohort = useCallback(async () => {
   252→    setLoading(true);
   253→    setErr(null);
   254→    try {
   255→      const res = await fetch("/api/v1/manager/cohort?managerId=MGR-1", { cache: "no-store" });
   256→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   257→      const json: ApiEnvelope<{ managerId: string; items: CohortItem[] }> = await res.json();
   258→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   259→      setItems(json.data.items);
   260→    } catch (e) {
   261→      setErr(e instanceof Error ? e.message : "Network error");
   262→    } finally {
   263→      setLoading(false);
   264→    }
   265→  }, []);
   266→
   267→  useEffect(() => { fetchCohort(); }, [fetchCohort]);
   268→
   269→  // Aggregate metrics
   270→  const stats = useMemo(() => {
   271→    if (!items) return { joiners: 0, waitingOnMe: 0, avgCompletion: 0 };
   272→    const joiners = items.length;
   273→    let waiting = 0;
   274→    let sumCompletion = 0;
   275→    for (const it of items) {
   276→      sumCompletion += it.aggregateCompletion;
   277→      for (const s of it.ownedSteps) {
   278→        if (s.status === "waiting_owner") waiting += 1;
   279→      }
   280→    }
   281→    return {
   282→      joiners,
   283→      waitingOnMe: waiting,
   284→      avgCompletion: joiners > 0 ? Math.round(sumCompletion / joiners) : 0,
   285→    };
   286→  }, [items]);
   287→
   288→  // Flatten owned steps across cohort (for "my obligations")
   289→  const allObligations = useMemo(() => {
   290→    if (!items) return [] as { employee: CohortItem; step: OwnedStep }[];
   291→    const out: { employee: CohortItem; step: OwnedStep }[] = [];
   292→    for (const it of items) {
   293→      for (const s of it.ownedSteps) {
   294→        out.push({ employee: it, step: s });
   295→      }
   296→    }
   297→    // Sort: waiting_owner first, then by dueAt asc
   298→    out.sort((a, b) => {
   299→      const aWait = a.step.status === "waiting_owner" ? 0 : 1;
   300→      const bWait = b.step.status === "waiting_owner" ? 0 : 1;
   301→      if (aWait !== bWait) return aWait - bWait;
   302→      const aDue = a.step.dueAt ? new Date(a.step.dueAt).getTime() : Number.MAX_SAFE_INTEGER;
   303→      const bDue = b.step.dueAt ? new Date(b.step.dueAt).getTime() : Number.MAX_SAFE_INTEGER;
   304→      return aDue - bDue;
   305→    });
   306→    return out;
   307→  }, [items]);
   308→
   309→  return (
   310→    <div
   311→      dir={dir}
   312→      className="max-w-[1180px] mx-auto px-4 sm:px-6 lg:px-8 py-6 lg:py-10"
   313→    >
   314→      <header className="flex items-center justify-between mb-6 flex-wrap gap-3">
   315→        <div>
   316→          <div className="flex items-center gap-2 text-[11px] font-semibold uppercase tracking-[0.18em]" style={{ color: "var(--brand-400)" }}>
   317→            <Users className="w-3.5 h-3.5" strokeWidth={1.75} />
   318→            <span>{locale === "ar" ? "وحدة المدير" : "Manager console"}</span>
   319→          </div>
   320→          <h1 className="mt-2 text-[26px] sm:text-[30px] font-semibold tracking-tight" style={{ color: "var(--text-hi)" }}>
   321→            {locale === "ar" ? "التزاماتك تجاه فريقك" : "Your obligations to your team"}
   322→          </h1>
   323→        </div>
   324→        <ManagerTabs />
   325→      </header>
   326→
   327→      <PrivacyBanner locale={locale} />
   328→
   329→      {loading ? (
   330→        <ManagerSkeleton />
   331→      ) : err ? (
   332→        <ErrorBanner message={err} onRetry={fetchCohort} />
   333→      ) : !items ? null : items.length === 0 ? (
   334→        <div className="ast-card p-10 flex flex-col items-center text-center">
   335→          <div
   336→            className="flex items-center justify-center w-14 h-14 rounded-full mb-4"
   337→            style={{ background: "var(--surface-2)", color: "var(--text-mid)" }}
   338→          >
   339→            <Inbox className="w-6 h-6" strokeWidth={1.75} />
   340→          </div>
   341→          <p className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   342→            {locale === "ar" ? "لا شيء بانتظارك." : "Nothing waiting on you."}
   343→          </p>
   344→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   345→            {locale === "ar"
   346→              ? "عندما يصل موظف جديد أو ينتقل أحد، ستظهر التزاماتك هنا."
   347→              : "When a joiner or mover lands on your team, your obligations will appear here."}
   348→          </p>
   349→        </div>
   350→      ) : (
   351→        <>
   352→          {/* Hero numbers */}
   353→          <section className="grid grid-cols-1 sm:grid-cols-3 gap-3 mb-5">
   354→            {/* Waiting on me — sand-gold (primary) */}
   355→            <div
   356→              className="ast-card p-5 relative overflow-hidden"
   357→              style={{
   358→                borderColor: "rgba(201,162,39,0.40)",
   359→                background: "linear-gradient(135deg, rgba(201,162,39,0.10) 0%, var(--surface-1) 60%)",
   360→              }}
   361→            >
   362→              <div
   363→                className="absolute top-0 left-0 right-0 h-[3px]"
   364→                style={{ background: "var(--accent-500)" }}
   365→              />
   366→              <p className="text-[11px] uppercase tracking-[0.14em] font-medium" style={{ color: "var(--accent-500)" }}>
   367→                {locale === "ar" ? "بانتظارك" : "Waiting on you"}
   368→              </p>
   369→              <p className="text-[44px] leading-none font-semibold mt-2 tabular ast-brass">
   370→                {stats.waitingOnMe}
   371→              </p>
   372→              <p className="text-[12px] mt-2" style={{ color: "var(--text-mid)" }}>
   373→                {locale === "ar"
   374→                  ? `خطوة تنتظر إكمالك عبر ${stats.joiners} موظف`
   375→                  : `step${stats.waitingOnMe === 1 ? "" : "s"} you own across ${stats.joiners} joiner${stats.joiners === 1 ? "" : "s"}`}
   376→              </p>
   377→            </div>
   378→
   379→            {/* Joiners count */}
   380→            <div className="ast-card p-5">
   381→              <p className="text-[11px] uppercase tracking-[0.14em] font-medium" style={{ color: "var(--text-lo)" }}>
   382→                {locale === "ar" ? "المنضمّون" : "Joiners"}
   383→              </p>
   384→              <p className="text-[44px] leading-none font-semibold mt-2 tabular" style={{ color: "var(--text-hi)" }}>
   385→                {stats.joiners}
   386→              </p>
   387→              <p className="text-[12px] mt-2" style={{ color: "var(--text-mid)" }}>
   388→                {locale === "ar" ? "في رحلتك الحالية" : "in your current cohort"}
   389→              </p>
   390→            </div>
   391→
   392→            {/* Aggregate completion */}
   393→            <div className="ast-card p-5">
   394→              <p className="text-[11px] uppercase tracking-[0.14em] font-medium" style={{ color: "var(--text-lo)" }}>
   395→                {locale === "ar" ? "نسبة الإنجاز" : "Avg completion"}
   396→              </p>
   397→              <p className="text-[44px] leading-none font-semibold mt-2 tabular" style={{ color: "var(--text-hi)" }}>
   398→                {stats.avgCompletion}<span className="text-[24px] font-normal" style={{ color: "var(--text-mid)" }}>%</span>
   399→              </p>
   400→              <div className="mt-3 h-1.5 rounded-full overflow-hidden" style={{ background: "var(--surface-3)" }}>
   401→                <motion.div
   402→                  initial={false}
   403→                  animate={{ width: `${stats.avgCompletion}%` }}
   404→                  transition={{ duration: 0.4 }}
   405→                  className="h-full rounded-full"
   406→                  style={{ background: "var(--brand-500)" }}
   407→                />
   408→              </div>
   409→            </div>
   410→          </section>
   411→
   412→          {/* Cohort / Joiners view */}
   413→          {view === "cohort" && (
   414→            <section>
   415→              <div className="flex items-center gap-2 mb-4">
   416→                <Users className="w-4 h-4" strokeWidth={1.75} style={{ color: "var(--brand-400)" }} />
   417→                <h2 className="text-[17px] font-semibold" style={{ color: "var(--text-hi)" }}>
   418→                  {locale === "ar" ? "المنضمّون لديك" : "My joiners"}
   419→                </h2>
   420→              </div>
   421→              <ul className="space-y-3">
   422→                {items.map((item) => {

   424→                  return (

   430→                      <div className="flex items-start gap-4 flex-wrap">
   431→                        <div className="flex-1 min-w-0">
   432→                          <p className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>

   435→                          <p className="text-[12px] mt-0.5" style={{ color: "var(--text-mid)" }}>
   436→                            {item.journeyType === "onboarding" ? (locale === "ar" ? "التحاق" : "Onboarding") : (locale === "ar" ? "تنقل" : "Mobility")}
   437→                            <span className="mx-1.5">·</span>

   948→function CompareColumn({
   949→  item,
   950→  locale,
   951→  index,
   952→}: {
   953→  item: CohortItem;
   954→  locale: "en" | "ar";
   955→  index: number;
   956→}) {
   957→  const waitingCount = item.ownedSteps.filter((s) => s.status === "waiting_owner").length;
   958→  const hasBlocked = item.ownedSteps.some((s) => s.status === "blocked");
   959→
   960→  // Build the 5-segment stage rail. Onboarding has 5 stages, mobility 4 —
   961→  // we always render 5 segments and pad with a "—" placeholder for mobility.
   962→  const journeyStages: string[] =
   963→    item.journeyType === "onboarding"
   964→      ? ["pre_boarding", "visa_processing", "first_day", "first_week", "first_90"]
   965→      : ["transfer_approved", "handover", "effective_date", "settle_new_role", "—"];
   966→  const currentIdx = Math.max(0, journeyStages.indexOf(item.stage));
   967→
   968→  return (
   969→    <motion.article
   970→      initial={{ opacity: 0, y: 6 }}
