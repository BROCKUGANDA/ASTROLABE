   144→   144→export const CALENDARS: Record<string, WorkingCalendar> = {
   145→   145→  DXB: { weekendDays: [6, 5], holidays: ["2024-12-02", "2024-12-03"] }, // Fri/Sat in UAE
   146→   146→  AUH: { weekendDays: [6, 5], holidays: ["2024-12-02", "2024-12-03"] },

   150→   150→/**
   151→   151→ * Add hours but skip non-working time (weekends + holidays).
   152→   152→ * For the prototype we approximate: if target_hours < 24 we use wall clock.
   153→   153→ * For longer SLAs we count only working hours per day (8-18, Mon-Fri in UAE).
   154→   154→ */
   155→   155→export function computeDueAt(startIso: string, targetHours: number, location: string): string {
   156→   156→  const start = new Date(startIso);
   157→   157→  // Short SLAs (<24h) — use wall clock from start
   158→   158→  if (targetHours <= 24) {
   159→   159→    return new Date(start.getTime() + targetHours * 3600 * 1000).toISOString();
   160→   160→  }
   161→   161→  // Longer SLAs — calendar working days
   162→   162→  const cal = CALENDARS[location] ?? CALENDARS.DXB;
   163→   163→  let remainingMs = targetHours * 3600 * 1000;
   164→   164→  let cursor = new Date(start);
   165→   165→  // For prototype: assume 8 working hours per working day
   166→   166→  const workHoursPerDay = 8;
   167→   167→  const safetyMax = 365 * 24; // days cap
   168→   168→  let iter = 0;
   169→   169→  while (remainingMs > 0 && iter < safetyMax) {
   170→   170→    iter++;
   171→   171→    const dayOfWeek = cursor.getUTCDay();
   172→   172→    const isoDay = cursor.toISOString().slice(0, 10);
   173→   173→    const isWeekend = cal.weekendDays.includes(dayOfWeek);
   174→   174→    const isHoliday = cal.holidays.includes(isoDay);
   175→   175→    if (isWeekend || isHoliday) {
   176→   176→      // skip full day
   177→   177→      cursor = new Date(cursor.getTime() + 24 * 3600 * 1000);
   178→   178→      continue;
   179→   179→    }
   180→   180→    const dayWorkMs = workHoursPerDay * 3600 * 1000;
   181→   181→    if (remainingMs <= dayWorkMs) {
   182→   182→      cursor = new Date(cursor.getTime() + remainingMs);
   183→   183→      remainingMs = 0;
   184→   184→    } else {
   185→   185→      remainingMs -= dayWorkMs;
   186→   186→      cursor = new Date(cursor.getTime() + 24 * 3600 * 1000);
   187→   187→    }
   188→   188→  }
   189→   189→  return cursor.toISOString();
   190→   190→}

   192→   192→/**
   193→   193→ * Breach level from elapsed/total ratio.
   194→   194→ * Clock pauses while waiting_employee (handled by caller).
   195→   195→ */
   196→   196→export function breachLevel(elapsedHours: number, targetHours: number): StepState["breachLevel"] {
   197→   197→  if (targetHours === 0) return "none";
   198→   198→  const ratio = elapsedHours / targetHours;
   199→   199→  if (ratio >= 2.0) return "hard";
   200→   200→  if (ratio >= 1.0) return "breached";
   201→   201→  if (ratio >= 0.8) return "due_soon";
   202→   202→  return "none";
   203→   203→}

   208→   208→export function computeStepState(
   209→   209→  employee: Employee,
   210→   210→  step: JourneyStep,
   211→   211→  employeeEvents: JourneyEvent[],
   212→   212→  asOf: Date
   213→   213→): StepState {
   214→   214→  // Filter events for this step at or before asOf
   215→   215→  const stepEvents = employeeEvents
   216→   216→    .filter((e) => e.stepKey === step.key && new Date(e.occurredAt) <= asOf)
   217→   217→    .sort((a, b) => a.occurredAt.localeCompare(b.occurredAt));
   218→   218→
   219→   219→  let status: StepStatus = "not_started";
   220→   220→  let startedAt: string | undefined;
   221→   221→  let completedAt: string | undefined;
   222→   222→  let isWaitingEmployee = false;
   223→   223→
   224→   224→  for (const ev of stepEvents) {
   225→   225→    if (ev.type === "completed") {
   226→   226→      status = "done";
   227→   227→      completedAt = ev.occurredAt;
   228→   228→    } else if (ev.type === "started") {
   229→   229→      if (status !== "done") {
   230→   230→        status = "in_progress";
   231→   231→        startedAt = ev.occurredAt;
   232→   232→      }
   233→   233→    } else if (ev.type === "waiting_employee") {
   234→   234→      status = "waiting_employee";
   235→   235→      isWaitingEmployee = true;
   236→   236→      if (!startedAt) startedAt = ev.occurredAt;
   237→   237→    } else if (ev.type === "waiting_owner") {
   238→   238→      if (status !== "done") status = "waiting_owner";
   239→   239→      if (!startedAt) startedAt = ev.occurredAt;
   240→   240→    } else if (ev.type === "blocked") {
   241→   241→      if (status !== "done") status = "blocked";
   242→   242→    } else if (ev.type === "skipped") {
   243→   243→      status = "skipped";
   244→   244→    }
   245→   245→  }
   246→   246→
   247→   247→  // If dependencies not done → blocked
   248→   248→  const blockedBy: string[] = [];
   249→   249→  if (status !== "done" && status !== "skipped") {
   250→   250→    for (const depKey of step.dependsOn) {
   251→   251→      const depStep = ALL_STEPS.find((s) => s.key === depKey);
   252→   252→      if (!depStep) continue;
   253→   253→      const depState = computeStepState(employee, depStep, employeeEvents, asOf);
   254→   254→      if (depState.status !== "done" && depState.status !== "skipped") {
   255→   255→        blockedBy.push(depKey);
   256→   256→      }
   257→   257→    }
   258→   258→    if (blockedBy.length > 0 && status === "not_started") {
   259→   259→      status = "blocked";
   260→   260→    }
   261→   261→  }
   262→   262→
   263→   263→  // Compute SLA
   264→   264→  const dueAt = startedAt ? computeDueAt(startedAt, step.targetHours, employee.location) : undefined;
   265→   265→  const elapsedHours = startedAt && status !== "done"
   266→   266→    ? (asOf.getTime() - new Date(startedAt).getTime()) / 3600000
   267→   267→    : 0;
   268→   268→  // Pause clock while waiting_employee
   269→   269→  const effectiveElapsed = isWaitingEmployee ? 0 : elapsedHours;
   270→   270→  const remainingHours = dueAt && status !== "done"
   271→   271→    ? Math.max(0, (new Date(dueAt).getTime() - asOf.getTime()) / 3600000)
   272→   272→    : 0;
   273→   273→
   274→   274→  const bl = (status === "done" || status === "skipped" || !startedAt)
   275→   275→    ? "none"
   276→   276→    : breachLevel(effectiveElapsed, step.targetHours);
   277→   277→
   278→   278→  // Compute downstream-at-risk count
   279→   279→  const downstreamAtRisk = computeDownstreamAtRisk(step.key);
   280→   280→
   281→   281→  return {
   282→   282→    stepKey: step.key,
   283→   283→    status,
   284→   284→    startedAt,
   285→   285→    dueAt,
   286→   286→    completedAt,
   287→   287→    blockedBy,
   288→   288→    breachLevel: bl,
   289→   289→    targetHours: step.targetHours,
   290→   290→    elapsedHours: effectiveElapsed,
   291→   291→    remainingHours,
   292→   292→    ownerGroup: step.ownerGroup,
   293→   293→    actor: step.actor,
   294→   294→    policyRefs: step.policyRefs,
   295→   295→    dependsOn: step.dependsOn,
   296→   296→    stage: step.stage,
   297→   297→    label: step.label,
   298→   298→    labelAr: step.labelAr,
   299→   299→    requestType: step.requestType,
   300→   300→    journeyType: step.journeyType,
   301→   301→    downstreamAtRisk,
   302→   302→  };
   303→   303→}
   304→   304→
   305→   305→function computeDownstreamAtRisk(stepKey: string): number {
   306→   306→  // Count steps that depend (transitively) on this step
   307→   307→  const visited = new Set<string>();
   308→   308→  const queue = [stepKey];
   309→   309→  let count = 0;
   310→   310→  while (queue.length > 0) {
   311→   311→    const cur = queue.shift()!;
   312→   312→    for (const s of ALL_STEPS) {
   313→   313→      if (s.dependsOn.includes(cur) && !visited.has(s.key)) {
   314→   314→        visited.add(s.key);
   315→   315→        count++;
   316→   316→        queue.push(s.key);
   317→   317→      }
   318→   318→    }
   319→   319→  }
   320→   320→  return count;
   321→   321→}

   326→   326→export function inferStage(
   327→   327→  employee: Employee,
   328→   328→  employeeEvents: JourneyEvent[],
   329→   329→  asOf: Date
   330→   330→): { stage: string; dayInStage: number } {
   331→   331→  const journeyStages = ALL_STAGES
   332→   332→    .filter((s) => s.journeyType === employee.journeyType)
   333→   333→    .sort((a, b) => a.order - b.order);
   334→   334→
   335→   335→  // Compute step states for all steps
   336→   336→  const journeySteps = ALL_STEPS.filter((s) => s.journeyType === employee.journeyType);
   337→   337→  const stepStates = journeySteps.map((s) => computeStepState(employee, s, employeeEvents, asOf));
   338→   338→
   339→   339→  // Stage = latest stage whose entry condition is satisfied and exit is not
   340→   340→  // For prototype: a stage is "entered" when its first step is started or completed.
   341→   341→  // A stage is "exited" when all its steps are done.
   342→   342→  let currentStage = journeyStages[0];
   343→   343→  for (const st of journeyStages) {
   344→   344→    const stSteps = stepStates.filter((s) => s.stage === st.key);
   345→   345→    if (stSteps.length === 0) continue;
   346→   346→    const anyStarted = stSteps.some((s) => s.status === "in_progress" || s.status === "done" || s.status === "waiting_employee" || s.status === "waiting_owner" || s.status === "blocked");
   347→   347→    const allDone = stSteps.every((s) => s.status === "done" || s.status === "skipped");
   348→   348→    if (anyStarted && !allDone) {
   349→   349→      currentStage = st;
   350→   350→      break; // conservative: first non-exited stage wins
   351→   351→    }
   352→   352→    if (allDone) {
   353→   353→      currentStage = st; // last fully-done stage; we'll fall through to next
   354→   354→    }
   355→   355→  }
   356→   356→
   357→   357→  // Day in stage: find first event in this stage
   358→   358→  const stageStepKeys = journeySteps.filter((s) => s.stage === currentStage.key).map((s) => s.key);
   359→   359→  const firstEventInStage = employeeEvents
   360→   360→    .filter((e) => stageStepKeys.includes(e.stepKey) && new Date(e.occurredAt) <= asOf)
   361→   361→    .sort((a, b) => a.occurredAt.localeCompare(b.occurredAt))[0];
   362→   362→  const dayInStage = firstEventInStage
   363→   363→    ? Math.max(1, Math.floor((asOf.getTime() - new Date(firstEventInStage.occurredAt).getTime()) / (24 * 3600 * 1000)) + 1)
   364→   364→    : 1;
   365→   365→
   366→   366→  return { stage: currentStage.key, dayInStage };
   367→   367→}

   372→   372→export function computeSignals(
   373→   373→  employee: Employee,
   374→   374→  stepStates: StepState[],
   375→   375→  employeeRequests: ServiceRequest[],
   376→   376→  employeePulse: PulseResponse[],
   377→   377→  asOf: Date
   378→   378→): { fired: FrictionSignal[]; suppressed: SignalFamily[] } {
   379→   379→  const fired: FrictionSignal[] = [];
   380→   380→  const suppressed: SignalFamily[] = [];
   381→   381→
   382→   382→  // Consent gates
   383→   383→  const pulseConsent = getConsent(employee.id, "pulse_capture");
   384→   384→  if (!pulseConsent) suppressed.push("PS");
   385→   385→
   386→   386→  // ---- TL family (timing) ----
   387→   387→  let tlFamilyTotal = 0;
   388→   388→  for (const s of stepStates) {
   389→   389→    if (s.status === "done" || s.status === "skipped" || s.status === "not_started") continue;
   390→   390→    // TL-01: step past target time (3 per step, family-capped)
   391→   391→    if (s.breachLevel === "breached" || s.breachLevel === "hard") {
   392→   392→      fired.push({
   393→   393→        code: "TL-01",
   394→   394→        family: "TL",
   395→   395→        weight: 3,
   396→   396→        evidenceRefs: [`step:${s.stepKey}`, `due:${s.dueAt}`, `elapsed:${s.elapsedHours.toFixed(1)}h`],
   397→   397→        description: `${s.label} is past its target time (${s.elapsedHours.toFixed(0)}h vs ${s.targetHours}h target).`,
   398→   398→        descriptionAr: `${s.labelAr} متأخر عن الوقت المستهدف.`,
   399→   399→      });
   400→   400→      tlFamilyTotal += 3;
   401→   401→    }
   402→   402→    // TL-02: step past 200% of target (hard breach, weight 6)
   403→   403→    if (s.breachLevel === "hard") {
   404→   404→      fired.push({
   405→   405→        code: "TL-02",
   406→   406→        family: "TL",
   407→   407→        weight: 6,
   408→   408→        evidenceRefs: [`step:${s.stepKey}`, `ratio:${(s.elapsedHours / s.targetHours).toFixed(2)}`],
   409→   409→        description: `${s.label} is past 200% of target time — a hard breach.`,
   410→   410→        descriptionAr: `${s.labelAr} تجاوز 200% من الوقت المستهدف — خرق جوهري.`,
   411→   411→      });
   412→   412→      tlFamilyTotal += 6;
   413→   413→    }
   414→   414→    // TL-05: No event for 72h while stage is active (weight 3)
   415→   415→    if (s.status === "in_progress" && s.startedAt) {
   416→   416→      const sinceStart = (asOf.getTime() - new Date(s.startedAt).getTime()) / 3600000;
   417→   417→      // Approximate: if no other event in 72h on any step in same stage
   418→   418→      // For prototype we check this step's last event
   419→   419→      if (sinceStart > 72) {
   420→   420→        // Check no other event for this step in last 72h
   421→   421→        // (simplified for prototype)
   422→   422→      }
   423→   423→    }
   424→   424→  }
   425→   425→
   426→   426→  // TL-04: Start date in ≤5 days and pre-boarding completion < 60% (weight 4)
   427→   427→  const daysToStart = (new Date(employee.startDate).getTime() - asOf.getTime()) / (24 * 3600 * 1000);
   428→   428→  if (daysToStart >= 0 && daysToStart <= 5 && employee.journeyType === "onboarding") {
   429→   429→    const preSteps = stepStates.filter((s) => s.stage === "pre_boarding");
   430→   430→    const doneCount = preSteps.filter((s) => s.status === "done").length;
   431→   431→    const pct = preSteps.length > 0 ? doneCount / preSteps.length : 0;
   432→   432→    if (pct < 0.6) {
   433→   433→      fired.push({
   434→   434→        code: "TL-04",
   435→   435→        family: "TL",
   436→   436→        weight: 4,
   437→   437→        evidenceRefs: [`startDate:${employee.startDate}`, `preBoardingPct:${pct.toFixed(2)}`],
   438→   438→        description: `Start date is in ${Math.ceil(daysToStart)} day(s) but pre-boarding is only ${Math.round(pct * 100)}% complete.`,
   439→   439→        descriptionAr: `تاريخ البدء بعد ${Math.ceil(daysToStart)} يوم/أيام لكن ما قبل الالتحاق مكتمل بنسبة ${Math.round(pct * 100)}% فقط.`,
   440→   440→      });
   441→   441→    }
   442→   442→  }
   443→   443→
   444→   444→  // ---- DP family (dependencies) ----
   445→   445→  for (const s of stepStates) {
   446→   446→    if (s.status === "done" || s.status === "skipped") continue;
   447→   447→    // DP-01: Incomplete step blocks ≥2 downstream steps (weight 5)
   448→   448→    if (s.downstreamAtRisk >= 2 && (s.status === "in_progress" || s.status === "blocked")) {
   449→   449→      fired.push({
   450→   450→        code: "DP-01",
   451→   451→        family: "DP",
   452→   452→        weight: 5,
   453→   453→        evidenceRefs: [`step:${s.stepKey}`, `downstream:${s.downstreamAtRisk}`],

   456→   456→      });
   457→   457→    }
   458→   458→    // DP-02: Visa / work-authorisation step stalled > 7 days (hard trip, weight 7)
   459→   459→    if (
   460→   460→      (s.stepKey === "ob_visa_application" || s.stepKey === "ob_visa_approval") &&
   461→   461→      s.status === "in_progress" &&
   462→   462→      s.startedAt
   463→   463→    ) {
   464→   464→      const stalled = (asOf.getTime() - new Date(s.startedAt).getTime()) / (24 * 3600 * 1000);

   476→   476→      }
   477→   477→    }
   478→   478→    // DP-03: Access provisioning blocked by missing ID or asset (weight 4)

   480→   480→      fired.push({

   488→   488→    }
   489→   489→    // DP-05: Transfer approved, releasing-manager handover absent > 5 days (weight 4)
   490→   490→    if (s.stepKey === "mb_release_handover" && s.status === "in_progress" && s.startedAt) {
   491→   491→      const stalled = (asOf.getTime() - new Date(s.startedAt).getTime()) / (24 * 3600 * 1000);
   492→   492→      if (stalled > 5) {
   493→   493→        fired.push({
   494→   494→          code: "DP-05",
   495→   495→          family: "DP",
   496→   496→          weight: 4,

   523→   523→        family: "RQ",
   524→   524→        weight: 4,

   534→   534→        family: "RQ",
   535→   535→        weight: 4,
   536→   536→        evidenceRefs: [`request:${r.ref}`, `returnCount:${r.returnCount}`],

   541→   541→  }

   544→   544→  if (pulseConsent && suppressed.indexOf("PS") === -1) {
   545→   545→    for (const p of employeePulse) {

   561→   561→          code: "PS-05",
   562→   562→          family: "PS",
   563→   563→          weight: 6,
   564→   564→          hardTrip: true,
   565→   565→          forceBlocked: false,

   569→   569→        });
   570→   570→      }

   572→   572→  } else {

   574→   574→    if (suppressed.indexOf("PS") !== -1) {
   575→   575→      // Do not add a signal; suppression is disclosed separately
   576→   576→    }
   577→   577→  }
   578→   578→
   579→   579→  // ---- MB family (mobility) ----
   580→   580→  if (employee.journeyType === "mobility" && employee.effectiveDate) {
   581→   581→    const daysSinceEffective = (asOf.getTime() - new Date(employee.effectiveDate).getTime()) / (24 * 3600 * 1000);

   586→   586→        code: "MB-01",
   587→   587→        family: "MB",
   588→   588→        weight: 4,

   603→   603→        fired.push({
   604→   604→          code: "MB-04",
   605→   605→          family: "MB",
   606→   606→          weight: 6,
   607→   607→          hardTrip: true,
   608→   608→          forceBlocked: true,
   609→   609→          evidenceRefs: [`step:mb_payroll_change`, `daysToCutoff:${daysToCutoff.toFixed(0)}`],
   610→   610→          description: `Payroll change unconfirmed and cycle cutoff is in ${daysToCutoff.toFixed(0)} days — hard trip.`,
   611→   611→          descriptionAr: `تغيير الراتب غير مؤكد وتقطع دورة الرواتب بعد ${daysToCutoff.toFixed(0)} أيام — تعطيل جوهري.`,
   612→   612→        });
   613→   613→      }
   614→   614→    }
   615→   615→    // MB-03: Dual reporting unresolved > 7 days after effective date (weight 3)
   616→   616→    const dualStep = stepStates.find((s) => s.stepKey === "mb_dual_reporting");
   617→   617→    if (dualStep && dualStep.status === "waiting_owner" && daysSinceEffective > 7) {

   619→   619→        code: "MB-03",
   620→   620→        family: "MB",
   621→   621→        weight: 3,
   622→   622→        evidenceRefs: [`step:mb_dual_reporting`, `daysSinceEffective:${daysSinceEffective.toFixed(0)}`],
   623→   623→        description: `Dual reporting unresolved for ${daysSinceEffective.toFixed(0)} days after effective date.`,
   624→   624→        descriptionAr: `التقارير المزدوجة غير محلولة منذ ${daysSinceEffective.toFixed(0)} أيام من تاريخ السريان.`,
   625→   625→      });
   626→   626→    }
   627→   627→  }
   628→   628→
   629→   629→  // ---- DQ family (data quality) ----
   630→   630→  // DQ-01: Manager or HRBP mapping missing (weight 4)
   631→   631→  if (!employee.managerId || !employee.hrbpId) {
   632→   632→    fired.push({
   633→   633→      code: "DQ-01",
   634→   634→      family: "DQ",
   635→   635→      weight: 4,
   636→   636→      evidenceRefs: [`employee:${employee.id}`],
   637→   637→      description: `Manager or HRBP mapping missing on employee record.`,
   638→   638→      descriptionAr: `تعيين المدير أو شريك الموارد البشرية مفقود في سجل الموظف.`,
   639→   639→    });
   640→   640→  }
   641→   641→  // DQ-05: Step references a policy absent from the corpus
   642→   642→  // (simplified — for prototype assume all policy refs resolve)
   643→   643→
   644→   644→  return { fired, suppressed };
   645→   645→}

   647→   647→/* ============================================================
   648→   648→   FUSION — bands from capped signal weights + hard trips
   649→   649→   ============================================================ */
   650→   650→export function fuse(signals: FrictionSignal[], suppressed: SignalFamily[]): {
   651→   651→  score: number;
   652→   652→  band: Band;
   653→   653→  hardTrips: string[];
   654→   654→  forcedBlocked: boolean;
   655→   655→} {
   656→   656→  // Apply family caps
   657→   657→  const familyTotals: Record<string, number> = {};
   658→   658→  const hardTrips: string[] = [];
   659→   659→  let forcedBlocked = false;
   660→   660→
   661→   661→  for (const sig of signals) {
   662→   662→    const fam = sig.family;
   663→   663→    const capped = Math.min(sig.weight, FAMILY_CAPS[fam]);
   664→   664→    familyTotals[fam] = (familyTotals[fam] ?? 0) + capped;
   665→   665→    if (sig.hardTrip) hardTrips.push(sig.code);
   666→   666→    if (sig.forceBlocked) forcedBlocked = true;
   667→   667→  }
   668→   668→
   669→   669→  let score = 0;
   670→   670→  for (const [fam, total] of Object.entries(familyTotals)) {
   671→   671→    score += Math.min(total, FAMILY_CAPS[fam as SignalFamily]);
   672→   672→  }
   673→   673→
   674→   674→  let band: Band = "steady";
   675→   675→  if (score >= 20) band = "blocked";
   676→   676→  else if (score >= 12) band = "friction";
   677→   677→  else if (score >= 6) band = "watch";
   678→   678→
   679→   679→  // Hard trips: any weight≥6 forces at least Friction
   680→   680→  if (hardTrips.length > 0 && band === "steady") band = "watch";
   681→   681→  if (hardTrips.length > 0 && (band === "steady" || band === "watch")) band = "friction";
   682→   682→  // DP-02, PS-05, MB-04 force Blocked
   683→   683→  if (hardTrips.some((c) => c === "DP-02" || c === "PS-05" || c === "MB-04") || forcedBlocked) {
   684→   684→    band = "blocked";
   685→   685→  }
   686→   686→
   687→   687→  return { score, band, hardTrips, forcedBlocked };
   688→   688→}

   693→   693→export function assess(
   694→   694→  employee: Employee,
   695→   695→  asOf: Date = new Date()
   696→   696→): Assessment {
   697→   697→  const employeeEvents = eventsForEmployee(employee.id);
   698→   698→  const employeeRequests = ALL_REQUESTS.filter((r) => r.employeeId === employee.id);
   699→   699→  const employeePulse = ALL_PULSE.filter((p) => p.employeeId === employee.id);
   700→   700→
   701→   701→  const journeySteps = ALL_STEPS.filter((s) => s.journeyType === employee.journeyType);
   702→   702→  const stepStates = journeySteps.map((s) => computeStepState(employee, s, employeeEvents, asOf));
   703→   703→
   704→   704→  const { fired, suppressed } = computeSignals(employee, stepStates, employeeRequests, employeePulse, asOf);
   705→   705→  const fusion = fuse(fired, suppressed);
   706→   706→
   707→   707→  const { stage, dayInStage } = inferStage(employee, employeeEvents, asOf);
   708→   708→
   709→   709→  return {
   710→   710→    employeeId: employee.id,
   711→   711→    asOf: asOf.toISOString(),
   712→   712→    band: fusion.band,
   713→   713→    score: fusion.score,
   714→   714→    signals: fired,
   715→   715→    suppressedFamilies: suppressed,
   716→   716→    mapVersion: "map-1.0.0",
   717→   717→    corpusVersion: "policies-2024-q4-v1",
   718→   718→    engineVersion: ENGINE_VERSION,
   719→   719→    stage,
   720→   720→    dayInStage,
   721→   721→  };
   722→   722→}

   731→   731→/* ============================================================
   732→   732→   NEXT BEST STEP — ranked
   733→   733→   ============================================================ */
   734→   734→export function nextBestStep(employee: Employee, asOf: Date = new Date()): NextBestStep | null {
   735→   735→  const employeeEvents = eventsForEmployee(employee.id);
   736→   736→  const journeySteps = ALL_STEPS.filter((s) => s.journeyType === employee.journeyType);
   737→   737→  const stepStates = journeySteps.map((s) => computeStepState(employee, s, employeeEvents, asOf));
   738→   738→
   739→   739→  // Candidates: not done, dependencies satisfied, in current or next stage
   740→   740→  const { stage } = inferStage(employee, employeeEvents, asOf);
   741→   741→  const journeyStages = ALL_STAGES.filter((s) => s.journeyType === employee.journeyType).sort((a, b) => a.order - b.order);
   742→   742→  const stageIdx = journeyStages.findIndex((s) => s.key === stage);
   743→   743→  const allowedStages = journeyStages.slice(stageIdx, stageIdx + 2).map((s) => s.key);
   744→   744→
   745→   745→  const candidates = stepStates.filter((s) => {
   746→   746→    if (s.status === "done" || s.status === "skipped") return false;
   747→   747→    if (!allowedStages.includes(s.stage)) return false;
   748→   748→    if (s.status === "blocked") return false;
   749→   749→    return true;
   750→   750→  });
   751→   751→
   752→   752→  if (candidates.length === 0) return null;
   753→   753→
   754→   754→  // Rank: hard breach > most downstream blocked > nearest due > employee-actionable > least effort
   755→   755→  candidates.sort((a, b) => {
   756→   756→    const aHard = a.breachLevel === "hard" ? 1 : 0;
   757→   757→    const bHard = b.breachLevel === "hard" ? 1 : 0;
   758→   758→    if (aHard !== bHard) return bHard - aHard;
   759→   759→    if (b.downstreamAtRisk !== a.downstreamAtRisk) return b.downstreamAtRisk - a.downstreamAtRisk;
   760→   760→    const aDue = a.dueAt ? new Date(a.dueAt).getTime() : Infinity;
   761→   761→    const bDue = b.dueAt ? new Date(b.dueAt).getTime() : Infinity;
   762→   762→    if (aDue !== bDue) return aDue - bDue;
   763→   763→    const aEmp = a.actor === "employee" ? 0 : 1;
   764→   764→    const bEmp = b.actor === "employee" ? 0 : 1;
   765→   765→    if (aEmp !== bEmp) return aEmp - bEmp;
   766→   766→    return a.targetHours - b.targetHours;
   767→   767→  });
   768→   768→
   769→   769→  const top = candidates[0];
   770→   770→  const step = journeySteps.find((s) => s.key === top.stepKey)!;
   771→   771→  let reason = `This unblocks ${top.downstreamAtRisk} step(s) and is the most urgent item.`;
   772→   772→  let reasonAr = `هذا يحرر ${top.downstreamAtRisk} خطوة/خطوات وهو العنصر الأكثر إلحاحاً.`;
   773→   773→  if (top.breachLevel === "hard") {
   774→   774→    reason = `${step.label} is past 200% of its target — this is the most urgent item.`;
   775→   775→    reasonAr = `${step.labelAr} تجاوز 200% من المستهدف — هذا هو العنصر الأكثر إلحاحاً.`;
   776→   776→  } else if (top.actor === "employee") {
   777→   777→    reason = `${step.label} is yours to do — and it unblocks the next ${top.downstreamAtRisk} step(s).`;
   778→   778→    reasonAr = `${step.labelAr} خطوتك للقيام بها — وتحرر ${top.downstreamAtRisk} خطوة/خطوات لاحقة.`;
   779→   779→  }
   780→   780→
   781→   781→  return {
   782→   782→    stepKey: top.stepKey,
   783→   783→    label: step.label,
   784→   784→    labelAr: step.labelAr,
   785→   785→    ownerGroup: step.ownerGroup,
   786→   786→    dueAt: top.dueAt,
   787→   787→    policyRefs: step.policyRefs,
   788→   788→    actor: step.actor,
   789→   789→    reason,
   790→   790→    reasonAr,
   791→   791→    requestType: step.requestType,
   792→   792→    primary: true,
   793→   793→  };
   794→   794→}

   796→   796→/* ============================================================
   797→   797→   ENGINE TRACE — judge mode visibility
   798→   798→   ============================================================ */
   799→   799→export function trace(employee: Employee, asOf: Date = new Date()): EngineTrace {
   800→   800→  const assessment = assess(employee, asOf);
   801→   801→  const employeeEvents = eventsForEmployee(employee.id);
   802→   802→  const journeySteps = ALL_STEPS.filter((s) => s.journeyType === employee.journeyType);
   803→   803→  const stepStates = journeySteps.map((s) => computeStepState(employee, s, employeeEvents, asOf));
   804→   804→  const employeeRequests = ALL_REQUESTS.filter((r) => r.employeeId === employee.id);
   805→   805→  const employeePulse = ALL_PULSE.filter((p) => p.employeeId === employee.id);
   806→   806→  const { fired, suppressed } = computeSignals(employee, stepStates, employeeRequests, employeePulse, asOf);
   807→   807→  const fusion = fuse(fired, suppressed);
   808→   808→  const nbs = nextBestStep(employee, asOf);
   809→   809→
   810→   810→  return {
   811→   811→    stageInference: {
   812→   812→      stage: assessment.stage,
   813→   813→      reason: `Latest stage with started activity and not all steps done (conservative tie-break).`,
   814→   814→    },
   815→   815→    signalsFired: fired.map((s) => ({
   816→   816→      code: s.code,
   817→   817→      weight: s.weight,
   818→   818→      family: s.family,
   819→   819→      evidence: s.evidenceRefs,
   820→   820→    })),
   821→   821→    signalsSuppressed: suppressed.map((family) => ({
   822→   822→      family,
   823→   823→      reason: family === "PS" ? "Pulse consent not granted — family suppressed, contributes 0." : "Unknown",
   824→   824→    })),
   825→   825→    band: {
   826→   826→      score: fusion.score,
   827→   827→      band: fusion.band,
   828→   828→      hardTrips: fusion.hardTrips,
   829→   829→      forcedBlocked: fusion.forcedBlocked,
