   111→/** Escape regex metachars in a term so we can build a literal matcher. */
   112→function escapeRegex(s: string): string {
   113→  return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   114→}
   115→
   116→/**
   117→ * Split a sentence into segments so matched terms render with a teal
   118→ * highlight (0.3 opacity background) and the rest renders as plain text.
   119→ * Match is case-insensitive but preserves the original text casing.
   120→ */
   121→function highlightMatches(text: string, terms: string[]): React.ReactNode {
   122→  if (terms.length === 0) return text;
   123→  const sorted = [...terms].sort((a, b) => b.length - a.length);
   124→  const pattern = new RegExp(`(${sorted.map(escapeRegex).join("|")})`, "gi");
   125→  const parts = text.split(pattern);
   126→  return parts.map((part, i) => {
   127→    if (i % 2 === 1) {
   128→      return (
   129→        <mark
   130→          key={i}
   131→          className="rounded-[3px] px-0.5"
   132→          style={{
   133→            background: "rgba(22,154,165,0.30)",
   134→            color: "inherit",
   135→            borderBottom: "1px solid rgba(22,154,165,0.50)",
   136→          }}
   137→        >
   138→          {part}
   139→        </mark>
   140→      );
   141→    }
   142→    return <span key={i}>{part}</span>;
   143→  });
   144→}

   220→
   221→  // Sync incoming initialQuery when the modal opens with a prefilled query.
   222→  useEffect(() => {
   223→    if (initialQuery !== undefined) {
   224→      setQuery(initialQuery);
   225→    }

   227→
   228→  const docSlug = useMemo(() => {
   229→    const f = DOC_FILTERS.find((d) => d.id === docFilter);
   230→    return f?.slug ?? null;
   231→  }, [docFilter]);
   232→
   233→  const runSearch = useCallback(async () => {
   234→    const q = query.trim();
   235→    if (q.length < 2) {
   236→      setHits(null);
   237→      setError(null);
   238→      return;
   239→    }
   240→    setLoading(true);
   241→    setError(null);

   244→        q,

   249→      const json: ApiEnvelope<{ items: PolicyHit[]; total: number }> = await res.json();
   250→      if (!json.ok || !json.data) {
   251→        throw new Error(json.error?.message ?? "Search failed");
   252→      }
   253→      setHits(json.data.items);
   254→    } catch (err) {
   255→      setError(err instanceof Error ? err.message : "Network error");
   256→      setHits([]);
   257→    } finally {
   258→      setLoading(false);
   259→    }
   260→  }, [query, langFilter, docSlug]);
   261→

   263→  useEffect(() => {
   264→    const t = setTimeout(() => {
   265→      runSearch();
   266→    }, 320);

   281→      setError(err instanceof Error ? err.message : "Network error");
   282→    } finally {
   283→      setDocLoading(false);
   284→    }
   285→  }, []);
   286→

   288→  const showEmptyState = hasQuery && !loading && !error && hits !== null && hits.length === 0;

   297→          border: "1px solid var(--border)",
   298→        }}

   300→        <Search className="w-4 h-4 shrink-0" strokeWidth={1.75} style={{ color: "var(--brand-400)" }} />
   301→        <input
   302→          type="text"
   303→          value={query}
   304→          onChange={(e) => setQuery(e.target.value)}
   305→          autoFocus
   306→          aria-label={locale === "ar" ? "ابحث في السياسات" : "Search policies"}
   307→          placeholder={
   308→            locale === "ar"
   309→              ? "ابحث عن بدل، فيزا، رواتب، تدريب…"
   310→              : "Search for allowance, visa, payroll, training…"
   311→          }
   312→          className="flex-1 bg-transparent outline-none text-[14px] placeholder:opacity-50"
   313→          style={{ color: "var(--text-hi)" }}
   314→        />
   315→        {query && (
   316→          <button
   317→            type="button"

   337→            style={{ color: "var(--text-lo)" }}
