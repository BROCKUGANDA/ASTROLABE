   126→/** Strip inline [^n] refs and footnote definitions; return clean markdown. */
   127→function stripCitations(md: string): string {
   128→  return md
   129→    // Strip footnote definition lines FIRST (before removing the [^n] markers they start with)
   130→    .replace(/^\[\^\d+\]:.*$/gm, "")
   131→    // Then strip inline [^n] references
   132→    .replace(/\[\^\d+\]/g, "")
   133→    .replace(/\n{3,}/g, "\n\n")
   134→    .trim();
   135→}
   136→
   137→function fmtMs(ms: number): string {
   138→  if (ms < 1000) return `${ms}ms`;
   139→  return `${(ms / 1000).toFixed(1)}s`;
   140→}

   223→      try {

   225→          method: "POST",
   226→          headers: { "Content-Type": "application/json" },
   227→          body: JSON.stringify({ question: trimmed, locale, employeeId: currentEmployeeId }),

   231→          throw new Error(json.error?.message ?? "Failed to get answer");

   234→          prev.map((t) =>

   236→              ? { ...t, answer: json.data!.answer, loading: false }
   237→              : t,
   238→          ),
   239→        );

   242→          prev.map((t) =>
   243→            t.id === turnId
   244→              ? {
   245→                  ...t,
   246→                  loading: false,
   247→                  error: err instanceof Error ? err.message : "Unknown error",
   248→                }
   249→              : t,
   250→          ),
   251→        );
   252→        toast.error(locale === "ar" ? "تعذّر الوصول إلى السياسة" : "Couldn't reach the policy service");

   254→        // Stop animations after fetch resolves (regardless of stage reached)
   255→        stageTimers.forEach(clearTimeout);
   256→        stageIntervals.forEach(clearInterval);
   257→        setActiveStage(-1);
   258→      }
   259→    },
   260→    [currentEmployeeId, locale],
   261→  );
   262→
   263→  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   264→    if (e.key === "Enter" && !e.shiftKey) {
   265→      e.preventDefault();

   271→
   272→  return (

   274→      {/* Header */}
   275→      <div className="mb-5 flex items-center gap-3">
   276→        <div
   277→          className="flex items-center justify-center w-10 h-10 rounded-full shrink-0"
   278→          style={{ background: "var(--g-veil)", color: "var(--brand-400)" }}
   279→          aria-hidden="true"
   280→        >
   281→          <Compass className="w-5 h-5" strokeWidth={1.75} />
   282→        </div>
   283→        <div className="min-w-0">
   284→          <h1
   285→            className="text-[20px] sm:text-[22px] font-semibold leading-tight"
   286→            style={{ color: "var(--text-hi)" }}
   287→          >
   288→            {locale === "ar" ? "اسأل عن السياسة" : "Ask about policy"}
   289→          </h1>
   290→          <p className="text-[13px] mt-0.5" style={{ color: "var(--text-mid)" }}>
   291→            {locale === "ar"
   292→              ? "أسألني أي شيء عن رحلتك أو سياساتنا. أُجيب من الوثائق الرسمية فقط — وإذا لم أكن متأكداً، أحيلك إلى الشخص المناسب."
   293→              : "Ask me anything about your journey or our policies. I answer from official documents only — and if I'm not sure, I hand you to the right person."}
   294→          </p>
   295→        </div>
   296→      </div>
   297→
   298→      {/* History (last 5 turns) — scrollable */}
   299→      {turns.length > 0 && (
   300→        <section
   301→          className="ast-card p-3 sm:p-4 mb-4"
   302→          aria-label={locale === "ar" ? "آخر الأسئلة" : "Recent questions"}

   304→          <div className="flex items-center justify-between mb-2 px-1">

   462→/* ============================================================
   463→   Three-stage progress — instrument style with pulse dots
   464→   connected by hairlines.
   465→   ============================================================ */
   466→
   467→function ThreeStageProgress({
   468→  activeStage,
   469→  stageElapsed,
   470→  locale,
   471→}: {
   472→  activeStage: number;
   473→  stageElapsed: number[];
   474→  locale: "en" | "ar";
   475→}) {
   476→  const labels =
   477→    locale === "ar"
   478→      ? ["يفحص رحلتك", "يقرأ السياسة", "يصيغ الإجابة"]
   479→      : ["checking your journey", "reading policy", "drafting"];
   480→  return (
   481→    <div>
   482→      <ol className="flex items-center gap-2">
   483→        {labels.map((label, i) => {
   484→          const isPast = i < activeStage;
   485→          const isActive = i === activeStage;
   486→          const isPending = i > activeStage;
   487→          return (
   488→            <li key={label} className="flex items-center gap-2 flex-1 min-w-0">
   489→              <span
   490→                className={cn(
   491→                  "inline-flex items-center justify-center w-3 h-3 rounded-full shrink-0",
   492→                  isActive && "ast-pulse-ring",
   493→                )}
   494→                style={{
   495→                  background: isPast
   496→                    ? "var(--brand-500)"
   497→                    : isActive
   498→                      ? "var(--brand-400)"
   499→                      : "var(--surface-3)",
   500→                  border: `1px solid ${
   501→                    isPast || isActive ? "var(--brand-500)" : "var(--border)"
   502→                  }`,
   503→                }}
   504→                aria-hidden="true"
   505→              />
   506→              <div className="flex-1 min-w-0">
   507→                <p
   508→                  className="text-[12px] font-medium leading-tight truncate"
   509→                  style={{
   510→                    color: isPast
   511→                      ? "var(--brand-400)"
   512→                      : isActive
   513→                        ? "var(--text-hi)"
   514→                        : "var(--text-lo)",
   515→                  }}
   516→                >
   517→                  {label}
   518→                </p>
   519→                {(isActive || isPast) && (
   520→                  <p
   521→                    className="text-[10px] tabular mt-0.5"
   522→                    style={{ color: "var(--text-lo)" }}
   523→                  >
   524→                    {fmtMs(stageElapsed[i] || 0)}
   525→                  </p>
   526→                )}
   527→              </div>
   528→              {i < labels.length - 1 && (
   529→                <span
   530→                  aria-hidden="true"
   531→                  className="hidden sm:block h-[1px] w-8 shrink-0"
   532→                  style={{
   533→                    background:
   534→                      isPast || isActive
   535→                        ? "var(--brand-500)"
   536→                        : "var(--border)",
   537→                  }}
   538→                />
   539→              )}
   540→            </li>
   541→          );
   542→        })}
   543→      </ol>
   544→      <p
   545→        className="text-[11px] mt-3"
   546→        style={{ color: "var(--text-lo)" }}
   547→      >
   548→        {locale === "ar"
   549→          ? "أبحث في سياساتك الرسمية فقط — لا تخمين."
   550→          : "I'm searching your official policies only — no guessing."}
   551→      </p>
   552→    </div>
   553→  );
   554→}
