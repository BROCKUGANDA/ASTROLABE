   161→   161→function shortDate(iso: string, locale: "en" | "ar"): string {
   162→   162→  try {
   163→   163→    const d = new Date(iso);
   164→   164→    return format(d, "d MMM yyyy · HH:mm") + (locale === "ar" ? " ت" : " UTC");
   165→   165→  } catch {
   166→   166→    return iso;
   167→   167→  }
