   153→function namedOwnerForGroup(group: string): { name: string; nameAr?: string; role: string; group: string } | null {
   154→  const o = ALL_OWNERS.find((x) => x.group === group);
   155→  if (!o) return null;

   159→function stageLabel(stageKey: string, journeyType: string, locale: "en" | "ar"): string {
   160→  const st = ALL_STAGES.find((s) => s.key === stageKey && s.journeyType === journeyType);
   161→  if (!st) return stageKey;
   162→  return locale === "ar" ? st.labelAr : st.label;
   163→}

   165→function stageShort(stageKey: string, journeyType: string, locale: "en" | "ar"): string {
   166→  const full = stageLabel(stageKey, journeyType, locale);
   167→  // Compact rail labels: take first 1–2 words
   168→  if (locale === "ar") return full;
   169→  const parts = full.split(" ");
   170→  return parts.length > 2 ? parts.slice(0, 2).join(" ") : full;
   171→}

   173→function fmtDateAndDistance(iso: string | undefined, locale: "en" | "ar"): string {
   174→  if (!iso) return locale === "ar" ? "في أي وقت" : "any time";
   175→  const d = new Date(iso);
   176→  const datePart = format(d, "EEEE d MMM");
   177→  // Use synthetic NOW from pack for deterministic "in 3 days" copy
   178→  const distance = formatDistance(d, PACK_NOW, { addSuffix: false });
   179→  return locale === "ar" ? `${datePart} · ${distance}` : `${datePart} · ${distance}`;
   180→}
   181→
   182→function lateCopy(step: Step, locale: "en" | "ar"): string {
   183→  if (step.breachLevel === "hard") {
   184→    const days = Math.max(1, Math.floor((step.elapsedHours - step.targetHours) / 24));
   185→    return locale === "ar"
   186→      ? `متأخر بـ ${days} أيام — خرق جوهري`
   187→      : `${days} day${days === 1 ? "" : "s"} late — hard breach`;
   188→  }
   189→  if (step.breachLevel === "breached") {
   190→    const days = Math.max(1, Math.floor((step.elapsedHours - step.targetHours) / 24));
   191→    return locale === "ar" ? `متأخر بـ ${days} أيام` : `${days} day${days === 1 ? "" : "s"} late`;
   192→  }
   193→  if (step.status === "blocked") {
   194→    return locale === "ar"
   195→      ? "محظور بانتظار خطوات أخرى"
   196→      : "Held up — waiting on another step";
   197→  }
   198→  return "";
   199→}

   201→function ownerLine(step: Step, locale: "en" | "ar"): string {
   202→  const grp = OWNER_GROUP_LABEL[step.ownerGroup];
   203→  const named = namedOwnerForGroup(step.ownerGroup);
   204→  if (!grp) return step.ownerGroup;
   205→  if (named) {
   206→    return locale === "ar" && named.nameAr
   207→      ? `مع ${named.nameAr} في ${grp.ar}`
   208→      : `with ${named.name} in ${grp.en}`;
   209→  }
   210→  return locale === "ar" ? `مع ${grp.ar}` : `with ${grp.en}`;
   211→}

   218→    : `about ${days} day${days === 1 ? "" : "s"} from start`;
   219→}

   636→function StageRail({ data }: { data: JourneyMeData }) {
   637→  const { locale } = useAppStore();
   638→  const journeyType = data.employee.journeyType;
   639→  const journeyStages = useMemo(
   640→    () => ALL_STAGES.filter((s) => s.journeyType === journeyType).sort((a, b) => a.order - b.order),
   641→    [journeyType]
   642→  );
   643→
   644→  const segState = (stageKey: string): "done" | "active" | "blocked" | "pending" => {
   645→    const stepsInStage = data.steps.filter((s) => s.stage === stageKey);
   646→    if (stepsInStage.length === 0) return "pending";
   647→    const allDone = stepsInStage.every((s) => s.status === "done" || s.status === "skipped");
   648→    if (allDone) return "done";
   649→    const anyBlocked = stepsInStage.some(
   650→      (s) => s.status === "blocked" || s.breachLevel === "hard"
   651→    );
   652→    if (anyBlocked) return "blocked";
   653→    if (stageKey === data.stage) return "active";
   654→    // Stage before current — if not all done, mark done visually (history)
   655→    const stageOrder = journeyStages.find((s) => s.key === stageKey)?.order ?? 0;
   656→    const currentOrder = journeyStages.find((s) => s.key === data.stage)?.order ?? 0;
   657→    if (stageOrder < currentOrder) return "done";
   658→    return "pending";
   659→  };
   660→
   661→  return (
   662→    <section className="ast-card p-5 sm:p-6" aria-label="Stage rail">
   663→      <div className="flex items-center gap-2 mb-4">
   664→        <Compass className="w-4 h-4" strokeWidth={1.75} style={{ color: "var(--brand-400)" }} />
   665→        <h3 className="text-[14px] font-semibold uppercase tracking-[0.12em]" style={{ color: "var(--text-hi)" }}>
   666→          {locale === "ar" ? "مسار الرحلة" : "Journey rail"}
   667→        </h3>
   668→      </div>
   669→      <ol className="grid gap-2" style={{ gridTemplateColumns: `repeat(${journeyStages.length}, minmax(0, 1fr))` }}>
   670→        {journeyStages.map((st) => {
   671→          const state = segState(st.key);
   672→          const isActive = st.key === data.stage;
   673→          const label = locale === "ar" ? st.labelAr : st.label;
   674→          return (
   675→            <li key={st.key} className="flex flex-col gap-2 min-w-0">
   676→              <div
   677→                className="ast-rail-seg"
   678→                data-state={state === "pending" ? undefined : state}
   679→                aria-label={`${label}: ${state}`}
   680→                style={state === "pending" ? { background: "var(--surface-3)" } : undefined}
   681→              />
   682→              <div className="min-w-0">
   683→                <p
   684→                  className="text-[11px] sm:text-[12px] font-medium leading-tight truncate"
   685→                  style={{
   686→                    color: isActive ? "var(--accent-500)" : state === "done" ? "var(--brand-400)" : state === "blocked" ? "var(--status-blocked)" : "var(--text-lo)",
   687→                  }}
   688→                >
   689→                  {stageShort(st.key, journeyType, locale)}
   690→                </p>
   691→                {isActive && (
   692→                  <p className="text-[10px] tabular mt-0.5" style={{ color: "var(--text-lo)" }}>
   693→                    {locale === "ar" ? `اليوم ${data.dayInStage}` : `Day ${data.dayInStage}`}
   694→                  </p>
   695→                )}
   696→              </div>
   697→            </li>
   698→          );
   699→        })}

   971→function PulseCard({
   972→  employeeId,
   973→  stage,
   974→  pulseGranted,
   975→  onGrantConsent,
   976→  onSubmit,
   977→  locale,
   978→}: {
   979→  employeeId: string;
   980→  stage: string;
   981→  pulseGranted: boolean;
   982→  onGrantConsent: () => Promise<void>;
   983→  onSubmit: (score: number) => Promise<void>;
   984→  locale: "en" | "ar";
   985→}) {
   986→  const [score, setScore] = useState<number | null>(null);
   987→  const [submitting, setSubmitting] = useState(false);
   988→  const [granting, setGranting] = useState(false);
   989→  const [done, setDone] = useState(false);
   990→
   991→  const handleGrant = async () => {
   992→    setGranting(true);
   993→    try {
   994→      await onGrantConsent();
   995→    } finally {
   996→      setGranting(false);
   997→    }
   998→  };
   999→
  1000→  const handleSubmit = async () => {
  1001→    if (score === null) return;
  1002→    setSubmitting(true);
  1003→    try {
  1004→      await onSubmit(score);
  1005→      setDone(true);
  1006→      setScore(null);
  1007→    } finally {
  1008→      setSubmitting(false);
  1009→    }
  1010→  };
  1011→
  1012→  return (
  1013→    <motion.section
  1014→      initial={{ opacity: 0, y: 8 }}
  1015→      animate={{ opacity: 1, y: 0 }}
  1016→      transition={{ duration: 0.24 }}
  1017→      className="ast-card p-5 sm:p-6 relative overflow-hidden"
  1018→      aria-label="Pulse check"
  1019→    >
