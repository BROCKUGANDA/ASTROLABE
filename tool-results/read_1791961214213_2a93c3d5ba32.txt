   111→   111→/** Escape regex metachars in a term so we can build a literal matcher. */
   112→   112→function escapeRegex(s: string): string {
   113→   113→  return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   114→   114→}
   115→   115→
   116→   116→/**
   117→   117→ * Split a sentence into segments so matched terms render with a teal
   118→   118→ * highlight (0.3 opacity background) and the rest renders as plain text.
   119→   119→ * Match is case-insensitive but preserves the original text casing.
   120→   120→ */
   121→   121→function highlightMatches(text: string, terms: string[]): React.ReactNode {
   122→   122→  if (terms.length === 0) return text;
   123→   123→  const sorted = [...terms].sort((a, b) => b.length - a.length);
   124→   124→  const pattern = new RegExp(`(${sorted.map(escapeRegex).join("|")})`, "gi");
   125→   125→  const parts = text.split(pattern);
   126→   126→  return parts.map((part, i) => {
   127→   127→    if (i % 2 === 1) {
   128→   128→      return (
   129→   129→        <mark
   130→   130→          key={i}
   131→   131→          className="rounded-[3px] px-0.5"
   132→   132→          style={{
   133→   133→            background: "rgba(22,154,165,0.30)",
   134→   134→            color: "inherit",
   135→   135→            borderBottom: "1px solid rgba(22,154,165,0.50)",
   136→   136→          }}
   137→   137→        >
   138→   138→          {part}
   139→   139→        </mark>
   140→   140→      );
   141→   141→    }
   142→   142→    return <span key={i}>{part}</span>;
   143→   143→  });
   144→   144→}
   145→   145→
   146→   146→function isStale(reviewDate: string): boolean {

   148→   148→}
