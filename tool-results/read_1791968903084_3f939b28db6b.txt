   206→function truncateHash(h: string): string {
   207→  // "0x" + first 8 chars total + "…" + last 4 chars.
   208→  // 0xabc4129f7e8d2c1b → 0xabc412…2c1b
   209→  if (h.length <= 14) return h;
   210→  return `${h.slice(0, 8)}…${h.slice(-4)}`;
   211→}

   213→function shortDate(iso: string, locale: "en" | "ar"): string {

   217→  } catch {
   218→    return iso;
   219→  }

   232→interface VerifyResult {
   233→  ok: boolean;
   234→  verifiedCount: number;
   235→  totalCount: number;
   236→  tamperedEventId?: string;
   237→  expectedPrev?: string;
   238→  foundPrev?: string;
   239→  lastVerifiedId?: string;
   240→}
   241→
   242→/** Walk the chain and verify every prev-hash links to the prior event's hash. */
   243→function verifyChain(chain: AuditEventRow[]): VerifyResult {
   244→  if (chain.length === 0) {
   245→    return { ok: true, verifiedCount: 0, totalCount: 0 };
   246→  }
   247→  // Genesis event: its prevHash must be the zero marker "0x0000000000000000".
   248→  const ZERO = "0x0000000000000000";
   249→  if (chain[0].prevHash !== ZERO) {
   250→    return {
   251→      ok: false,
   252→      verifiedCount: 0,
   253→      totalCount: chain.length,
   254→      tamperedEventId: chain[0].id,
   255→      expectedPrev: ZERO,
   256→      foundPrev: chain[0].prevHash,
   257→      lastVerifiedId: undefined,
   258→    };
   259→  }
   260→  let lastVerifiedId: string | undefined = chain[0].id;
   261→  for (let i = 1; i < chain.length; i += 1) {
   262→    const prev = chain[i - 1];
   263→    const cur = chain[i];
   264→    if (cur.prevHash !== prev.hash) {
   265→      return {
   266→        ok: false,
   267→        verifiedCount: i,
   268→        totalCount: chain.length,
   269→        tamperedEventId: cur.id,
   270→        expectedPrev: prev.hash,
   271→        foundPrev: cur.prevHash,
   272→        lastVerifiedId,
   273→      };
   274→    }
   275→    lastVerifiedId = cur.id;
   276→  }
   277→  return {
   278→    ok: true,
   279→    verifiedCount: chain.length,
   280→    totalCount: chain.length,
   281→  };
   282→}

   285→  return src.map((e) => ({ ...e }));

   288→/** Apply a tamper to a fresh copy of the chain. Returns the tampered copy and the set of affected ids. */
   289→function applyTamper(type: TamperType): { chain: AuditEventRow[]; tamperedIds: string[] } {
   290→  const chain = cloneChain(AUDIT_EVENTS);
   291→  const tamperedIds: string[] = [];
   292→  if (type === "alter_hash") {
   293→    // ae-003's prevHash is rewritten to a fake value, breaking the link
   294→    // at ae-003. (Matches the spec's example verification message.)
   295→    const target = chain.find((e) => e.id === "ae-003");
   296→    if (target) {
   297→      target.prevHash = "0xFAKEFAKEFAKEFAKE";
   298→      tamperedIds.push("ae-003");
   299→    }
   300→  } else if (type === "delete_event") {
   301→    // Remove ae-002 → ae-003's prevHash no longer links to the prior event (ae-001).
   302→    const idx = chain.findIndex((e) => e.id === "ae-002");
   303→    if (idx >= 0) {
   304→      chain.splice(idx, 1);
   305→      const orphan = chain.find((e) => e.id === "ae-003");
   306→      if (orphan) tamperedIds.push(orphan.id);
   307→    }
   308→  } else if (type === "reorder_events") {
   309→    // Swap ae-002 and ae-003 → both become orphans of the new ordering.
   310→    const a = chain.findIndex((e) => e.id === "ae-002");
   311→    const b = chain.findIndex((e) => e.id === "ae-003");
   312→    if (a >= 0 && b >= 0) {
   313→      const tmp = chain[a];
   314→      chain[a] = chain[b];
   315→      chain[b] = tmp;
   316→      tamperedIds.push("ae-002", "ae-003");
   317→    }
   318→  } else if (type === "inject_fake") {
   319→    // Insert a synthetic event between ae-002 and ae-003 with a wrong prev hash.
   320→    const idx = chain.findIndex((e) => e.id === "ae-003");
   321→    if (idx >= 0) {
   322→      const fake: AuditEventRow = {
   323→        id: "ae-fake",
   324→        actor: "system",
   325→        action: "ingestion",
   326→        subject: "pack",
   327→        subjectId: "synthetic-fake-v1",
   328→        hash: "0xDEADBEEFDEADBEEF",
   329→        prevHash: "0xWRONGPREV000000",
   330→        at: "2024-10-19T11:00:00Z",
   331→      };
   332→      chain.splice(idx, 0, fake);
   333→      tamperedIds.push("ae-fake");
   334→    }
   335→  }
   336→  return { chain, tamperedIds };
   337→}

   428→/** Small teal sparkline (7-day synthetic trend) for ingestion stat tiles. */
   429→function Sparkline({
   430→  data,
   431→  width = 88,
   432→  height = 22,
   433→}: {
   434→  data: number[];
   435→  width?: number;
   436→  height?: number;
   437→}) {
   438→  const reduce = useReducedMotion();
   439→  const max = Math.max(...data, 1);
   440→  const min = Math.min(...data, 0);
   441→  const range = max - min || 1;
   442→  const stepX = width / Math.max(data.length - 1, 1);
   443→  const pts: Array<[number, number]> = data.map((v, i) => {
   444→    const x = i * stepX;
   445→    const y = height - ((v - min) / range) * (height - 4) - 2;
   446→    return [x, y];
   447→  });
   448→  const path = pts
   449→    .map(([x, y], i) => `${i === 0 ? "M" : "L"} ${x.toFixed(1)} ${y.toFixed(1)}`)
   450→    .join(" ");
   451→  const last = pts[pts.length - 1];
   452→  if (!last) return null;
   453→  return (
   454→    <svg width={width} height={height} className="block" aria-hidden="true">
   455→      <motion.path
   456→        d={path}
   457→        fill="none"
   458→        stroke="var(--brand-400)"
   459→        strokeWidth={1.4}
   460→        strokeLinecap="round"
   461→        strokeLinejoin="round"
   462→        initial={reduce ? { pathLength: 1 } : { pathLength: 0 }}
   463→        animate={{ pathLength: 1 }}
   464→        transition={{ duration: 0.6, ease: "easeOut" }}
   465→      />
   466→      <motion.circle
   467→        cx={last[0]}
   468→        cy={last[1]}
   469→        r={1.6}
   470→        fill="var(--brand-400)"
   471→        initial={reduce ? { opacity: 1 } : { opacity: 0 }}
   472→        animate={{ opacity: 1 }}
   473→        transition={{ delay: reduce ? 0 : 0.45, duration: 0.2 }}
   474→      />
   475→    </svg>
   476→  );
   477→}
