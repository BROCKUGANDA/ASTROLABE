   489→/* ============================================================
   490→   HERO PARALLAX — 3D astrolabe + hero text
   491→   - Astrolabe canvas shifts max 8px opposite the cursor
   492→   - Hero text shifts max 4px with the cursor
   493→   - 60fps via framer-motion useMotionValue + useTransform
   494→   - Disabled on touch devices and when prefers-reduced-motion
   495→   ============================================================ */
   496→
   497→function usePrefersReducedMotion(): boolean {
   498→  const [reduce, setReduce] = useState(false);
   499→  useEffect(() => {
   500→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   501→    const update = () => setReduce(mq.matches);
   502→    update();
   503→    mq.addEventListener("change", update);
   504→    return () => mq.removeEventListener("change", update);
   505→  }, []);
   506→  return reduce;
   507→}
   508→
   509→function useFinePointer(): boolean {
   510→  const [fine, setFine] = useState(false);
   511→  useEffect(() => {
   512→    const mq = window.matchMedia("(pointer: fine)");
   513→    const update = () => setFine(mq.matches);
   514→    update();
   515→    mq.addEventListener("change", update);
   516→    return () => mq.removeEventListener("change", update);
   517→  }, []);
   518→  return fine;
   519→}

   521→function HeroParallax() {
   522→  const locale = useAppStore((s) => s.locale);
   523→  const setView = useAppStore((s) => s.setView);
   524→  const reduce = usePrefersReducedMotion();
   525→  const fine = useFinePointer();
   526→  const enabled = !reduce && fine;
   527→
   528→  // Normalised cursor offset from the centre of the hero, in [-0.5, 0.5].
   529→  const mx = useMotionValue(0);
   530→  const my = useMotionValue(0);
   531→
   532→  // Astrolabe: 8px in the opposite direction of the cursor.
   533→  const astX: MotionValue<number> = useTransform(mx, [-0.5, 0.5], [8, -8]);
   534→  const astY: MotionValue<number> = useTransform(my, [-0.5, 0.5], [8, -8]);
   535→  // Hero text: 4px in the same direction as the cursor.
   536→  const textX: MotionValue<number> = useTransform(mx, [-0.5, 0.5], [-4, 4]);
   537→  const textY: MotionValue<number> = useTransform(my, [-0.5, 0.5], [-4, 4]);
   538→
   539→  const handleMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
   540→    if (!enabled) return;
   541→    const rect = e.currentTarget.getBoundingClientRect();
   542→    mx.set((e.clientX - rect.left) / rect.width - 0.5);
   543→    my.set((e.clientY - rect.top) / rect.height - 0.5);
   544→  };
   545→
   546→  const handleMouseLeave = () => {
   547→    mx.set(0);
   548→    my.set(0);
   549→  };
