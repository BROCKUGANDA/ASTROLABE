   244→

   315→function ScenariosView({ onLoad }: { onLoad: (employeeId: string) => void }) {
   316→  const locale = useAppStore((s) => s.locale);
   317→  const setEmployee = useAppStore((s) => s.setEmployee);
   318→  const setOffset = useAppStore((s) => s.setAsOfOffset);
   319→  const [scenarios, setScenarios] = useState<Scenario[] | null>(null);
   320→  const [employees, setEmployees] = useState<Record<string, EmployeeLite> | null>(null);
   321→  const [previewAssess, setPreviewAssess] = useState<Record<string, Assessment | null>>({});
   322→  const [selectedId, setSelectedId] = useState<string | null>(null);
   323→  const [err, setErr] = useState<string | null>(null);
   324→  const [loading, setLoading] = useState(true);
   325→
   326→  const fetchAll = useCallback(async () => {
   327→    setLoading(true);
   328→    setErr(null);
   329→    try {
   330→      const [scRes, empRes] = await Promise.all([
   331→        fetch("/api/v1/scenarios", { cache: "no-store" }),
   332→        fetch("/api/v1/employees", { cache: "no-store" }),
   333→      ]);
   334→      if (!scRes.ok || !empRes.ok) throw new Error("HTTP error");
   335→      const scJson: ApiEnvelope<{ scenarios: Scenario[] }> = await scRes.json();
   336→      const empJson: ApiEnvelope<{ items: EmployeeLite[] }> = await empRes.json();
   337→      if (!scJson.ok || !scJson.data || !empJson.ok || !empJson.data) {
   338→        throw new Error(scJson.error?.message ?? empJson.error?.message ?? "Unknown error");
   339→      }
   340→      setScenarios(scJson.data.scenarios);
   341→      const map: Record<string, EmployeeLite> = {};
   342→      for (const e of empJson.data.items) map[e.id] = e;
   343→      setEmployees(map);
   344→    } catch (e) {
   345→      setErr(e instanceof Error ? e.message : "Network error");
   346→    } finally {
   347→      setLoading(false);
   348→    }
   349→  }, []);
   350→
   351→  useEffect(() => { fetchAll(); }, [fetchAll]);
   352→
   353→  // Live preview assessment for selected scenario (recomputes with asOf offset)
   354→  const offsetHours = useAppStore((s) => s.asOfOffsetHours);
   355→  useEffect(() => {
   356→    if (!selectedId || !scenarios) return;
   357→    const sc = scenarios.find((s) => s.id === selectedId);
   358→    if (!sc) return;
   359→    const asOf = new Date(PACK_NOW.getTime() + offsetHours * 3600 * 1000).toISOString();
   360→    let cancelled = false;
   361→    fetch(`/api/v1/journey/${encodeURIComponent(sc.employeeId)}/assessment?asOf=${encodeURIComponent(asOf)}`, { cache: "no-store" })
   362→      .then((r) => r.json())
   363→      .then((json: ApiEnvelope<{ assessment: Assessment; trace: EngineTrace }>) => {
   364→        if (cancelled) return;
   365→        if (json.ok && json.data) {
   366→          setPreviewAssess((prev) => ({ ...prev, [sc.id]: json.data!.assessment }));
   367→        }
   368→      })
   369→      .catch(() => {});
   370→    return () => { cancelled = true; };
   371→  }, [selectedId, scenarios, offsetHours]);
   372→
   373→  const handleLoad = useCallback((sc: Scenario) => {
   374→    setEmployee(sc.employeeId);
   375→    onLoad(sc.employeeId);
   376→  }, [onLoad, setEmployee]);
   377→
   378→  if (loading) {
   379→    return (
   380→      <div className="space-y-3" aria-hidden="true">
   381→        <div className="ast-card h-24 ast-shimmer rounded-[14px]" />
   382→        {Array.from({ length: 5 }).map((_, i) => (
   383→          <div key={i} className="ast-card p-4 h-24 ast-shimmer rounded-[12px]" />
   384→        ))}
   385→      </div>
   386→    );
   387→  }
   388→  if (err) return <ErrorBanner message={err} onRetry={fetchAll} />;
   389→
   390→  return (
   391→    <div className="space-y-5">
   392→      <div>
   393→        <h2 className="text-[17px] font-semibold" style={{ color: "var(--text-hi)" }}>
   394→          {locale === "ar" ? "السيناريوهات الخمسة" : "Five scenarios"}
   395→        </h2>
   396→        <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   397→          {locale === "ar"
   398→            ? "كل بطاقة تُحمّل موظفاً في الأداة وتعرض التصنيف المتوقع. حرّك الساعة لرؤية المحرك يعيد التقييم."
   399→            : "Each card loads an employee into the instrument and previews the expected band. Drag the clock to watch the engine re-assess."}
   400→        </p>
   401→      </div>
   402→
   403→      <AsOfClock />
   404→
   405→      <ul className="grid grid-cols-1 lg:grid-cols-2 gap-3">
   406→        {scenarios?.map((sc) => {
   407→          const emp = employees?.[sc.employeeId];
   408→          const a = previewAssess[sc.id];
   409→          const isSelected = selectedId === sc.id;
   410→          return (
   411→            <motion.li
   412→              key={sc.id}
   413→              initial={false}
   414→              whileHover={{ y: -2 }}
   415→              onClick={() => { setSelectedId(sc.id); setEmployee(sc.employeeId); }}
   416→              className={cn("ast-card p-4 sm:p-5 cursor-pointer ast-focus transition-colors")}
   417→              style={{
   418→                borderColor: isSelected ? "rgba(22,154,165,0.40)" : "var(--border)",
   419→                background: isSelected ? "rgba(22,154,165,0.04)" : "var(--surface-1)",
   420→              }}
   421→            >
   422→              <div className="flex items-baseline justify-between gap-3 mb-2">
   423→                <div className="flex items-center gap-2">
   424→                  <span
   425→                    className="font-mono text-[12px] tabular px-2 py-0.5 rounded"
   426→                    style={{ background: "var(--surface-3)", color: "var(--brand-400)" }}
   427→                  >
   428→                    {sc.id}
   429→                  </span>
   430→                  <h3 className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   431→                    {sc.title}
   432→                  </h3>
   433→                </div>
   434→                {a && <BandChip band={a.band} score={a.score} locale={locale} size="sm" />}
   435→              </div>
   436→              <p className="text-[12px] leading-relaxed" style={{ color: "var(--text-mid)" }}>
   437→                {sc.description}
   438→              </p>
   439→              <div className="flex items-center justify-between mt-3 flex-wrap gap-2">
   440→                <p className="text-[11px] font-mono tabular" style={{ color: "var(--text-lo)" }} dir="ltr">
   441→                  {sc.employeeId}{emp ? ` · ${emp.displayName}` : ""}
   442→                </p>
   443→                <button
   444→                  onClick={(e) => { e.stopPropagation(); handleLoad(sc); }}
   445→                  className="ast-focus inline-flex items-center gap-1.5 h-9 px-3 rounded-full text-[12px] font-medium"
   446→                  style={{
   447→                    background: "var(--brand-500)",
   448→                    color: "var(--surface-0)",
   449→                    border: "1px solid var(--brand-500)",
   450→                  }}
   451→                >
   452→                  <PlayCircle className="w-3.5 h-3.5" strokeWidth={1.75} />
   453→                  {locale === "ar" ? "حمّل السيناريو" : "Load scenario"}
   454→                </button>
   455→              </div>
   456→            </motion.li>
   457→          );
   458→        })}
   459→      </ul>
   460→    </div>
   461→  );
   462→}
