   265→/* ============================================================
   266→   Component
   267→   ============================================================ */
   268→export function ManagerScreen() {
   269→  const locale = useAppStore((s) => s.locale);
   270→  const view = useAppStore((s) => s.managerView);
   271→  const dir = locale === "ar" ? "rtl" : "ltr";
   272→
   273→  const [items, setItems] = useState<CohortItem[] | null>(null);
   274→  const [err, setErr] = useState<string | null>(null);
   275→  const [loading, setLoading] = useState(true);
   276→  // Per-manager nudge history (PRD §11-205). Seeded with 5 historical
   277→  // entries so the bar chart demonstrates that nudges convert.
   278→  const [nudgeHistory, setNudgeHistory] = useState<NudgeHistoryEntry[]>(SEED_NUDGE_HISTORY);
   279→
   280→  const fetchCohort = useCallback(async () => {
   281→    setLoading(true);
   282→    setErr(null);
   283→    try {
   284→      const res = await fetch("/api/v1/manager/cohort?managerId=MGR-1", { cache: "no-store" });
   285→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   286→      const json: ApiEnvelope<{ managerId: string; items: CohortItem[] }> = await res.json();
   287→      if (!json.ok || !json.data) throw new Error(json.error?.message ?? "Unknown error");
   288→      setItems(json.data.items);
   289→    } catch (e) {
   290→      setErr(e instanceof Error ? e.message : "Network error");
   291→    } finally {
   292→      setLoading(false);
   293→    }
   294→  }, []);
   295→
   296→  useEffect(() => { fetchCohort(); }, [fetchCohort]);
   297→
   298→  // Aggregate metrics
   299→  const stats = useMemo(() => {
   300→    if (!items) return { joiners: 0, waitingOnMe: 0, avgCompletion: 0 };
   301→    const joiners = items.length;
   302→    let waiting = 0;
   303→    let sumCompletion = 0;
   304→    for (const it of items) {
   305→      sumCompletion += it.aggregateCompletion;
   306→      for (const s of it.ownedSteps) {
   307→        if (s.status === "waiting_owner") waiting += 1;
   308→      }
   309→    }
   310→    return {
   311→      joiners,
   312→      waitingOnMe: waiting,
   313→      avgCompletion: joiners > 0 ? Math.round(sumCompletion / joiners) : 0,
   314→    };
   315→  }, [items]);
   316→
   317→  // Flatten owned steps across cohort (for "my obligations")
   318→  const allObligations = useMemo(() => {
   319→    if (!items) return [] as { employee: CohortItem; step: OwnedStep }[];
   320→    const out: { employee: CohortItem; step: OwnedStep }[] = [];
   321→    for (const it of items) {
   322→      for (const s of it.ownedSteps) {
   323→        out.push({ employee: it, step: s });
   324→      }
   325→    }
   326→    // Sort: waiting_owner first, then by dueAt asc
   327→    out.sort((a, b) => {
   328→      const aWait = a.step.status === "waiting_owner" ? 0 : 1;
   329→      const bWait = b.step.status === "waiting_owner" ? 0 : 1;
   330→      if (aWait !== bWait) return aWait - bWait;
   331→      const aDue = a.step.dueAt ? new Date(a.step.dueAt).getTime() : Number.MAX_SAFE_INTEGER;
   332→      const bDue = b.step.dueAt ? new Date(b.step.dueAt).getTime() : Number.MAX_SAFE_INTEGER;
   333→      return aDue - bDue;
   334→    });
   335→    return out;
   336→  }, [items]);
   337→
   338→  // Late steps = waiting_owner AND (dueAt < NOW OR breachLevel breached/hard).
   339→  // Used by the late-step nudge banner + per-step nudge icon.
   340→  const lateSteps = useMemo(() => {
   341→    return allObligations
   342→      .filter(({ step }) => isStepLate(step))
   343→      .map(({ employee, step }) => ({
   344→        employee,
   345→        step,
   346→        daysLate: daysLateFor(step.dueAt),
   347→      }));
   348→  }, [allObligations]);
   349→
   350→  const handleNudgeDaily = useCallback(() => {
   351→    toast.success(
   352→      locale === "ar"
   353→        ? "جُدول التنبيه اليومي"
   354→        : "Daily nudge scheduled",
   355→      {
   356→        description: locale === "ar"
   357→          ? "سنُذكّرك كل صباح حتى تنتهي هذه الخطوات."
   358→          : "We'll remind you every morning until these are done.",
   359→      }
   360→    );
   361→  }, [locale]);
   362→
   363→  const handleSnoozeToday = useCallback(() => {
   364→    setNudgeHistory((prev) => [
   365→      {
   366→        id: `nh-live-${Date.now()}`,
   367→        date: new Date(PACK_NOW.getTime() - 0).toISOString(),
   368→        stepLabel: locale === "ar" ? "كل الخطوات المتأخرة" : "All late steps",
   369→        employeeName: locale === "ar" ? "أنت" : "You",
   370→        action: "snoozed" as NudgeAction,
   371→      },
   372→      ...prev,
   373→    ]);
   374→    toast.success(
   375→      locale === "ar"
   376→        ? "أُجّل حتى الغد 09:00"
   377→        : "Snoozed until tomorrow 09:00",
   378→      {
   379→        description: locale === "ar"
   380→          ? "سنُذكّرك عندها."
   381→          : "We'll remind you then.",
   382→      }
   383→    );
   384→  }, [locale]);
   385→
   386→  const handleNudge = useCallback(
   387→    (schedule: NudgeSchedule, step: OwnedStep, employee: CohortItem) => {
   388→      const labels: Record<NudgeSchedule, { en: string; ar: string }> = {
   389→        "1h":       { en: "in 1 hour",       ar: "بعد ساعة" },
   390→        "4h":       { en: "in 4 hours",      ar: "بعد 4 ساعات" },
   391→        tomorrow:   { en: "tomorrow at 09:00", ar: "غداً 09:00" },
   392→        never:      { en: "Never",            ar: "أبدًا" },
   393→      };
   394→      const lbl = labels[schedule];
   395→      if (schedule === "never") {
   396→        setNudgeHistory((prev) => [
   397→          {
   398→            id: `nh-live-${Date.now()}`,
   399→            date: new Date(PACK_NOW.getTime()).toISOString(),
   400→            stepLabel: step.label,
   401→            employeeName: employee.employeeName,
   402→            action: "ignored" as NudgeAction,
   403→          },
   404→          ...prev,
   405→        ]);
   406→        toast.warning(
   407→          locale === "ar" ? "لن نُذكّرك بهذه الخطوة" : "We won't nudge you about this step",
   408→          { description: `${step.label} · ${employee.employeeName}` }
   409→        );
   410→      } else {
   411→        setNudgeHistory((prev) => [
   412→          {
   413→            id: `nh-live-${Date.now()}`,
   414→            date: new Date(PACK_NOW.getTime()).toISOString(),
   415→            stepLabel: step.label,
   416→            employeeName: employee.employeeName,
   417→            action: "snoozed" as NudgeAction,
   418→          },
   419→          ...prev,
   420→        ]);
   421→        toast.success(
   422→          locale === "ar" ? "جُدول التنبيه" : "Nudge scheduled",
   423→          {
   424→            description: `${step.label} · ${employee.employeeName} · ${locale === "ar" ? lbl.ar : lbl.en}`,
   425→          }
   426→        );
   427→      }
   428→    },
   429→    [locale]
   430→  );
   431→
   432→  return (
   433→    <div
   434→      dir={dir}
   435→      className="max-w-[1180px] mx-auto px-4 sm:px-6 lg:px-8 py-6 lg:py-10"
   436→    >
   437→      <header className="flex items-center justify-between mb-6 flex-wrap gap-3">
   438→        <div>
   439→          <div className="flex items-center gap-2 text-[11px] font-semibold uppercase tracking-[0.18em]" style={{ color: "var(--brand-400)" }}>
   440→            <Users className="w-3.5 h-3.5" strokeWidth={1.75} />
   441→            <span>{locale === "ar" ? "وحدة المدير" : "Manager console"}</span>
   442→          </div>
   443→          <h1 className="mt-2 text-[26px] sm:text-[30px] font-semibold tracking-tight" style={{ color: "var(--text-hi)" }}>
   444→            {locale === "ar" ? "التزاماتك تجاه فريقك" : "Your obligations to your team"}
   445→          </h1>
   446→        </div>
   447→        <ManagerTabs />
   448→      </header>
   449→
   450→      <PrivacyBanner locale={locale} />
   451→
   452→      {loading ? (
   453→        <ManagerSkeleton />
   454→      ) : err ? (
   455→        <ErrorBanner message={err} onRetry={fetchCohort} />
   456→      ) : !items ? null : items.length === 0 ? (
   457→        <div className="ast-card p-10 flex flex-col items-center text-center">
   458→          <div
   459→            className="flex items-center justify-center w-14 h-14 rounded-full mb-4"
   460→            style={{ background: "var(--surface-2)", color: "var(--text-mid)" }}
   461→          >
   462→            <Inbox className="w-6 h-6" strokeWidth={1.75} />
   463→          </div>
   464→          <p className="text-[15px] font-medium" style={{ color: "var(--text-hi)" }}>
   465→            {locale === "ar" ? "لا شيء بانتظارك." : "Nothing waiting on you."}
   466→          </p>
   467→          <p className="text-[13px] mt-1" style={{ color: "var(--text-mid)" }}>
   468→            {locale === "ar"
   469→              ? "عندما يصل موظف جديد أو ينتقل أحد، ستظهر التزاماتك هنا."
   470→              : "When a joiner or mover lands on your team, your obligations will appear here."}
   471→          </p>
   472→        </div>
   473→      ) : (
   474→        <>
   475→          {/* Hero numbers */}
   476→          <section className="grid grid-cols-1 sm:grid-cols-3 gap-3 mb-5">
   477→            {/* Waiting on me — sand-gold (primary) */}
   478→            <div
   479→              className="ast-card p-5 relative overflow-hidden"
   480→              style={{
   481→                borderColor: "rgba(201,162,39,0.40)",
   482→                background: "linear-gradient(135deg, rgba(201,162,39,0.10) 0%, var(--surface-1) 60%)",
   483→              }}
   484→            >
   485→              <div
   486→                className="absolute top-0 left-0 right-0 h-[3px]"
   487→                style={{ background: "var(--accent-500)" }}
   488→              />
   489→              <p className="text-[11px] uppercase tracking-[0.14em] font-medium" style={{ color: "var(--accent-500)" }}>
   490→                {locale === "ar" ? "بانتظارك" : "Waiting on you"}
   491→              </p>
   492→              <p className="text-[44px] leading-none font-semibold mt-2 tabular ast-brass">
   493→                {stats.waitingOnMe}
   494→              </p>
   495→              <p className="text-[12px] mt-2" style={{ color: "var(--text-mid)" }}>
   496→                {locale === "ar"
   497→                  ? `خطوة تنتظر إكمالك عبر ${stats.joiners} موظف`
   498→                  : `step${stats.waitingOnMe === 1 ? "" : "s"} you own across ${stats.joiners} joiner${stats.joiners === 1 ? "" : "s"}`}
   499→              </p>
   500→            </div>
   501→
   502→            {/* Joiners count */}
   503→            <div className="ast-card p-5">

   724→/* ---------- Late step detection (PRD §11-205) ---------- */
   725→
   726→/** A step is "late" when ANY of:
   727→ *  1. status is waiting_owner AND its dueAt is in the past, OR
   728→ *  2. breachLevel is breached/hard (regardless of status — covers
   729→ *     in_progress steps that have blown past their SLA target too).
   730→ *
   731→ * Per PRD §11-205: "(status=waiting_owner AND dueAt < NOW,
   732→ * OR breachLevel=breached/hard)". */
   733→function isStepLate(step: OwnedStep): boolean {
   734→  // (2) breachLevel breach/hard → always late.
   735→  if (step.breachLevel === "breached" || step.breachLevel === "hard") return true;
   736→  // (1) waiting_owner + past dueAt.
   737→  if (step.status !== "waiting_owner") return false;
   738→  if (step.dueAt) {
   739→    try {
   740→      const due = new Date(step.dueAt);
   741→      if (due.getTime() < PACK_NOW.getTime()) return true;
   742→    } catch {
   743→      /* ignore parse errors */
   744→    }
   745→  }
   746→  return false;
   747→}
   748→
   749→function daysLateFor(dueAt?: string): number {
   750→  if (!dueAt) return 0;
   751→  try {
   752→    const due = new Date(dueAt);
   753→    return Math.max(0, differenceInCalendarDays(PACK_NOW, due));
   754→  } catch {
   755→    return 0;
   756→  }
   757→}
