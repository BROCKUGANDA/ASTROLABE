   542→export function getConsent(employeeId: string, purpose: ConsentRecord["purpose"]): boolean {
   543→  const records = consentRecords
   544→    .filter((r) => r.employeeId === employeeId && r.purpose === purpose)
   545→    .sort((a, b) => b.grantedAt.localeCompare(a.grantedAt));
   546→  if (records.length === 0) {
   547→    if (purpose === "journey_assistance") return true;
   548→    if (purpose === "manager_visibility") return true;
   549→    if (purpose === "aggregate_analytics") return true;
   550→    return false;
   551→  }
   552→  const latest = records[0];
   553→  return latest.granted && !latest.revokedAt;
   554→}
