   145→function namedOwnerForGroup(group: string): { name: string; nameAr?: string; role: string; group: string } | null {
   146→  const o = ALL_OWNERS.find((x) => x.group === group);
   147→  if (!o) return null;

   151→function stageLabel(stageKey: string, journeyType: string, locale: "en" | "ar"): string {
   152→  const st = ALL_STAGES.find((s) => s.key === stageKey && s.journeyType === journeyType);
   153→  if (!st) return stageKey;
   154→  return locale === "ar" ? st.labelAr : st.label;
   155→}

   157→function stageShort(stageKey: string, journeyType: string, locale: "en" | "ar"): string {
   158→  const full = stageLabel(stageKey, journeyType, locale);
   159→  // Compact rail labels: take first 1–2 words
   160→  if (locale === "ar") return full;
   161→  const parts = full.split(" ");
   162→  return parts.length > 2 ? parts.slice(0, 2).join(" ") : full;
   163→}

   165→function fmtDateAndDistance(iso: string | undefined, locale: "en" | "ar"): string {
   166→  if (!iso) return locale === "ar" ? "في أي وقت" : "any time";
   167→  const d = new Date(iso);
   168→  const datePart = format(d, "EEEE d MMM");
   169→  // Use synthetic NOW from pack for deterministic "in 3 days" copy
   170→  const distance = formatDistance(d, PACK_NOW, { addSuffix: false });
   171→  return locale === "ar" ? `${datePart} · ${distance}` : `${datePart} · ${distance}`;
   172→}

   174→function lateCopy(step: Step, locale: "en" | "ar"): string {
   175→  if (step.breachLevel === "hard") {
   176→    const days = Math.max(1, Math.floor((step.elapsedHours - step.targetHours) / 24));
   177→    return locale === "ar"
   178→      ? `متأخر بـ ${days} أيام — خرق جوهري`
   179→      : `${days} day${days === 1 ? "" : "s"} late — hard breach`;
   180→  }
   181→  if (step.breachLevel === "breached") {
   182→    const days = Math.max(1, Math.floor((step.elapsedHours - step.targetHours) / 24));
   183→    return locale === "ar" ? `متأخر بـ ${days} أيام` : `${days} day${days === 1 ? "" : "s"} late`;
   184→  }
   185→  if (step.status === "blocked") {
   186→    return locale === "ar"
   187→      ? "محظور بانتظار خطوات أخرى"
   188→      : "Held up — waiting on another step";
   189→  }
   190→  return "";
   191→}

   193→function ownerLine(step: Step, locale: "en" | "ar"): string {
   194→  const grp = OWNER_GROUP_LABEL[step.ownerGroup];
   195→  const named = namedOwnerForGroup(step.ownerGroup);
   196→  if (!grp) return step.ownerGroup;
   197→  if (named) {
   198→    return locale === "ar" && named.nameAr
   199→      ? `مع ${named.nameAr} في ${grp.ar}`
   200→      : `with ${named.name} in ${grp.en}`;
   201→  }
   202→  return locale === "ar" ? `مع ${grp.ar}` : `with ${grp.en}`;
   203→}

   209→    ? `يُتوقع خلال نحو ${days} يوم`
   210→    : `about ${days} day${days === 1 ? "" : "s"} from start`;
   211→}

   490→/* Parse "docSlug#section" → CitationChip shape */
   491→function parsePolicyRef(ref: string): { docSlug: string; title: string; section: string; effectiveDate: string } {
   492→  const [docSlug, section] = ref.split("#");
   493→  const titleMap: Record<string, string> = {
   494→    "relocation-policy": "Relocation Policy",
   495→    "immigration-policy": "Immigration Policy",
   496→    "it-policy": "IT Provisioning Policy",
   497→    "onboarding-policy": "Onboarding Policy",
   498→    "learning-policy": "Learning Policy",
   499→    "mobility-policy": "Internal Mobility Policy",
   500→    "payroll-policy": "Payroll Policy",
   501→    "facilities-policy": "Facilities Policy",
   502→  };
   503→  return {
   504→    docSlug,
   505→    title: titleMap[docSlug] ?? docSlug,

   624→/* ---------- Stage Rail ---------- */
   625→
   626→function StageRail({ data }: { data: JourneyMeData }) {
   627→  const { locale } = useAppStore();
   628→  const journeyType = data.employee.journeyType;
   629→  const journeyStages = useMemo(
   630→    () => ALL_STAGES.filter((s) => s.journeyType === journeyType).sort((a, b) => a.order - b.order),
   631→    [journeyType]
   632→  );
   633→
   634→  const segState = (stageKey: string): "done" | "active" | "blocked" | "pending" => {
   635→    const stepsInStage = data.steps.filter((s) => s.stage === stageKey);
   636→    if (stepsInStage.length === 0) return "pending";
   637→    const allDone = stepsInStage.every((s) => s.status === "done" || s.status === "skipped");
   638→    if (allDone) return "done";
   639→    const anyBlocked = stepsInStage.some(
   640→      (s) => s.status === "blocked" || s.breachLevel === "hard"
   641→    );
   642→    if (anyBlocked) return "blocked";
   643→    if (stageKey === data.stage) return "active";
   644→    // Stage before current — if not all done, mark done visually (history)
   645→    const stageOrder = journeyStages.find((s) => s.key === stageKey)?.order ?? 0;
   646→    const currentOrder = journeyStages.find((s) => s.key === data.stage)?.order ?? 0;
   647→    if (stageOrder < currentOrder) return "done";
   648→    return "pending";
   649→  };
   650→
   651→  return (
   652→    <section className="ast-card p-5 sm:p-6" aria-label="Stage rail">
   653→      <div className="flex items-center gap-2 mb-4">
   654→        <Compass className="w-4 h-4" strokeWidth={1.75} style={{ color: "var(--brand-400)" }} />
   655→        <h3 className="text-[14px] font-semibold uppercase tracking-[0.12em]" style={{ color: "var(--text-hi)" }}>
   656→          {locale === "ar" ? "مسار الرحلة" : "Journey rail"}
   657→        </h3>
   658→      </div>
   659→      <ol className="grid gap-2" style={{ gridTemplateColumns: `repeat(${journeyStages.length}, minmax(0, 1fr))` }}>
   660→        {journeyStages.map((st) => {
   661→          const state = segState(st.key);
   662→          const isActive = st.key === data.stage;
   663→          const label = locale === "ar" ? st.labelAr : st.label;
   664→          return (
   665→            <li key={st.key} className="flex flex-col gap-2 min-w-0">
   666→              <div
   667→                className="ast-rail-seg"
   668→                data-state={state === "pending" ? undefined : state}
   669→                aria-label={`${label}: ${state}`}
   670→                style={state === "pending" ? { background: "var(--surface-3)" } : undefined}
   671→              />
   672→              <div className="min-w-0">
   673→                <p
   674→                  className="text-[11px] sm:text-[12px] font-medium leading-tight truncate"
   675→                  style={{
   676→                    color: isActive ? "var(--accent-500)" : state === "done" ? "var(--brand-400)" : state === "blocked" ? "var(--status-blocked)" : "var(--text-lo)",
   677→                  }}
   678→                >
   679→                  {stageShort(st.key, journeyType, locale)}
   680→                </p>
   681→                {isActive && (
   682→                  <p className="text-[10px] tabular mt-0.5" style={{ color: "var(--text-lo)" }}>

   996→        toast.success(locale === "ar" ? "تم تسجيل البدء" : "Step started", {
   997→          description: locale === "ar" ? nbs.labelAr : nbs.label,
   998→        });
   999→        return;
  1000→      }
  1001→      const idempotencyKey = `idm-${emp.id}-${requestType}-${Date.now().toString(36)}`;
  1002→      try {
  1003→        const res = await fetch("/api/v1/requests", {
  1004→          method: "POST",
  1005→          headers: { "Content-Type": "application/json" },
  1006→          body: JSON.stringify({
  1007→            employeeId: emp.id,
  1008→            type: requestType,
  1009→            payload: { stepKey: nbs.stepKey, label: nbs.label },
  1010→            idempotencyKey,
  1011→          }),
  1012→        });
  1013→        const json: ApiEnvelope<{ ref: string; owner?: { name: string } }> = await res.json();
  1014→        if (json.ok && json.data) {
  1015→          const ownerName = json.data.owner?.name ?? "the team";
  1016→          toast.success(locale === "ar" ? "تم إرسال الطلب" : "Request sent", {
  1017→            description: locale === "ar"
  1018→              ? `${nbs.labelAr} — ${ownerName} سيتولى الأمر`
  1019→              : `${nbs.label} — ${ownerName} will handle it`,
  1020→          });
  1021→        } else {
  1022→          toast.error(locale === "ar" ? "تعذّر الإرسال" : "Couldn't send", {
  1023→            description: json.error?.message ?? "Please try again.",
  1024→          });
  1025→        }
  1026→      } catch {
  1027→        toast.error(locale === "ar" ? "خطأ في الشبكة" : "Network error", {
  1028→          description: locale === "ar" ? "حاول مرة أخرى." : "Please retry.",
  1029→        });
  1030→      }
  1031→    },
  1032→    [data, locale]
  1033→  );
  1034→
  1035→  const handleGrantConsent = useCallback(async () => {
  1036→    try {
  1037→      const res = await fetch("/api/v1/consent", {
  1038→        method: "POST",
  1039→        headers: { "Content-Type": "application/json" },
  1040→        body: JSON.stringify({
  1041→          employeeId,
  1042→          purpose: "pulse_capture",
  1043→          granted: true,
  1044→          version: "v1",
  1045→        }),
  1046→      });
  1047→      const json: ApiEnvelope<{ receiptId: string }> = await res.json();
  1048→      if (json.ok) {
  1049→        setPulseGranted(true);
  1050→        toast.success(locale === "ar" ? "تم تفعيل التقدير" : "Pulse capture enabled", {
  1051→          description: locale === "ar"
  1052→            ? `رقم الإيصال: ${json.data?.receiptId ?? ""}`
  1053→            : `Receipt: ${json.data?.receiptId ?? ""}`,
  1054→        });
  1055→      } else {
  1056→        toast.error(json.error?.message ?? "Failed");
  1057→      }
  1058→    } catch {
  1059→      toast.error(locale === "ar" ? "خطأ في الشبكة" : "Network error");
  1060→    }
  1061→  }, [employeeId, locale]);
  1062→
  1063→  const handleSubmitPulse = useCallback(
  1064→    async (score: number) => {
  1065→      if (!data) return;
  1066→      try {
  1067→        const res = await fetch("/api/v1/pulse", {
  1068→          method: "POST",
  1069→          headers: { "Content-Type": "application/json" },
  1070→          body: JSON.stringify({
  1071→            employeeId,
  1072→            stage: data.stage,
  1073→            score,
  1074→            consentId: `CN-pulse-${employeeId}`,
  1075→          }),
  1076→        });
  1077→        const json: ApiEnvelope<{ id: string }> = await res.json();
  1078→        if (json.ok) {
  1079→          toast.success(locale === "ar" ? "تم الإرسال" : "Pulse recorded", {
  1080→            description: locale === "ar" ? "شكراً لمشاركتك." : "Thank you for sharing.",
  1081→          });
  1082→        } else if (json.error?.code === "E_CONSENT_REQUIRED") {
  1083→          setPulseGranted(false);
  1084→          toast.error(locale === "ar" ? "يلزم الموافقة" : "Consent required", {
  1085→            description: json.error.hint ?? "",
  1086→          });
  1087→        } else {
  1088→          toast.error(json.error?.message ?? "Failed");
  1089→        }
  1090→      } catch {
  1091→        toast.error(locale === "ar" ? "خطأ في الشبكة" : "Network error");
  1092→      }
  1093→    },
  1094→    [data, employeeId, locale]
  1095→  );
  1096→
  1097→  /* Render ---------------------------------------------------- */
  1098→
  1099→  return (
  1100→    <div
  1101→      dir={dir}
  1102→      className="min-h-screen flex flex-col"
  1103→      style={{ background: "var(--surface-0)" }}
  1104→    >
  1105→      <main className="flex-1 w-full max-w-[1180px] mx-auto px-4 sm:px-6 lg:px-8 py-6 sm:py-8 space-y-6">
  1106→        {loading ? (
  1107→          <TodaySkeleton />
  1108→        ) : error ? (
  1109→          <ErrorBanner
  1110→            message={error.message}
  1111→            requestId={error.requestId}
  1112→            onRetry={() => void load()}
  1113→          />
  1114→        ) : data ? (
  1115→          <>
  1116→            <HeroStrip data={data} />
  1117→
  1118→            <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
  1119→              <NextBestStepCard data={data} onStart={handleStartStep} />
  1120→              <LateItemsPanel data={data} />
  1121→            </div>
  1122→
  1123→            <StageRail data={data} />
  1124→
  1125→            <ComingUp data={data} />
  1126→
  1127→            <PulseCard
  1128→              employeeId={employeeId}
  1129→              stage={data.stage}
  1130→              pulseGranted={pulseGranted}
  1131→              onGrantConsent={handleGrantConsent}
  1132→              onSubmit={handleSubmitPulse}
  1133→              locale={locale}
  1134→            />
  1135→
  1136→            {/* Version footer */}
  1137→            <p
  1138→              className="text-[11px] font-mono tabular text-center pt-2 pb-4"
  1139→              style={{ color: "var(--text-lo)" }}
  1140→            >
  1141→              {data.engineVersion} · {data.mapVersion} · {data.corpusVersion}
  1142→            </p>
  1143→          </>
  1144→        ) : null}
  1145→      </main>
  1146→    </div>
  1147→  );
  1148→}
