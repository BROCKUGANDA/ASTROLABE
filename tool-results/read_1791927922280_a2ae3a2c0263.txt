   202→   202→/* ---------- Helpers ---------- */
   203→   203→
   204→   204→function latestConsent(records: ConsentRecord[], purpose: Purpose): ConsentRecord | undefined {
   205→   205→  return records
   206→   206→    .filter((r) => r.purpose === purpose)
   207→   207→    .sort((a, b) => b.grantedAt.localeCompare(a.grantedAt))[0];
   208→   208→}
   209→   209→
   210→   210→function purposeGranted(
   211→   211→  records: ConsentRecord[],
   212→   212→  overrides: Record<string, boolean>,
   213→   213→  meta: PurposeMeta
   214→   214→): boolean {
   215→   215→  if (typeof overrides[meta.key] === "boolean") return overrides[meta.key];
   216→   216→  const latest = latestConsent(records, meta.key);
   217→   217→  if (latest) return latest.granted;
   218→   218→  return meta.defaultOn;
   219→   219→}
